   106→// ---------- debounced search hook ----------
   107→function useDebouncedValue<T>(value: T, delay = 350): T {
   108→  const [debounced, setDebounced] = useState(value)
   109→  useEffect(() => {
   110→    const t = setTimeout(() => setDebounced(value), delay)
   111→    return () => clearTimeout(t)
   112→  }, [value, delay])
   113→  return debounced
   114→}

   635→function renderPageButtons(current: number, total: number): (number | '...')[] {
   636→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1)
   637→  const out: (number | '...')[] = [1]
   638→  if (current > 3) out.push('...')
   639→  for (let i = Math.max(2, current - 1); i <= Math.min(total - 1, current + 1); i++) out.push(i)
   640→  if (current < total - 2) out.push('...')
   641→  out.push(total)
   642→  return out
   643→}
