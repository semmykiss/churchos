   999→function DonationFormDialog({
  1000→  mode, donationId, open, onOpenChange, onSaved,
  1001→}: {
  1002→  mode: 'create' | 'edit'
  1003→  donationId?: string
  1004→  open: boolean
  1005→  onOpenChange: (o: boolean) => void
  1006→  onSaved: () => void
  1007→}) {
  1008→  const { organization } = useAuth()
  1009→  const currency = organization?.currency
  1010→  const [donorName, setDonorName] = useState('')
  1011→  const [memberId, setMemberId] = useState('')
  1012→  const [memberSearch, setMemberSearch] = useState('')
  1013→  const [fundId, setFundId] = useState('')
  1014→  const [amount, setAmount] = useState('')
  1015→  const [method, setMethod] = useState<string>('cash')
  1016→  const [reference, setReference] = useState('')
  1017→  const [status, setStatus] = useState<string>('confirmed')
  1018→  const [donatedAt, setDonatedAt] = useState(new Date().toISOString().slice(0, 10))
  1019→  const [saving, setSaving] = useState(false)
  1020→
  1021→  const { data: fundsData } = useQuery<{ items: FundListItem[] }>({
  1022→    queryKey: ['finance', 'funds'],
  1023→    queryFn: async () => {
  1024→      const res = await fetch('/api/finance/funds')
  1025→      if (!res.ok) return { items: [] }
  1026→      return res.json()
  1027→    },
  1028→  })
  1029→
  1030→  // member search
  1031→  const debouncedMemberSearch = useDebouncedValue(memberSearch, 350)
  1032→  const { data: memberResults } = useQuery<{ items: { id: string; firstName: string; lastName: string; memberId?: string | null }[] }>({
  1033→    queryKey: ['members', 'lookup', debouncedMemberSearch],
  1034→    queryFn: async () => {
  1035→      if (!debouncedMemberSearch) return { items: [] }
  1036→      const params = new URLSearchParams({ q: debouncedMemberSearch, page: '1', pageSize: '10' })
  1037→      const res = await fetch(`/api/members?${params}`)
  1038→      if (!res.ok) return { items: [] }
  1039→      return res.json()
  1040→    },
  1041→    enabled: open && !!debouncedMemberSearch,
  1042→  })
  1043→
  1044→  // Load existing donation for edit mode
  1045→  const { data: existing } = useQuery<DonationItem>({
  1046→    queryKey: ['finance', 'donation', donationId],
  1047→    queryFn: async () => {
  1048→      const res = await fetch(`/api/finance/donations/${donationId}`)
  1049→      if (!res.ok) throw new Error('Failed to load')
  1050→      return res.json()
  1051→    },
  1052→    enabled: open && mode === 'edit' && !!donationId,
  1053→  })
  1054→
  1055→  useEffect(() => {
  1056→    if (mode === 'edit' && existing) {
  1057→      setDonorName(existing.donorName ?? '')
  1058→      setMemberId(existing.memberId ?? '')
  1059→      setFundId(existing.fundId ?? '')
  1060→      setAmount(String(existing.amount))
  1061→      setMethod(existing.method)
  1062→      setReference(existing.reference ?? '')
  1063→      setStatus(existing.status)
  1064→      setDonatedAt(new Date(existing.donatedAt).toISOString().slice(0, 10))
  1065→    } else if (mode === 'create') {
  1066→      setDonorName(''); setMemberId(''); setFundId(''); setAmount('')
  1067→      setMethod('cash'); setReference(''); setStatus('confirmed')
  1068→      setDonatedAt(new Date().toISOString().slice(0, 10))
  1069→    }
  1070→  }, [mode, existing])
  1071→
  1072→  const submit = async () => {
  1073→    if (!amount || isNaN(Number(amount)) || Number(amount) <= 0) {
  1074→      toast.error('Enter a valid amount'); return
  1075→    }
  1076→    if (!memberId && !donorName.trim()) {
  1077→      toast.error('Donor name or member is required'); return
  1078→    }
  1079→    setSaving(true)
  1080→    try {
  1081→      const payload: any = {
  1082→        donorName: donorName.trim() || null,
  1083→        memberId: memberId || null,
  1084→        fundId: fundId || null,
  1085→        amount: Number(amount),
  1086→        method,
  1087→        reference: reference.trim() || null,
  1088→        status,
  1089→        donatedAt,
  1090→      }
  1091→      const url = mode === 'create'
  1092→        ? '/api/finance/donations'
  1093→        : `/api/finance/donations/${donationId}`
  1094→      const method_ = mode === 'create' ? 'POST' : 'PATCH'
  1095→      const res = await fetch(url, {
  1096→        method: method_,
  1097→        headers: { 'Content-Type': 'application/json' },
  1098→        body: JSON.stringify(payload),
  1099→      })
  1100→      if (!res.ok) {
  1101→        const e = await res.json().catch(() => ({}))
