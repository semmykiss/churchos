   121→function useDebouncedValue<T>(value: T, delay = 350): T {
   122→  const [debounced, setDebounced] = useState(value)
   123→  useEffect(() => {
   124→    const t = setTimeout(() => setDebounced(value), delay)
   125→    return () => clearTimeout(t)
   126→  }, [value, delay])
   127→  return debounced
   128→}

   130→function initials(name: string): string {
   131→  const parts = name.split(' ').filter(Boolean)
   132→  return ((parts[0]?.[0] ?? '') + (parts[1]?.[0] ?? '')).toUpperCase() || '?'
   133→}

   135→function isOverdue(due: string | null | undefined, status: string): boolean {
   136→  if (!due || status === 'done') return false
   137→  return new Date(due).getTime() < Date.now()
   138→}

   542→        {task.description && (

   733→// ---------- Task Form Dialog (create / edit) ----------
   734→function TaskFormDialog({
   735→  open,
   736→  onOpenChange,
   737→  editId,
   738→  departments,
   739→  assignees,
   740→  onSaved,
   741→}: {
   742→  open: boolean
   743→  onOpenChange: (v: boolean) => void
   744→  editId?: string | null
   745→  departments: DeptRef[]
   746→  assignees: AssigneeOption[]
   747→  onSaved: () => void
   748→}) {
   749→  const isEdit = !!editId
   750→  const [title, setTitle] = useState('')
   751→  const [description, setDescription] = useState('')
   752→  const [departmentId, setDepartmentId] = useState('')
   753→  const [assigneeId, setAssigneeId] = useState('')
   754→  const [priority, setPriority] = useState('normal')
   755→  const [status, setStatus] = useState('todo')
   756→  const [dueDate, setDueDate] = useState('')
   757→
   758→  // Prefill edit
   759→  useEffect(() => {
   760→    if (open && editId) {
   761→      fetch(`/api/tasks/${editId}`)
   762→        .then((r) => r.json())
   763→        .then((d) => {
   764→          if (d && !d.error) {
   765→            setTitle(d.title || '')
   766→            setDescription(d.description || '')
   767→            setDepartmentId(d.departmentId || '')
   768→            setAssigneeId(d.assigneeId || '')
   769→            setPriority(d.priority || 'normal')
   770→            setStatus(d.status || 'todo')
   771→            setDueDate(d.dueDate ? d.dueDate.slice(0, 10) : '')
   772→          }
   773→        })
   774→        .catch(() => {})
   775→    } else if (open && !editId) {
   776→      // Defer reset past the effect body to avoid cascading renders.
   777→      void Promise.resolve().then(() => {
   778→        setTitle(''); setDescription(''); setDepartmentId('')
   779→        setAssigneeId(''); setPriority('normal'); setStatus('todo'); setDueDate('')
   780→      })
   781→    }
   782→  }, [open, editId])
   783→
   784→  const saveMutation = useMutation({
   785→    mutationFn: async () => {
   786→      const payload: Record<string, unknown> = {
   787→        title,
   788→        description: description.trim() || null,
   789→        priority,
   790→        status,
   791→        dueDate: dueDate ? new Date(dueDate).toISOString() : null,
   792→      }
   793→      if (departmentId && departmentId !== '__none') payload.departmentId = departmentId
   794→      else payload.departmentId = null
   795→      if (assigneeId && assigneeId !== '__none') payload.assigneeId = assigneeId
   796→      else payload.assigneeId = null
   797→
   798→      const url = isEdit ? `/api/tasks/${editId}` : '/api/tasks'
   799→      const method = isEdit ? 'PATCH' : 'POST'
   800→      const res = await fetch(url, {
   801→        method,
   802→        headers: { 'Content-Type': 'application/json' },
   803→        body: JSON.stringify(payload),
   804→      })
   805→      if (!res.ok) {
   806→        const e = await res.json().catch(() => ({}))
   807→        throw new Error(e.error || 'Save failed')
   808→      }
   809→      return res.json()
   810→    },
   811→    onSuccess: () => {
   812→      toast.success(isEdit ? 'Task updated' : 'Task created')
   813→      onOpenChange(false)
   814→      onSaved()
   815→    },
   816→    onError: (err: Error) => toast.error(err.message),
   817→  })
   818→
   819→  return (
   820→    <Dialog open={open} onOpenChange={onOpenChange}>
   821→      <DialogContent className="sm:max-w-2xl">
   822→        <DialogHeader>
   823→          <DialogTitle>{isEdit ? 'Edit Task' : 'New Task'}</DialogTitle>
   824→          <DialogDescription>
   825→            {isEdit ? 'Update task details.' : 'Coordinate ministry work with a structured task.'}
   826→          </DialogDescription>
   827→        </DialogHeader>
   828→        <div className="grid gap-4 py-2">
   829→          <div className="grid gap-2">
   830→            <Label htmlFor="title">Title</Label>
   831→            <Input id="title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Prepare Sunday bulletin" />
   832→          </div>
   833→          <div className="grid gap-2">
   834→            <Label htmlFor="description">Description</Label>
   835→            <Textarea
   836→              id="description"
   837→              value={description}
   838→              onChange={(e) => setDescription(e.target.value)}
   839→              rows={3}
   840→              placeholder="Optional context, scope, or notes"
   841→            />
   842→          </div>
   843→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   844→            <div className="grid gap-2">
   845→              <Label>Department</Label>
   846→              <Select value={departmentId || '__none'} onValueChange={(v) => setDepartmentId(v === '__none' ? '' : v)}>
   847→                <SelectTrigger><SelectValue placeholder="No department" /></SelectTrigger>
   848→                <SelectContent>

   863→            <div className="grid gap-2">
   864→              <Label>Assignee</Label>
   865→              <Select value={assigneeId || '__none'} onValueChange={(v) => setAssigneeId(v === '__none' ? '' : v)}>
