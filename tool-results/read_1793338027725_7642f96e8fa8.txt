    12→let s = 0x9e3779b9;
    13→function rnd() {
    14→  s |= 0; s = (s + 0x6d2b79f5) | 0;
    15→  let t = Math.imul(s ^ (s >>> 15), 1 | s);
    16→  t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    17→  return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    18→}
    19→const pick = <T,>(arr: T[]): T => arr[Math.floor(rnd() * arr.length)];

    22→function hashPassword(password: string): string {
    23→  const salt = randomBytes(16).toString("hex");
    24→  const hash = scryptSync(password, salt, 64).toString("hex");
    25→  return `${salt}:${hash}`;
    26→}

    31→function addMonths(dt: Date, n: number) { const x = new Date(dt); x.setUTCMonth(x.getUTCMonth() + n); return x; }
    32→function sundayBefore(dt: Date): Date { const x = new Date(dt); x.setUTCDate(x.getUTCDate() - x.getUTCDay()); return x; }

    34→function age(bd: Date, at = new Date()) {
    35→  let a = at.getUTCFullYear() - bd.getUTCFullYear();
    36→  const m = at.getUTCMonth() - bd.getUTCMonth();
    37→  if (m < 0 || (m === 0 && at.getUTCDate() < bd.getUTCDate())) a--;
    38→  return a;
    39→}
    40→function ageGroup(bd: Date | null, at = new Date()): string {
    41→  if (!bd) return "adults";

    44→  if (a < 6) return "preschool";
    45→  if (a < 12) return "kids";
    46→  if (a < 18) return "students";
    47→  return "adults";

   898→  function pickN<T,>(arr: T[], n: number): T[] {
   899→    const copy = [...arr]; const out: T[] = [];
   900→    while (out.length < n && copy.length) out.push(copy.splice(Math.floor(rnd() * copy.length), 1)[0]);
   901→    return out;
   902→  }
