   392→// ---------- manage dialog ----------
   393→function ManageDepartmentDialog({
   394→  id, open, onOpenChange, canEdit, onEdit,
   395→}: {
   396→  id: string
   397→  open: boolean
   398→  onOpenChange: (o: boolean) => void
   399→  canEdit: boolean
   400→  onEdit: () => void
   401→}) {
   402→  const queryClient = useQueryClient()
   403→  const [addMemberOpen, setAddMemberOpen] = useState(false)
   404→  const [newMemberId, setNewMemberId] = useState('')
   405→  const [newMemberRole, setNewMemberRole] = useState<string>('member')
   406→  const [memberSearch, setMemberSearch] = useState('')
   407→
   408→  const { data, isLoading, isError } = useQuery<DeptDetail>({
   409→    queryKey: ['department', id],
   410→    queryFn: async () => {
   411→      const res = await fetch(`/api/departments/${id}`)
   412→      if (!res.ok) {
   413→        const e = await res.json().catch(() => ({}))
   414→        throw new Error(e.error || 'Failed to load department')
   415→      }
   416→      return res.json()
   417→    },
   418→    enabled: open,
   419→  })
   420→
   421→  // Search members to add (only when canEdit)
   422→  const memberSearchDebounced = useDebounced(memberSearch, 350)
   423→  const { data: memberResults } = useQuery<{ items: { id: string; firstName: string; lastName: string; memberId?: string | null }[] }>({
   424→    queryKey: ['members', 'list', memberSearchDebounced, '', 1, 20],
   425→    queryFn: async () => {
   426→      const params = new URLSearchParams({ page: '1', pageSize: '20' })
   427→      if (memberSearchDebounced) params.set('q', memberSearchDebounced)
   428→      const res = await fetch(`/api/members?${params}`)
   429→      if (!res.ok) return { items: [] }
   430→      return res.json()
   431→    },
   432→    enabled: addMemberOpen && !!memberSearchDebounced,
   433→  })
   434→
   435→  const addMutation = useMutation({
   436→    mutationFn: async ({ memberId, role }: { memberId: string; role: string }) => {
   437→      const res = await fetch(`/api/departments/${id}`, {
   438→        method: 'PATCH',
   439→        headers: { 'Content-Type': 'application/json' },
   440→        body: JSON.stringify({ addMemberId: memberId, memberRole: role }),
   441→      })
   442→      if (!res.ok) {
   443→        const e = await res.json().catch(() => ({}))
   444→        throw new Error(e.error || 'Failed to add member')
   445→      }
   446→      return res.json()
   447→    },
   448→    onSuccess: () => {
   449→      toast.success('Member added to department')
   450→      queryClient.invalidateQueries({ queryKey: ['department', id] })
   451→      queryClient.invalidateQueries({ queryKey: ['departments'] })
   452→      setNewMemberId('')
   453→      setNewMemberRole('member')
   454→      setAddMemberOpen(false)
   455→    },
   456→    onError: (e: Error) => toast.error(e.message),
   457→  })
   458→
   459→  const removeMutation = useMutation({
   460→    mutationFn: async (memberId: string) => {
   461→      const res = await fetch(`/api/departments/${id}`, {
   462→        method: 'PATCH',
   463→        headers: { 'Content-Type': 'application/json' },
   464→        body: JSON.stringify({ removeMemberId: memberId }),
   465→      })
   466→      if (!res.ok) {
   467→        const e = await res.json().catch(() => ({}))
   468→        throw new Error(e.error || 'Failed to remove member')
   469→      }
   470→      return res.json()
   471→    },
   472→    onSuccess: () => {
   473→      toast.success('Member removed')
   474→      queryClient.invalidateQueries({ queryKey: ['department', id] })
   475→      queryClient.invalidateQueries({ queryKey: ['departments'] })
   476→    },
   477→    onError: (e: Error) => toast.error(e.message),
   478→  })
   479→
   480→  return (
   481→    <Dialog open={open} onOpenChange={onOpenChange}>
   482→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto scroll-thin">
   483→        {isLoading ? (
   484→          <div className="space-y-3 py-4">
   485→            <Skeleton className="h-7 w-1/2" />
   486→            <Skeleton className="h-32 w-full" />
   487→            <Skeleton className="h-40 w-full" />
   488→          </div>
   489→        ) : isError ? (
   490→          <div className="py-10 text-center">
   491→            <p className="text-sm font-medium">Failed to load department</p>
   492→          </div>
   493→        ) : data ? (
   494→          <>
   495→            <DialogHeader>
   496→              <div className="flex items-start gap-3">
   497→                <span
   498→                  className="size-10 rounded-xl grid place-items-center shrink-0 ring-1 ring-inset ring-black/5 dark:ring-white/10"
   499→                  style={{ background: `${data.color ?? '#6366f1'}1a`, color: data.color ?? '#6366f1' }}
   500→                >
   501→                  <Boxes className="size-5" />
   502→                </span>
   503→                <div className="min-w-0 flex-1">
   504→                  <DialogTitle className="text-xl truncate">{data.name}</DialogTitle>
   505→                  <DialogDescription className="mt-0.5">
   506→                    {data.description || 'No description provided.'}
   507→                  </DialogDescription>
   508→                </div>
   509→                {canEdit && (
   510→                  <Button variant="outline" size="sm" onClick={onEdit} className="gap-2">
   511→                    <Pencil className="size-3.5" /> Edit
   512→                  </Button>
   513→                )}
   514→              </div>
   515→            </DialogHeader>
   516→
   517→            {/* Stats */}
   518→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   519→              <MiniStat icon={Users} value={data.counts.members} label="Members" />
   520→              <MiniStat icon={Boxes} value={data.counts.units} label="Units" />
   521→              <MiniStat icon={ListChecks} value={data.counts.tasks} label="Tasks" />
   522→              <MiniStat icon={CalendarDays} value={data.counts.events} label="Events" />
   523→            </div>
   524→
   525→            {/* Head card */}
   526→            {data.head && (
   527→              <div className="rounded-lg border border-amber-500/30 bg-amber-500/5 p-3 flex items-center gap-3">
   528→                <Avatar className="size-10">
   529→                  {data.head.avatarUrl && <AvatarImage src={data.head.avatarUrl} alt={data.head.name} />}
   530→                  <AvatarFallback className="text-xs bg-amber-500/15 text-amber-700 dark:text-amber-300">
   531→                    {initials(data.head.name)}
   532→                  </AvatarFallback>

   966→                  ))}

   980→          </Button>
   981→        </DialogFooter>
   982→      </DialogContent>
   983→    </Dialog>

   987→// ---------- small debounce util ----------
   988→function useDebounced<T>(value: T, delay = 350): T {
   989→  const [v, setV] = useState(value)
   990→  useEffect(() => {
   991→    const t = setTimeout(() => setV(value), delay)
   992→    return () => clearTimeout(t)
   993→  }, [value, delay])
   994→  return v
   995→}
   996→
