   139→function useDebouncedValue<T>(value: T, delay = 350): T {
   140→  const [debounced, setDebounced] = useState(value)
   141→  useEffect(() => {
   142→    const t = setTimeout(() => setDebounced(value), delay)
   143→    return () => clearTimeout(t)
   144→  }, [value, delay])
   145→  return debounced
   146→}

   423→function RolesTab({
   424→  canManage, goMatrix, goUsers,
   425→}: {
   426→  canManage: boolean
   427→  goMatrix: (roleId: string) => void
   428→  goUsers: (roleId: string) => void
   429→}) {
   430→  const queryClient = useQueryClient()
   431→  const [search, setSearch] = useState('')
   432→  const [filter, setFilter] = useState<'all' | 'system' | 'custom' | 'active' | 'inactive'>('all')
   433→  const [createOpen, setCreateOpen] = useState(false)
   434→  const [editRole, setEditRole] = useState<RoleListItem | null>(null)
   435→  const [cloneRole, setCloneRole] = useState<RoleListItem | null>(null)
   436→  const [deleteRole, setDeleteRole] = useState<RoleListItem | null>(null)
   437→
   438→  const { data, isLoading, isError, refetch } = useQuery<RoleListResponse>({
   439→    queryKey: ['access', 'roles'] as const,
   440→    queryFn: async () => {
   441→      const res = await fetch('/api/access/roles')
   442→      if (!res.ok) {
   443→        const e = await res.json().catch(() => ({}))
   444→        throw new Error(e.error || 'Failed to load roles')
   445→      }
   446→      return res.json()
   447→    },
   448→  })
   449→
   450→  // Stats: also need active users count + total grants. We piggy-back on the
   451→  // legacy /api/access endpoint for totalUsers + totalGrants since it already
   452→  // aggregates them server-side.
   453→  const { data: legacy } = useQuery<AccessMatrixResponse>({
   454→    queryKey: ['access', 'matrix-legacy'] as const,
   455→    queryFn: async () => {
   456→      const res = await fetch('/api/access')
   457→      if (!res.ok) throw new Error('Failed to load matrix')
   458→      return res.json()
   459→    },
   460→    enabled: !isLoading,
   461→  })
   462→
   463→  const roles = data?.roles ?? []
   464→  const filtered = useMemo(() => {
   465→    let out = roles
   466→    if (filter === 'system') out = out.filter((r) => r.isSystem)
   467→    else if (filter === 'custom') out = out.filter((r) => !r.isSystem)
   468→    else if (filter === 'active') out = out.filter((r) => r.active)
   469→    else if (filter === 'inactive') out = out.filter((r) => !r.active)
   470→    if (search.trim()) {
   471→      const q = search.toLowerCase()
   472→      out = out.filter((r) =>
   473→        r.name.toLowerCase().includes(q) ||
   474→        r.label.toLowerCase().includes(q) ||
   475→        (r.description ?? '').toLowerCase().includes(q),
   476→      )
   477→    }
   478→    return out
   479→  }, [roles, filter, search])
   480→
   481→  const totalGrants = useMemo(() => {
   482→    if (!legacy) return 0
   483→    let s = 0
   484→    for (const r of legacy.roles) for (const m of legacy.modules) s += legacy.matrix[r]?.[m]?.length ?? 0
   485→    return s
   486→  }, [legacy])
   487→
   488→  // Mutations
   489→  const retireMutation = useMutation({
   490→    mutationFn: async ({ role, active }: { role: RoleListItem; active: boolean }) => {
   491→      const res = await fetch(`/api/access/roles/${role.id}`, {
   492→        method: 'PATCH',
   493→        headers: { 'Content-Type': 'application/json' },
   494→        body: JSON.stringify({ active }),
   495→      })
   496→      if (!res.ok) {
   497→        const e = await res.json().catch(() => ({}))
   498→        throw new Error(e.error || `Failed to ${active ? 'activate' : 'retire'} role`)
   499→      }
   500→      return res.json()
   501→    },
   502→    onSuccess: (_d, { role, active }) => {
   503→      toast.success(`${role.label} ${active ? 'activated' : 'retired'}`)
   504→      void queryClient.invalidateQueries({ queryKey: ['access', 'roles'] })
   505→      void queryClient.invalidateQueries({ queryKey: ['access'] })
   506→    },
   507→    onError: (e: Error) => toast.error(e.message),
   508→  })
