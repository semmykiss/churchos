   422→// ---------- calendar view ----------
   423→function CalendarView({ items, type, loading }: { items: CelebrationItem[]; type: string; loading: boolean }) {
   424→  const today = new Date()
   425→  const [viewMonth, setViewMonth] = useState(today.getMonth())
   426→  const [viewYear, setViewYear] = useState(today.getFullYear())
   427→
   428→  // Build calendar days
   429→  const firstDay = new Date(viewYear, viewMonth, 1)
   430→  const lastDay = new Date(viewYear, viewMonth + 1, 0)
   431→  const startWeekday = (firstDay.getDay() + 6) % 7 // Monday-first
   432→  const totalDays = lastDay.getDate()
   433→  const cells: (number | null)[] = []
   434→  for (let i = 0; i < startWeekday; i++) cells.push(null)
   435→  for (let d = 1; d <= totalDays; d++) cells.push(d)
   436→  while (cells.length % 7 !== 0) cells.push(null)
   437→
   438→  // Index items by day-of-month for current view
   439→  const byDay = new Map<number, CelebrationItem[]>()
   440→  for (const it of items) {
   441→    if (it.month === viewMonth + 1) {
   442→      const list = byDay.get(it.day) ?? []
   443→      list.push(it)
   444→      byDay.set(it.day, list)
   445→    }
   446→  }
   447→
   448→  const goPrev = () => {
   449→    if (viewMonth === 0) { setViewMonth(11); setViewYear(viewYear - 1) }
   450→    else setViewMonth(viewMonth - 1)
   451→  }
   452→  const goNext = () => {
   453→    if (viewMonth === 11) { setViewMonth(0); setViewYear(viewYear + 1) }
   454→    else setViewMonth(viewMonth + 1)
   455→  }
   456→  const goToday = () => { setViewMonth(today.getMonth()); setViewYear(today.getFullYear()) }
   457→
   458→  return (
   459→    <Card className="border-border/60">
   460→      <CardContent className="p-5">
   461→        <div className="flex items-center justify-between mb-4">
   462→          <div>
   463→            <p className="text-sm font-semibold">{MONTHS[viewMonth]} {viewYear}</p>
   464→            <p className="text-[11px] text-muted-foreground">{items.length} {type} this view</p>
   465→          </div>
   466→          <div className="flex items-center gap-1.5">
   467→            <Button variant="outline" size="sm" onClick={goPrev} className="h-8">Prev</Button>
   468→            <Button variant="ghost" size="sm" onClick={goToday} className="h-8">Today</Button>
   469→            <Button variant="outline" size="sm" onClick={goNext} className="h-8">Next</Button>
   470→          </div>
   471→        </div>
   472→        <div className="grid grid-cols-7 gap-1 text-center mb-2">
   473→          {['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'].map((d) => (
   474→            <div key={d} className="text-[10px] uppercase font-medium text-muted-foreground py-1">{d}</div>
   475→          ))}
   476→        </div>
   477→        {loading ? (
   478→          <div className="grid grid-cols-7 gap-1">
   479→            {Array.from({ length: 35 }).map((_, i) => <Skeleton key={i} className="aspect-square rounded-md" />)}
   480→          </div>
   481→        ) : (
   482→          <div className="grid grid-cols-7 gap-1">
   483→            {cells.map((day, i) => {
   484→              if (day === null) return <div key={i} className="aspect-square rounded-md bg-muted/20" />
   485→              const dayItems = byDay.get(day) ?? []
   486→              const isToday = day === today.getDate() && viewMonth === today.getMonth() && viewYear === today.getFullYear()
   487→              return (
   488→                <div
   489→                  key={i}
   490→                  className={cn(
   491→                    'aspect-square rounded-md border border-border/40 p-1 flex flex-col items-center justify-start gap-0.5 relative',
   492→                    isToday ? 'bg-primary/8 border-primary/40' : 'bg-card',
   493→                    dayItems.length > 0 && 'hover:shadow-premium transition-shadow',
   494→                  )}
   495→                >
   496→                  <span className={cn('text-[10px] font-semibold nums', isToday ? 'text-primary' : 'text-muted-foreground')}>{day}</span>
   497→                  <div className="flex flex-wrap gap-0.5 justify-center">
   498→                    {dayItems.slice(0, 4).map((it) => {
   499→                      const isBirthday = type === 'birthdays'
   500→                      return (
   501→                        <span
   502→                          key={it.id}
   503→                          title={`${it.name} · ${isBirthday ? 'Birthday' : 'Anniversary'}`}
   504→                          className={cn(
   505→                            'inline-block size-1.5 rounded-full',
   506→                            isBirthday ? 'bg-amber-500' : 'bg-rose-500',
   507→                          )}
   508→                        />
   509→                      )
   510→                    })}
   511→                    {dayItems.length > 4 && (
   512→                      <span className="text-[8px] text-muted-foreground nums">+{dayItems.length - 4}</span>
   513→                    )}
   514→                  </div>
   515→                </div>
   516→              )
   517→            })}
   518→          </div>
   519→        )}
   520→        <div className="flex items-center gap-4 mt-4 text-[11px]">
   521→          <span className="flex items-center gap-1.5"><Cake className="size-3 text-amber-500" /> Birthdays</span>
   522→          <span className="flex items-center gap-1.5"><Heart className="size-3 text-rose-500" /> Anniversaries</span>
   523→        </div>
   524→      </CardContent>
   525→    </Card>
   526→  )

   530→function GreetDialog({ target, onClose }: { target: CelebrationItem | null; onClose: () => void }) {
   531→  const [message, setMessage] = useState('')
   532→  const greetMut = useMutation({
   533→    mutationFn: async (vars: { memberId: string; type: string; message?: string }) => {
   534→      const res = await fetch('/api/celebrations/greet', {
   535→        method: 'POST',
   536→        headers: { 'Content-Type': 'application/json' },
   537→        body: JSON.stringify(vars),
   538→      })
   539→      if (!res.ok) {
   540→        const data = await res.json().catch(() => ({}))
   541→        throw new Error(data.error || 'Failed to send greeting')
   542→      }
   543→      return res.json()
   544→    },
   545→    onSuccess: (_data, vars) => {
   546→      toast.success(`Greeting sent to ${target?.name ?? 'celebrant'}`)
   547→      setMessage('')
   548→      onClose()
   549→    },
   550→    onError: (e: Error) => toast.error(e.message),
   551→  })
   552→
   553→  return (
   554→    <Dialog open={!!target} onOpenChange={(o) => !o && onClose()}>
   555→      <DialogContent className="sm:max-w-[480px]">
   556→        <DialogHeader>
   557→          <DialogTitle className="flex items-center gap-2">
   558→            <Heart className="size-4 text-rose-500" /> Send a greeting
   559→          </DialogTitle>
   560→          <DialogDescription>
   561→            {target ? `Send a personal greeting to ${target.name}. They'll receive an in-app notification.` : ''}
   562→          </DialogDescription>
   563→        </DialogHeader>
   564→        <div className="space-y-3 py-2">
   565→          {target && (
   566→            <div className="flex items-center gap-3 rounded-xl border border-border/60 bg-muted/30 p-3">
   567→              <Avatar className="size-10">
   568→                <AvatarImage src={target.photoUrl ?? undefined} alt={target.firstName} />

   570→              </Avatar>
   571→              <div>
   572→                <p className="text-sm font-medium">{target.name}</p>
   573→                <p className="text-[11px] text-muted-foreground">
   574→                  {target.day} {MONTHS[target.month - 1]} · in {target.daysUntil}d
   575→                </p>
   576→              </div>
   577→            </div>

   579→          <div className="space-y-1.5">

   581→            <Textarea

   587→            />

   589→              Leave blank to send the default greeting.

   599→                memberId: target.id,

   605→            className="gap-2"
   606→          >

   608→          </Button>
   609→        </DialogFooter>
   610→      </DialogContent>
   611→    </Dialog>
   612→  )
