   392→   392→// ---------- manage dialog ----------
   393→   393→function ManageDepartmentDialog({
   394→   394→  id, open, onOpenChange, canEdit, onEdit,
   395→   395→}: {
   396→   396→  id: string
   397→   397→  open: boolean
   398→   398→  onOpenChange: (o: boolean) => void
   399→   399→  canEdit: boolean
   400→   400→  onEdit: () => void
   401→   401→}) {
   402→   402→  const queryClient = useQueryClient()
   403→   403→  const [addMemberOpen, setAddMemberOpen] = useState(false)
   404→   404→  const [newMemberId, setNewMemberId] = useState('')
   405→   405→  const [newMemberRole, setNewMemberRole] = useState<string>('member')
   406→   406→  const [memberSearch, setMemberSearch] = useState('')
   407→   407→
   408→   408→  const { data, isLoading, isError } = useQuery<DeptDetail>({
   409→   409→    queryKey: ['department', id],
   410→   410→    queryFn: async () => {
   411→   411→      const res = await fetch(`/api/departments/${id}`)
   412→   412→      if (!res.ok) {
   413→   413→        const e = await res.json().catch(() => ({}))
   414→   414→        throw new Error(e.error || 'Failed to load department')
   415→   415→      }
   416→   416→      return res.json()
   417→   417→    },
   418→   418→    enabled: open,
   419→   419→  })
   420→   420→
   421→   421→  // Search members to add (only when canEdit)
   422→   422→  const memberSearchDebounced = useDebounced(memberSearch, 350)
   423→   423→  const { data: memberResults } = useQuery<{ items: { id: string; firstName: string; lastName: string; memberId?: string | null }[] }>({
   424→   424→    queryKey: ['members', 'list', memberSearchDebounced, '', 1, 20],
   425→   425→    queryFn: async () => {
   426→   426→      const params = new URLSearchParams({ page: '1', pageSize: '20' })
   427→   427→      if (memberSearchDebounced) params.set('q', memberSearchDebounced)
   428→   428→      const res = await fetch(`/api/members?${params}`)
   429→   429→      if (!res.ok) return { items: [] }
   430→   430→      return res.json()
   431→   431→    },
   432→   432→    enabled: addMemberOpen && !!memberSearchDebounced,
   433→   433→  })
   434→   434→
   435→   435→  const addMutation = useMutation({
   436→   436→    mutationFn: async ({ memberId, role }: { memberId: string; role: string }) => {
   437→   437→      const res = await fetch(`/api/departments/${id}`, {
   438→   438→        method: 'PATCH',
   439→   439→        headers: { 'Content-Type': 'application/json' },
   440→   440→        body: JSON.stringify({ addMemberId: memberId, memberRole: role }),
   441→   441→      })
   442→   442→      if (!res.ok) {
   443→   443→        const e = await res.json().catch(() => ({}))
   444→   444→        throw new Error(e.error || 'Failed to add member')
   445→   445→      }
   446→   446→      return res.json()
   447→   447→    },
   448→   448→    onSuccess: () => {
   449→   449→      toast.success('Member added to department')
   450→   450→      queryClient.invalidateQueries({ queryKey: ['department', id] })

   987→   987→// ---------- small debounce util ----------
   988→   988→function useDebounced<T>(value: T, delay = 350): T {
   989→   989→  const [v, setV] = useState(value)
   990→   990→  useEffect(() => {
   991→   991→    const t = setTimeout(() => setV(value), delay)
   992→   992→    return () => clearTimeout(t)
   993→   993→  }, [value, delay])
   994→   994→  return v
   995→   995→}
