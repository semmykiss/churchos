   395→// ---------- manage dialog ----------
   396→function ManageDepartmentDialog({
   397→  id, open, onOpenChange, canEdit, onEdit,
   398→}: {
   399→  id: string
   400→  open: boolean
   401→  onOpenChange: (o: boolean) => void
   402→  canEdit: boolean
   403→  onEdit: () => void
   404→}) {
   405→  const queryClient = useQueryClient()
   406→  const [addMemberOpen, setAddMemberOpen] = useState(false)
   407→  const [newMemberId, setNewMemberId] = useState('')
   408→  const [newMemberRole, setNewMemberRole] = useState<string>('member')
   409→  const [memberSearch, setMemberSearch] = useState('')
   410→
   411→  const { data, isLoading, isError } = useQuery<DeptDetail>({
   412→    queryKey: ['department', id],
   413→    queryFn: async () => {
   414→      const res = await fetch(`/api/departments/${id}`)
   415→      if (!res.ok) {
   416→        const e = await res.json().catch(() => ({}))
   417→        throw new Error(e.error || 'Failed to load department')
   418→      }
   419→      return res.json()
   420→    },
   421→    enabled: open,
   422→  })
   423→
   424→  // Search members to add (only when canEdit)
   425→  const memberSearchDebounced = useDebounced(memberSearch, 350)
   426→  const { data: memberResults } = useQuery<{ items: { id: string; firstName: string; lastName: string; memberId?: string | null }[] }>({
   427→    queryKey: ['members', 'list', memberSearchDebounced, '', 1, 20],
   428→    queryFn: async () => {
   429→      const params = new URLSearchParams({ page: '1', pageSize: '20' })
   430→      if (memberSearchDebounced) params.set('q', memberSearchDebounced)
   431→      const res = await fetch(`/api/members?${params}`)
   432→      if (!res.ok) return { items: [] }
   433→      return res.json()
   434→    },
   435→    enabled: addMemberOpen && !!memberSearchDebounced,
   436→  })
   437→
   438→  const addMutation = useMutation({
   439→    mutationFn: async ({ memberId, role }: { memberId: string; role: string }) => {
   440→      const res = await fetch(`/api/departments/${id}`, {
   441→        method: 'PATCH',
   442→        headers: { 'Content-Type': 'application/json' },
   443→        body: JSON.stringify({ addMemberId: memberId, memberRole: role }),
   444→      })
   445→      if (!res.ok) {
   446→        const e = await res.json().catch(() => ({}))
   447→        throw new Error(e.error || 'Failed to add member')
   448→      }
   449→      return res.json()
   450→    },
   451→    onSuccess: () => {
   452→      toast.success('Member added to department')
   453→      queryClient.invalidateQueries({ queryKey: ['department', id] })
   454→      queryClient.invalidateQueries({ queryKey: ['departments'] })
   455→      setNewMemberId('')
   456→      setNewMemberRole('member')
   457→      setAddMemberOpen(false)
   458→    },
   459→    onError: (e: Error) => toast.error(e.message),
   460→  })
   461→
   462→  const removeMutation = useMutation({
   463→    mutationFn: async (memberId: string) => {
   464→      const res = await fetch(`/api/departments/${id}`, {
   465→        method: 'PATCH',
   466→        headers: { 'Content-Type': 'application/json' },
   467→        body: JSON.stringify({ removeMemberId: memberId }),
   468→      })
   469→      if (!res.ok) {
   470→        const e = await res.json().catch(() => ({}))
   471→        throw new Error(e.error || 'Failed to remove member')
   472→      }
   473→      return res.json()
   474→    },
   475→    onSuccess: () => {
   476→      toast.success('Member removed')
   477→      queryClient.invalidateQueries({ queryKey: ['department', id] })
   478→      queryClient.invalidateQueries({ queryKey: ['departments'] })
   479→    },
   480→    onError: (e: Error) => toast.error(e.message),
   481→  })
   482→
   483→  return (
   484→    <Dialog open={open} onOpenChange={onOpenChange}>
   485→      <DialogContent className="sm:max-w-3xl max-h-[90vh] overflow-y-auto scroll-thin">
   486→        {isLoading ? (
   487→          <div className="space-y-3 py-4">
   488→            <Skeleton className="h-7 w-1/2" />
   489→            <Skeleton className="h-32 w-full" />
   490→            <Skeleton className="h-40 w-full" />
   491→          </div>
   492→        ) : isError ? (
   493→          <div className="py-10 text-center">
   494→            <p className="text-sm font-medium">Failed to load department</p>
   495→          </div>
   496→        ) : data ? (
   497→          <>
   498→            <DialogHeader>
   499→              <div className="flex items-start gap-3">
   500→                <span
   501→                  className="size-10 rounded-lg grid place-items-center shrink-0"
   502→                  style={{ background: `${data.color ?? '#0d9488'}1a`, color: data.color ?? '#0d9488' }}
   503→                >
   504→                  <Boxes className="size-5" />
   505→                </span>
   506→                <div className="min-w-0 flex-1">
   507→                  <DialogTitle className="text-xl truncate">{data.name}</DialogTitle>
   508→                  <DialogDescription className="mt-0.5">
   509→                    {data.description || 'No description provided.'}
   510→                  </DialogDescription>
   511→                </div>
   512→                {canEdit && (
   513→                  <Button variant="outline" size="sm" onClick={onEdit}>
   514→                    <Pencil className="size-3.5 mr-1.5" /> Edit
   515→                  </Button>
   516→                )}
   517→              </div>
   518→            </DialogHeader>
   519→
   520→            {/* Stats */}
   521→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   522→              <MiniStat icon={Users} value={data.counts.members} label="Members" />
   523→              <MiniStat icon={Boxes} value={data.counts.units} label="Units" />
   524→              <MiniStat icon={ListChecks} value={data.counts.tasks} label="Tasks" />
   525→              <MiniStat icon={CalendarDays} value={data.counts.events} label="Events" />
   526→            </div>
   527→
   528→            {/* Head card */}
   529→            {data.head && (
   530→              <div className="rounded-lg border border-amber-500/30 bg-amber-500/5 p-3 flex items-center gap-3">

   723→// ---------- form dialog (create / edit) ----------
   724→interface DeptFormState {
   725→  name: string
   726→  description: string
   727→  color: string
   728→  headUserId: string
   729→}
   730→const EMPTY_DEPT: DeptFormState = { name: '', description: '', color: PRESET_COLORS[0], headUserId: '' }
   731→
   732→function DeptFormDialog({
   733→  mode, departmentId, open, onOpenChange, onSaved,
   734→}: {
   735→  mode: 'create' | 'edit'
   736→  departmentId?: string
   737→  open: boolean
   738→  onOpenChange: (o: boolean) => void
   739→  onSaved: () => void
   740→}) {
   741→  const [form, setForm] = useState<DeptFormState>(EMPTY_DEPT)
   742→  const [saving, setSaving] = useState(false)
   743→  const [headCandidates, setHeadCandidates] = useState<{ id: string; name: string; email: string }[]>([])
   744→
   745→  // Load department heads candidates — fetch users via /api/me? No endpoint. Use known seed users.
   746→  useEffect(() => {
   747→    if (!open) return
   748→    // Provide a small list of known staff candidates (department_head role users)
   749→    setHeadCandidates([
   750→      { id: 'user-worship-head', name: 'Michael Bello', email: 'worship@gracechapel.org' },
   751→      { id: 'user-media-head', name: 'Tunde Lawal', email: 'media@gracechapel.org' },
   752→      { id: 'user-worship-asst', name: 'Sarah Obi', email: 'assistant@gracechapel.org' },
   753→      { id: 'user-pastor', name: 'Pastor Samuel Adeyemi', email: 'pastor@gracechapel.org' },
   754→      { id: 'user-admin', name: 'Daniel Okafor', email: 'admin@gracechapel.org' },
   755→    ])
   756→  }, [open])
   757→
   758→  useEffect(() => {
   759→    if (!open) return
   760→    if (mode === 'edit' && departmentId) {
   761→      ;(async () => {
   762→        try {
   763→          const res = await fetch(`/api/departments/${departmentId}`)
   764→          if (!res.ok) throw new Error('Failed')
   765→          const d: DeptDetail = await res.json()
   766→          setForm({
   767→            name: d.name,
   768→            description: d.description ?? '',
   769→            color: d.color ?? PRESET_COLORS[0],
   770→            headUserId: d.head?.id ?? '',
   771→          })
   772→        } catch {
   773→          toast.error('Failed to load department for editing')
   774→          onOpenChange(false)
   775→        }
   776→      })()
   777→    } else {
   778→      setForm(EMPTY_DEPT)
   779→    }
   780→  }, [mode, departmentId, open, onOpenChange])
   781→
   782→  const submit = async () => {
   783→    if (!form.name.trim()) {
   784→      toast.error('Department name is required')
   785→      return
   786→    }
   787→    setSaving(true)
   788→    try {
   789→      const payload: Record<string, unknown> = {
   790→        name: form.name.trim(),
   791→        description: form.description.trim() || null,
   792→        color: form.color,
   793→        headUserId: form.headUserId || null,
   794→      }
   795→      const url = mode === 'create' ? '/api/departments' : `/api/departments/${departmentId}`
   796→      const method = mode === 'create' ? 'POST' : 'PATCH'
   797→      const res = await fetch(url, {
   798→        method,
   799→        headers: { 'Content-Type': 'application/json' },
   800→        body: JSON.stringify(payload),
   801→      })
   802→      if (!res.ok) {
   803→        const e = await res.json().catch(() => ({}))
   804→        throw new Error(e.error || `Failed to ${mode} department`)
   805→      }
   806→      toast.success(mode === 'create' ? 'Department created' : 'Department updated')
   807→      onSaved()
   808→    } catch (e: unknown) {
   809→      const msg = e instanceof Error ? e.message : 'Save failed'
   810→      toast.error(msg)
   811→    } finally {
   812→      setSaving(false)
   813→    }
   814→  }
   815→
   816→  return (
   817→    <Dialog open={open} onOpenChange={onOpenChange}>
   818→      <DialogContent className="sm:max-w-lg">
   819→        <DialogHeader>
   820→          <DialogTitle>{mode === 'create' ? 'Add Department' : 'Edit Department'}</DialogTitle>
   821→          <DialogDescription>
   822→            {mode === 'create'
   823→              ? 'Create a new ministry department or unit.'
   824→              : 'Update department details. Changes will be audited.'}
   825→          </DialogDescription>
   826→        </DialogHeader>
   827→
   828→        <div className="space-y-4">
   829→          <div className="space-y-1.5">
   830→            <Label className="text-xs font-medium">Name *</Label>
   831→            <Input value={form.name} onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))} placeholder="e.g. Worship & Music" />
   832→          </div>
   833→
   834→          <div className="space-y-1.5">
   835→            <Label className="text-xs font-medium">Description</Label>
   836→            <Textarea rows={2} value={form.description} onChange={(e) => setForm((f) => ({ ...f, description: e.target.value }))} placeholder="Short description of the department's mission" />
   837→          </div>
   838→
   839→          <div className="space-y-1.5">
   840→            <Label className="text-xs font-medium">Color</Label>
   841→            <div className="flex flex-wrap gap-2">
   842→              {PRESET_COLORS.map((c) => (
   843→                <button
   844→                  key={c}
   845→                  type="button"
   846→                  onClick={() => setForm((f) => ({ ...f, color: c }))}
   847→                  className={`size-7 rounded-full border-2 transition-transform ${form.color === c ? 'scale-110 border-foreground' : 'border-transparent'}`}
   848→                  style={{ background: c }}
   849→                  aria-label={`Color ${c}`}
   850→                />
   851→              ))}
   852→            </div>
   853→          </div>
   854→
   855→          <div className="space-y-1.5">
   856→            <Label className="text-xs font-medium">Department Head</Label>
   857→            <Select value={form.headUserId || 'none'} onValueChange={(v) => setForm((f) => ({ ...f, headUserId: v === 'none' ? '' : v }))}>
   858→              <SelectTrigger className="w-full"><SelectValue placeholder="Select a head (optional)" /></SelectTrigger>
   859→              <SelectContent>
   860→                <SelectItem value="none">— None —</SelectItem>
   861→                {headCandidates.map((u) => (
   862→                  <SelectItem key={u.id} value={u.id}>{u.name} ({u.email})</SelectItem>
   863→                ))}
   864→              </SelectContent>
   865→            </Select>
   866→          </div>
   867→        </div>
   868→
   869→        <DialogFooter>
   870→          <DialogClose asChild>
   871→            <Button variant="outline" disabled={saving}>Cancel</Button>
   872→          </DialogClose>
   873→          <Button onClick={submit} disabled={saving}>
   874→            {saving ? 'Saving…' : mode === 'create' ? 'Create' : 'Save changes'}
   875→            {!saving && <ArrowRight className="size-3.5 ml-1.5" />}
   876→          </Button>
   877→        </DialogFooter>
   878→      </DialogContent>
   879→    </Dialog>
   880→  )
   881→}

   883→// ---------- small debounce util ----------
   884→function useDebounced<T>(value: T, delay = 350): T {
   885→  const [v, setV] = useState(value)
   886→  useEffect(() => {
   887→    const t = setTimeout(() => setV(value), delay)
   888→    return () => clearTimeout(t)
   889→  }, [value, delay])
   890→  return v
   891→}
