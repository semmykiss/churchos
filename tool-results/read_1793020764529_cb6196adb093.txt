   102→function FileIcon({ filename, className }: { filename?: string | null; className?: string }) {
   103→  const ext = (filename ?? '').split('.').pop()?.toLowerCase() ?? ''
   104→  if (['jpg', 'jpeg', 'png', 'gif'].includes(ext)) return <FileImage className={className} />
   105→  if (['xlsx', 'xls', 'csv'].includes(ext)) return <FileSpreadsheet className={className} />
   106→  if (['pptx', 'ppt'].includes(ext)) return <Presentation className={className} />
   107→  if (ext === 'txt') return <FileType className={className} />
   108→  return <File className={className} />
   109→}

   466→const ALLOWED_EXTS = ['pdf', 'doc', 'docx', 'xlsx', 'pptx', 'txt', 'jpg', 'jpeg', 'png']
   467→const MAX_FILE_MB = 10
   468→
   469→function DocumentFormDialog({
   470→  open, onOpenChange, editId, onSaved,
   471→}: {
   472→  open: boolean
   473→  onOpenChange: (v: boolean) => void
   474→  editId?: string
   475→  onSaved: () => void
   476→}) {
   477→  const [title, setTitle] = useState('')
   478→  const [description, setDescription] = useState('')
   479→  const [category, setCategory] = useState('policy')
   480→  const [accessLevel, setAccessLevel] = useState('staff')
   481→  const [file, setFile] = useState<File | null>(null)
   482→  const [loading, setLoading] = useState(false)
   483→
   484→  useQuery({
   485→    queryKey: ['document', editId] as const,
   486→    queryFn: async () => {
   487→      if (!editId) return null
   488→      const res = await fetch(`/api/documents/${editId}`)
   489→      if (!res.ok) {
   490→        const e = await res.json().catch(() => ({}))
   491→        throw new Error(e.error || 'Failed to load')
   492→      }
   493→      const d: DocumentItem = await res.json()
   494→      setTitle(d.title)
   495→      setDescription(d.description ?? '')
   496→      setCategory(d.category)
   497→      setAccessLevel(d.accessLevel)
   498→      return d
   499→    },
   500→    enabled: !!editId && open,
   501→  })
   502→
   503→  useEffect(() => {
   504→    if (!open && !editId) {
   505→      setTitle('')
   506→      setDescription('')
   507→      setCategory('policy')
   508→      setAccessLevel('staff')
   509→      setFile(null)
   510→    }
   511→  }, [open, editId])
   512→
   513→  const isEdit = !!editId
   514→
   515→  const validateFile = (f: File): string | null => {
   516→    if (f.size > MAX_FILE_MB * 1024 * 1024) {
   517→      return `File too large. Max ${MAX_FILE_MB}MB.`
   518→    }
   519→    const ext = (f.name.split('.').pop() ?? '').toLowerCase()
   520→    if (!ALLOWED_EXTS.includes(ext)) {
   521→      return `File type .${ext || '???'} not allowed. Allowed: ${ALLOWED_EXTS.join(', ')}`
   522→    }
   523→    return null
   524→  }
   525→
   526→  const onFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   527→    const f = e.target.files?.[0]
   528→    if (!f) {
   529→      setFile(null)
   530→      return
   531→    }
   532→    const err = validateFile(f)
   533→    if (err) {
   534→      toast.error(err)
   535→      e.target.value = ''
   536→      setFile(null)
   537→      return
   538→    }
   539→    setFile(f)
   540→  }
   541→
   542→  const save = async () => {
   543→    if (!title.trim()) {
   544→      toast.error('Title is required')
   545→      return
   546→    }
   547→    setLoading(true)
   548→    try {
   549→      if (isEdit) {
   550→        // Edit: only metadata — file replace is not supported via this dialog (would need a separate replace flow)
   551→        const res = await fetch(`/api/documents/${editId}`, {
   552→          method: 'PATCH',
   553→          headers: { 'Content-Type': 'application/json' },
   554→          body: JSON.stringify({
   555→            title: title.trim(),
   556→            description: description.trim() || null,
   557→            category,
   558→            accessLevel,
   559→          }),
   560→        })
   561→        if (!res.ok) {
   562→          const e = await res.json().catch(() => ({}))
   563→          throw new Error(e.error || 'Save failed')
   564→        }
   565→        toast.success('Document updated')
   566→        onSaved()
   567→        return
   568→      }
   569→      // Create with multipart
   570→      const form = new FormData()
   571→      form.set('title', title.trim())
   572→      form.set('description', description.trim())
   573→      form.set('category', category)
   574→      form.set('accessLevel', accessLevel)
   575→      if (file) form.set('file', file)
   576→      const res = await fetch('/api/documents', { method: 'POST', body: form })
   577→      if (!res.ok) {
   578→        const e = await res.json().catch(() => ({}))
   579→        throw new Error(e.error || 'Upload failed')
   580→      }
   581→      const data = await res.json()
   582→      toast.success(file ? 'Document uploaded' : 'Document created')
   583→      onSaved()
   584→    } catch (e: unknown) {
   585→      toast.error(e instanceof Error ? e.message : 'Save failed')
   586→    } finally {
   587→      setLoading(false)
   588→    }
   589→  }
   590→
   591→  return (
   592→    <Dialog open={open} onOpenChange={onOpenChange}>
   593→      <DialogContent className="sm:max-w-[600px] max-h-[90vh] overflow-y-auto scroll-thin">
   594→        <DialogHeader>
   595→          <DialogTitle>{isEdit ? 'Edit Document' : 'Upload Document'}</DialogTitle>
   596→          <DialogDescription>
   597→            {isEdit
   598→              ? 'Update document metadata. File replace is not supported here.'
   599→              : 'Upload a file or create metadata-only entry. Access levels enforce server-side visibility.'}
   600→          </DialogDescription>
   601→        </DialogHeader>
   602→        <div className="space-y-4 py-2">
   603→          <div>
   604→            <Label htmlFor="doc-title" className="text-sm font-medium">Title <span className="text-destructive">*</span></Label>
   605→            <Input
   606→              id="doc-title"
   607→              value={title}
   608→              onChange={(e) => setTitle(e.target.value)}
   609→              placeholder="e.g. Volunteer Code of Conduct"
   610→              className="mt-1.5 h-9"
   611→              maxLength={200}
   612→            />
   613→          </div>
   614→          <div>
   615→            <Label htmlFor="doc-desc" className="text-sm font-medium">Description</Label>
   616→            <Textarea
   617→              id="doc-desc"
   618→              value={description}
   619→              onChange={(e) => setDescription(e.target.value)}
   620→              placeholder="Short summary of the document…"
   621→              rows={3}
   622→              className="mt-1.5"
   623→              maxLength={1000}
   624→            />
   625→          </div>
   626→          <div className="grid grid-cols-2 gap-3">
   627→            <div>
   628→              <Label htmlFor="doc-cat" className="text-sm font-medium">Category</Label>
   629→              <Select value={category} onValueChange={setCategory}>
   630→                <SelectTrigger id="doc-cat" className="mt-1.5 h-9"><SelectValue /></SelectTrigger>
   631→                <SelectContent>
   632→                  {CATEGORIES.filter((c) => c.value).map((c) => (
   633→                    <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   634→                  ))}
   635→                </SelectContent>
   636→              </Select>
   637→            </div>
   638→            <div>
   639→              <Label htmlFor="doc-access" className="text-sm font-medium">Access level</Label>
   640→              <Select value={accessLevel} onValueChange={setAccessLevel}>
   641→                <SelectTrigger id="doc-access" className="mt-1.5 h-9"><SelectValue /></SelectTrigger>
   642→                <SelectContent>
   643→                  {ACCESS_LEVELS.filter((a) => a.value).map((a) => (
   644→                    <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>
   645→                  ))}
   646→                </SelectContent>
   647→              </Select>
   648→            </div>
   649→          </div>
   650→          {!isEdit && (
   651→            <div>
   652→              <Label htmlFor="doc-file" className="text-sm font-medium">File (optional)</Label>
   653→              <div className="mt-1.5 border-2 border-dashed border-border/60 rounded-xl p-5 text-center hover:border-primary/40 transition-colors">
   654→                <input
   655→                  id="doc-file"
   656→                  type="file"
   657→                  onChange={onFileChange}
   658→                  accept=".pdf,.doc,.docx,.xlsx,.pptx,.txt,.jpg,.jpeg,.png"
   659→                  className="hidden"
   660→                />
   661→                <Label htmlFor="doc-file" className="cursor-pointer inline-flex flex-col items-center">
   662→                  <Upload className="size-6 text-muted-foreground mb-2" />
   663→                  <span className="text-sm font-medium">{file ? file.name : 'Click to choose a file'}</span>
   664→                  <span className="text-xs text-muted-foreground mt-1">
   665→                    Max {MAX_FILE_MB}MB · {ALLOWED_EXTS.join(', ')}
   666→                  </span>
   667→                </Label>
   668→              </div>
   669→              {file && (
   670→                <div className="mt-2 flex items-center justify-between rounded-lg bg-muted/40 px-3 py-2 text-xs">
   671→                  <span className="truncate max-w-[80%]">{file.name}</span>
   672→                  <span className="text-muted-foreground nums">{(file.size / 1024 / 1024).toFixed(2)}MB</span>
   673→                  <button
   674→                    type="button"
   675→                    onClick={() => setFile(null)}
   676→                    className="text-muted-foreground hover:text-destructive"
   677→                    aria-label="Remove file"
   678→                  >
   679→                    <X className="size-3.5" />
   680→                  </button>
   681→                </div>
   682→              )}
   683→            </div>
   684→          )}
   685→          <div className="rounded-lg bg-muted/40 border border-border/60 p-3 text-xs text-muted-foreground flex items-start gap-2">
   686→            <Info className="size-3.5 mt-0.5 shrink-0" />
   687→            <div>
   688→              <span className="font-medium text-foreground">Access levels:</span>{' '}
   689→              public (everyone), members (member+), staff (church_admin/pastor/dept_head+), leaders (pastor/church_admin/dept_head), finance (finance_officer+), pastoral (pastor/super_admin).
   690→            </div>
   691→          </div>
   692→        </div>
   693→        <DialogFooter>
   694→          <DialogClose asChild><Button variant="outline">Cancel</Button></DialogClose>
   695→          <Button onClick={save} disabled={loading || !title.trim()} className="gap-2">
   696→            {loading ? 'Saving…' : isEdit ? 'Save changes' : 'Upload'}
   697→          </Button>
   698→        </DialogFooter>
   699→      </DialogContent>
   700→    </Dialog>
   701→  )
   702→}
