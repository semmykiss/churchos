   220→// ---------- report dialog container ----------
   221→function ReportDialog({
   222→  type, open, onOpenChange, canExport,
   223→}: {
   224→  type: AnyReport['type']
   225→  open: boolean
   226→  onOpenChange: (o: boolean) => void
   227→  canExport: boolean
   228→}) {
   229→  const [from, setFrom] = useState('')
   230→  const [to, setTo] = useState('')
   231→
   232→  const params = useMemo(() => {
   233→    const p = new URLSearchParams({ type })
   234→    if (from) p.set('from', from)
   235→    if (to) p.set('to', to)
   236→    return p.toString()
   237→  }, [type, from, to])
   238→
   239→  const { data, isLoading, isError, refetch } = useQuery<AnyReport>({
   240→    queryKey: ['reports', type, from, to],
   241→    queryFn: async () => {
   242→      const res = await fetch(`/api/reports?${params}`)
   243→      if (!res.ok) {
   244→        const e = await res.json().catch(() => ({}))
   245→        throw new Error(e.error || 'Failed to load report')
   246→      }
   247→      return res.json()
   248→    },
   249→    enabled: open,
   250→  })
   251→
   252→  const titles: Record<string, string> = {
   253→    membership: 'Membership Report',
   254→    attendance: 'Attendance Report',
   255→    finance: 'Finance Report',
   256→    visitors: 'Visitor Follow-up Report',
   257→    departments: 'Department Participation Report',
   258→  }
   259→
   260→  const exportCsv = () => {
   261→    try {
   262→      if (!data) return
   263→      toast.loading('Generating CSV…', { id: 'rep-export' })
   264→      let header: string[] = []
   265→      let rows: (string | number)[][] = []
   266→      let filename = type
   267→      if (data.type === 'membership') {
   268→        header = ['Status', 'Count']
   269→        rows = data.byStatus.map((s) => [s.status, s.count])
   270→        filename = `membership-by-status`
   271→      } else if (data.type === 'attendance') {
   272→        header = ['Service', 'Sessions']
   273→        rows = data.byService.map((s) => [s.serviceName, s.sessions])
   274→        filename = `attendance-by-service`
   275→      } else if (data.type === 'finance') {
   276→        header = ['Fund', 'Income', 'Expense', 'Net']
   277→        rows = data.byFund.map((f) => [f.fundName, f.income, f.expense, f.net])
   278→        filename = `finance-by-fund`
   279→      } else if (data.type === 'departments') {
   280→        header = ['Department', 'Head', 'Members', 'Tasks', 'Events', 'Units']
   281→        rows = data.items.map((d) => [d.name, d.head ?? '', d.members, d.tasks, d.events, d.units])
   282→        filename = `departments`
   283→      } else if (data.type === 'visitors') {
   284→        header = ['Source', 'Visitors']
   285→        rows = data.byReferral.map((r) => [r.source, r.count])
   286→        filename = `visitors-by-source`
   287→      }
   288→      const csv = [header, ...rows]
   289→        .map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(','))
   290→        .join('\n')
   291→      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8' })
   292→      const url = URL.createObjectURL(blob)
   293→      const a = document.createElement('a')
   294→      a.href = url
   295→      a.download = `${filename}-${new Date().toISOString().slice(0, 10)}.csv`
   296→      a.click()
   297→      URL.revokeObjectURL(url)
   298→      toast.success(`Exported ${rows.length} rows`, { id: 'rep-export' })
   299→    } catch {
   300→      toast.error('Export failed', { id: 'rep-export' })
   301→    }
   302→  }
   303→
   304→  return (
   305→    <Dialog open={open} onOpenChange={onOpenChange}>
   306→      <DialogContent className="max-w-4xl lg:max-w-[1000px] max-h-[90vh] overflow-y-auto scroll-thin">
   307→        <DialogHeader>
   308→          <DialogTitle>{titles[type] ?? 'Report'}</DialogTitle>
   309→          <DialogDescription>Filtered by date range. CSV export uses the same access scope.</DialogDescription>
   310→        </DialogHeader>
   311→
   312→        <ReportFilters
   313→          from={from}
   314→          to={to}
   315→          setFrom={setFrom}
   316→          setTo={setTo}
   317→          onReset={() => { setFrom(''); setTo('') }}
   318→        />
   319→
   320→        {isLoading ? (
   321→          <div className="space-y-3">
   322→            <Skeleton className="h-24 rounded-xl" />
   323→            <Skeleton className="h-64 rounded-xl" />
   324→          </div>
   325→        ) : isError ? (
   326→          <EmptyState
   327→            icon={AlertTriangle}
   328→            title="Failed to load report"
   329→            description="Please try again or adjust your filters."
   330→            action={
   331→              <Button variant="outline" size="sm" onClick={() => refetch()} className="gap-2">
   332→                <RefreshCw className="size-3.5" /> Retry
   333→              </Button>
   334→            }
   335→          />
   336→        ) : !data ? (
   337→          <p className="text-sm text-muted-foreground text-center py-6">No data.</p>
   338→        ) : data.type === 'membership' ? (
   339→          <MembershipReportView data={data} />
   340→        ) : data.type === 'attendance' ? (
   341→          <AttendanceReportView data={data} />
   342→        ) : data.type === 'finance' ? (
   343→          <FinanceReportView data={data} />
   344→        ) : data.type === 'visitors' ? (
   345→          <VisitorsReportView data={data} />
   346→        ) : data.type === 'departments' ? (
   347→          <DepartmentsReportView data={data} />
   348→        ) : null}
   349→
   350→        <DialogFooter>
   351→          <DialogClose asChild><Button variant="outline">Close</Button></DialogClose>
   352→          {canExport && data && (
   353→            <Button onClick={exportCsv} variant="outline" className="gap-2">
   354→              <Download className="size-3.5" /> Export CSV
   355→            </Button>
   356→          )}
   357→        </DialogFooter>
   358→      </DialogContent>
   359→    </Dialog>
   360→  )
   361→}
