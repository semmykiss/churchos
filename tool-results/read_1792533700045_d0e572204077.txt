   255→   255→function PrayerRequestFormCard({ enabled }: { enabled: boolean }) {
   256→   256→  const queryClient = useQueryClient()
   257→   257→  const [request, setRequest] = useState('')
   258→   258→  const [isPrivate, setIsPrivate] = useState(true)
   259→   259→  const [loading, setLoading] = useState(false)
   260→   260→
   261→   261→  const submit = async () => {
   262→   262→    if (!request.trim()) {
   263→   263→      toast.error('Please write your prayer request first')
   264→   264→      return
   265→   265→    }
   266→   266→    setLoading(true)
   267→   267→    try {
   268→   268→      const res = await fetch('/api/pastoral/prayer', {
   269→   269→        method: 'POST',
   270→   270→        headers: { 'Content-Type': 'application/json' },
   271→   271→        body: JSON.stringify({ request: request.trim(), private: isPrivate }),
   272→   272→      })
   273→   273→      if (!res.ok) {
   274→   274→        const e = await res.json().catch(() => ({}))
   275→   275→        throw new Error(e.error || 'Submit failed')
   276→   276→      }
   277→   277→      toast.success('Prayer request submitted — our team is praying with you.')
   278→   278→      setRequest('')
   279→   279→      queryClient.invalidateQueries({ queryKey: ['pastoral', 'prayer'] })
   280→   280→    } catch (e: unknown) {
   281→   281→      toast.error(e instanceof Error ? e.message : 'Submit failed')
   282→   282→    } finally {
   283→   283→      setLoading(false)
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  return (
   288→   288→    <Card className={enabled ? 'border-border/60' : 'opacity-60'}>
   289→   289→      <CardHeader>
   290→   290→        <CardTitle className="flex items-center gap-2 text-base">
   291→   291→          <MessageCircle className="size-4 text-primary" /> Submit a Prayer Request
   292→   292→        </CardTitle>
   293→   293→        <CardDescription className="text-xs">
   294→   294→          Share what you would like the pastoral team to pray about.
   295→   295→        </CardDescription>
   296→   296→      </CardHeader>
   297→   297→      <CardContent className="space-y-3">
   298→   298→        <Textarea
   299→   299→          value={request}
   300→   300→          onChange={(e) => setRequest(e.target.value)}
   301→   301→          placeholder="e.g. Please pray for healing for my mother…"
   302→   302→          rows={5}
   303→   303→          maxLength={2000}
   304→   304→        />
   305→   305→        <div className="flex items-center justify-between rounded-lg border border-border/60 p-3">
   306→   306→          <div>
   307→   307→            <Label htmlFor="pr-private" className="font-medium text-sm">Keep private</Label>

   311→   311→        </div>
