   136→function useDebouncedValue<T>(value: T, delay = 350): T {
   137→  const [debounced, setDebounced] = useState(value)
   138→  useEffect(() => {
   139→    const t = setTimeout(() => setDebounced(value), delay)
   140→    return () => clearTimeout(t)
   141→  }, [value, delay])
   142→  return debounced
   143→}

   500→                      <TableHead className="text-xs uppercase tracking-wide text-muted-foreground font-medium">End</TableHead>
   501→                      <TableHead className="pr-4 text-xs uppercase tracking-wide text-muted-foreground font-medium">Status</TableHead>
   502→                    </TableRow>
   503→                  </TableHeader>
   504→                  <TableBody>
   505→                    {bookingsData!.items.map((b) => (
   506→                      <TableRow key={b.id} className="hover:bg-muted/40 border-border/60">
   507→                        <TableCell className="pl-4 font-medium text-sm">{b.resource}</TableCell>
   508→                        <TableCell className="text-sm">{b.title}</TableCell>
   509→                        <TableCell className="text-muted-foreground text-sm">{b.bookedBy ?? '—'}</TableCell>
   510→                        <TableCell className="text-sm nums">{formatDate(b.startAt)}</TableCell>
   511→                        <TableCell className="text-sm nums">{formatDate(b.endAt)}</TableCell>
   512→                        <TableCell className="pr-4">
   513→                          <Badge variant="outline" className={`text-[10px] font-medium ${b.status === 'pending' ? 'border-amber-500/30 text-amber-700' : ''}`}>
   514→                            {titleCase(b.status)}
   515→                          </Badge>
   516→                        </TableCell>
   517→                      </TableRow>
   518→                    ))}
   519→                  </TableBody>
   520→                </Table>
   521→              </CardContent>
   522→            </Card>
   523→          )}
   524→        </TabsContent>
   525→      </Tabs>
   526→
   527→      {/* Detail dialog */}
   528→      <AssetDetailDialog id={detailId} open={detailOpen} onOpenChange={setDetailOpen} />
   529→
   530→      {/* Add/Edit form */}
   531→      <AssetFormDialog
   532→        open={addOpen}
   533→        onOpenChange={setAddOpen}
   534→        onSaved={invalidate}
   535→        currency={currency}
   536→      />
   537→      <AssetFormDialog
   538→        open={editOpen}
   539→        onOpenChange={setEditOpen}
   540→        editId={editId}
   541→        onSaved={invalidate}
   542→        currency={currency}
   543→      />
   544→
   545→      {/* Delete dialog */}
   546→      <AlertDialog open={!!deleteId} onOpenChange={(o) => !o && setDeleteId(null)}>
   547→        <AlertDialogContent>
   548→          <AlertDialogHeader>
   549→            <AlertDialogTitle>Delete this asset?</AlertDialogTitle>
   550→            <AlertDialogDescription>
   551→              You are about to permanently delete &quot;{deleteName}&quot; from the asset register. This cannot be undone.
   552→            </AlertDialogDescription>
   553→          </AlertDialogHeader>
   554→          <AlertDialogFooter>
   555→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   556→            <AlertDialogAction
   557→              className="bg-destructive text-white hover:bg-destructive/90"
   558→              onClick={async () => {
   559→                if (!deleteId) return
   560→                try {
   561→                  const res = await fetch(`/api/assets/${deleteId}`, { method: 'DELETE' })
   562→                  if (!res.ok) {
   563→                    const e = await res.json().catch(() => ({}))
   564→                    throw new Error(e.error || 'Delete failed')
   565→                  }
   566→                  toast.success('Asset deleted')
   567→                  invalidate()
   568→                } catch (e) {
   569→                  toast.error((e as Error).message)
   570→                } finally {
   571→                  setDeleteId(null)
   572→                }
   573→              }}
   574→            >
   575→              Delete
   576→            </AlertDialogAction>
   577→          </AlertDialogFooter>
   578→        </AlertDialogContent>
   579→      </AlertDialog>
   580→    </div>
   581→  )
   582→}

   584→// ---------- Asset Detail Dialog ----------
   585→function AssetDetailDialog({
   586→  id,
   587→  open,
   588→  onOpenChange,
   589→}: {
   590→  id: string | null
   591→  open: boolean
   592→  onOpenChange: (v: boolean) => void
   593→}) {
   594→  const { data, isLoading } = useQuery({
   595→    // Keyed under the plural 'assets' root so the list mutations'
   596→    // invalidateQueries({ queryKey: ['assets'] }) also refreshes this detail.
   597→    queryKey: ['assets', 'detail', id],
   598→    queryFn: async () => {
   599→      const res = await fetch(`/api/assets/${id}`)
   600→      if (!res.ok) {
   601→        const e = await res.json().catch(() => ({}))
   602→        throw new Error(e.error || 'Failed to load asset')
   603→      }
   604→      return res.json()
   605→    },
   606→    enabled: !!id && open,
   607→  })
   608→  const { organization } = useAuth()
   609→  const currency = organization?.currency ?? 'NGN'
   610→
   611→  return (
   612→    <Dialog open={open} onOpenChange={onOpenChange}>
   613→      <DialogContent className="sm:max-w-[500px]">
   614→        <DialogHeader>
   615→          <DialogTitle>Asset details</DialogTitle>
   616→          <DialogDescription>
   617→            {data ? data.name : 'Loading…'}
   618→          </DialogDescription>
   619→        </DialogHeader>
   620→        {isLoading ? (
   621→          <div className="space-y-3 py-4">
   622→            <Skeleton className="h-4 w-full" />
   623→            <Skeleton className="h-4 w-3/4" />
   624→            <Skeleton className="h-4 w-1/2" />
   625→          </div>
   626→        ) : data ? (
   627→          <div className="grid grid-cols-2 gap-3 py-2 text-sm">
   628→            <DetailRow label="Tag" value={data.tag} />
   629→            <DetailRow label="Category" value={titleCase(data.category)} />
   630→            <DetailRow label="Condition" value={titleCase(data.condition)} />
   631→            <DetailRow label="Location" value={data.location} />
   632→            <DetailRow label="Custodian" value={data.custodian} />
   633→            <DetailRow label="Acquired" value={data.acquiredAt ? formatDate(data.acquiredAt) : null} />
   634→            <DetailRow label="Value" value={data.value != null ? formatCurrency(data.value, currency) : null} />
   635→            <DetailRow label="Created" value={formatDate(data.createdAt)} />
   636→          </div>
   637→        ) : null}
   638→      </DialogContent>
   639→    </Dialog>
   640→  )
   641→}
   642→
   643→function DetailRow({ label, value }: { label: string; value: string | null | undefined }) {
   644→  return (
   645→    <div className="rounded-xl bg-muted/30 p-3">
   646→      <p className="text-xs text-muted-foreground">{label}</p>
   647→      <p className="font-medium mt-0.5 nums">{value || '—'}</p>
   648→    </div>
   649→  )
   650→}

   652→// ---------- Asset Form Dialog (create / edit) ----------
   653→function AssetFormDialog({
   654→  open,
   655→  onOpenChange,
   656→  editId,
   657→  onSaved,
   658→  currency,
   659→}: {
   660→  open: boolean
   661→  onOpenChange: (v: boolean) => void
   662→  editId?: string | null
   663→  onSaved: () => void
   664→  currency: string
   665→}) {
   666→  const isEdit = !!editId
   667→  const [name, setName] = useState('')
   668→  const [tag, setTag] = useState('')
   669→  const [category, setCategory] = useState('equipment')
   670→  const [location, setLocation] = useState('')
   671→  const [custodian, setCustodian] = useState('')
   672→  const [condition, setCondition] = useState('good')
   673→  const [acquiredAt, setAcquiredAt] = useState('')
   674→  const [value, setValue] = useState('')
   675→
   676→  useEffect(() => {
   677→    if (open && editId) {
   678→      fetch(`/api/assets/${editId}`)
   679→        .then((r) => r.json())
   680→        .then((d) => {
   681→          if (d && !d.error) {
   682→            setName(d.name || '')
   683→            setTag(d.tag || '')
   684→            setCategory(d.category || 'equipment')
   685→            setLocation(d.location || '')
   686→            setCustodian(d.custodian || '')
   687→            setCondition(d.condition || 'good')
   688→            setAcquiredAt(d.acquiredAt ? d.acquiredAt.slice(0, 10) : '')
   689→            setValue(d.value != null ? String(d.value) : '')
   690→          }
   691→        })
   692→        .catch(() => {})
   693→    } else if (open && !editId) {
   694→      void Promise.resolve().then(() => {
   695→        setName(''); setTag(''); setCategory('equipment'); setLocation('')
   696→        setCustodian(''); setCondition('good'); setAcquiredAt(''); setValue('')
   697→      })
   698→    }
   699→  }, [open, editId])
   700→
   701→  const saveMutation = useMutation({
   702→    mutationFn: async () => {
   703→      const payload: Record<string, unknown> = {
   704→        name,
   705→        tag: tag.trim() || null,
   706→        category,
   707→        location: location.trim() || null,
   708→        custodian: custodian.trim() || null,
   709→        condition,
   710→        acquiredAt: acquiredAt ? new Date(acquiredAt).toISOString() : null,
   711→        value: value !== '' ? Number(value) : null,
   712→      }
   713→      const url = isEdit ? `/api/assets/${editId}` : '/api/assets'
   714→      const method = isEdit ? 'PATCH' : 'POST'
   715→      const res = await fetch(url, {
   716→        method,
   717→        headers: { 'Content-Type': 'application/json' },
   718→        body: JSON.stringify(payload),
   719→      })
   720→      if (!res.ok) {
   721→        const e = await res.json().catch(() => ({}))
   722→        throw new Error(e.error || 'Save failed')
   723→      }
   724→      return res.json()
   725→    },
   726→    onSuccess: () => {
   727→      toast.success(isEdit ? 'Asset updated' : 'Asset added')
   728→      onOpenChange(false)
   729→      onSaved()
   730→    },
   731→    onError: (err: Error) => toast.error(err.message),
   732→  })
   733→
   734→  return (
   735→    <Dialog open={open} onOpenChange={onOpenChange}>
   736→      <DialogContent className="sm:max-w-[600px]">
   737→        <DialogHeader>
   738→          <DialogTitle>{isEdit ? 'Edit Asset' : 'Add Asset'}</DialogTitle>
   739→          <DialogDescription>
   740→            {isEdit ? 'Update asset details.' : 'Register a new church asset.'} ({currency})
   741→          </DialogDescription>
   742→        </DialogHeader>
   743→        <div className="grid grid-cols-1 gap-4 py-2 sm:grid-cols-2">
   744→          <div className="grid gap-2 sm:col-span-2">
   745→            <Label htmlFor="name" className="text-sm font-medium">Name</Label>
   746→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Yamaha PSR-S975 Keyboard" className="h-9" />
   747→          </div>
   748→          <div className="grid gap-2">
   749→            <Label htmlFor="tag" className="text-sm font-medium">Tag / Asset ID</Label>
   750→            <Input id="tag" value={tag} onChange={(e) => setTag(e.target.value)} placeholder="e.g. GCI-AST-1234" className="h-9" />
   751→          </div>
   752→          <div className="grid gap-2">
   753→            <Label className="text-sm font-medium">Category</Label>
   754→            <Select value={category} onValueChange={setCategory}>
   755→              <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   756→              <SelectContent>
   757→                {CATEGORY_OPTIONS.filter((c) => c.value).map((c) => (
   758→                  <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   759→                ))}
   760→              </SelectContent>
   761→            </Select>
   762→          </div>
   763→          <div className="grid gap-2">
   764→            <Label htmlFor="location" className="text-sm font-medium">Location</Label>
   765→            <Input id="location" value={location} onChange={(e) => setLocation(e.target.value)} placeholder="e.g. Main Hall" className="h-9" />
   766→          </div>
   767→          <div className="grid gap-2">
   768→            <Label htmlFor="custodian" className="text-sm font-medium">Custodian</Label>
   769→            <Input id="custodian" value={custodian} onChange={(e) => setCustodian(e.target.value)} placeholder="e.g. Media Dept" className="h-9" />
   770→          </div>
   771→          <div className="grid gap-2">
   772→            <Label className="text-sm font-medium">Condition</Label>
   773→            <Select value={condition} onValueChange={setCondition}>
   774→              <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   775→              <SelectContent>
   776→                {CONDITION_OPTIONS.filter((c) => c.value).map((c) => (
   777→                  <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   778→                ))}
   779→              </SelectContent>
   780→            </Select>
   781→          </div>
   782→          <div className="grid gap-2">
   783→            <Label htmlFor="acquiredAt" className="text-sm font-medium">Acquired date</Label>
   784→            <Input id="acquiredAt" type="date" value={acquiredAt} onChange={(e) => setAcquiredAt(e.target.value)} className="h-9" />
   785→          </div>
   786→          <div className="grid gap-2 sm:col-span-2">
   787→            <Label htmlFor="value" className="text-sm font-medium">Value ({currency})</Label>
   788→            <Input
   789→              id="value"
   790→              type="number"
   791→              step="0.01"
   792→              value={value}
   793→              onChange={(e) => setValue(e.target.value)}
   794→              placeholder="Optional — current value in organization currency"
   795→              className="h-9 nums"
   796→            />
   797→          </div>
   798→        </div>
   799→        <DialogFooter>
   800→          <DialogClose asChild>
   801→            <Button variant="outline">Cancel</Button>
   802→          </DialogClose>
   803→          <Button onClick={() => saveMutation.mutate()} disabled={saveMutation.isPending || !name.trim()} className="gap-2">
   804→            {saveMutation.isPending ? 'Saving…' : isEdit ? 'Save changes' : 'Add asset'}
   805→          </Button>
   806→        </DialogFooter>
   807→      </DialogContent>
   808→    </Dialog>
   809→  )
   810→}
