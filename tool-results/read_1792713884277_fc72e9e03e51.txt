   252→function useChatTheme(): [ChatTheme, (t: ChatTheme) => void] {
   253→  const [theme, setThemeState] = useState<ChatTheme>(() => {
   254→    if (typeof localStorage === 'undefined') return 'default'
   255→    try {
   256→      const saved = localStorage.getItem('churchos:chat-theme-v2') as ChatTheme | null
   257→      return saved && THEMES.some((t) => t.value === saved) ? saved : 'default'
   258→    } catch {
   259→      return 'default'
   260→    }
   261→  })
   262→  const setter = useCallback((t: ChatTheme) => {
   263→    setThemeState(t)
   264→    try { localStorage.setItem('churchos:chat-theme-v2', t) } catch { /* ignore */ }
   265→  }, [])
   266→  return [theme, setter]
   267→}

   324→function isSameDay(a: Date, b: Date): boolean {
   325→  return a.getFullYear() === b.getFullYear()
   326→    && a.getMonth() === b.getMonth()
   327→    && a.getDate() === b.getDate()

   330→function dayLabel(date: Date): string {
   331→  const today = new Date()
   332→  const yesterday = new Date()
   333→  yesterday.setDate(today.getDate() - 1)
   334→  if (isSameDay(date, today)) return 'Today'
   335→  if (isSameDay(date, yesterday)) return 'Yesterday'
   336→  return formatDate(date, { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })

   352→function nameInitials(name: string): string {
   353→  const parts = name.trim().split(/\s+/)
   354→  if (parts.length === 0) return '?'
   355→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
   356→  return (parts[0][0] + parts[1][0]).toUpperCase()

   359→function renderBody(body: string, search?: string): ReactNode {
   360→  const mentionParts = body.split(/(@[A-Za-z][A-Za-z'-]+(?:\s[A-Za-z][A-Za-z'-]+)?)/g)
   361→  const q = search?.trim().toLowerCase()
   362→  const nodes: ReactNode[] = []
   363→  mentionParts.forEach((p, i) => {
   364→    if (!p) return
   365→    if (p.startsWith('@') && p.length > 1) {
   366→      // If mention itself matches search, wrap entire mention in mark
   367→      if (q && p.toLowerCase().includes(q)) {
   368→        nodes.push(
   369→          <mark key={`m${i}`} className="bg-yellow-300/70 dark:bg-yellow-500/40 rounded px-0.5">
   370→            <span className="font-semibold text-primary">{p}</span>
   371→          </mark>,
   372→        )
   373→      } else {
   374→        nodes.push(<span key={`m${i}`} className="font-semibold text-primary">{p}</span>)
   375→      }
   376→      return
   377→    }
   378→    if (q) {
   379→      const lower = p.toLowerCase()
   380→      let idx = lower.indexOf(q)
   381→      let cursor = 0
   382→      const k = `t${i}`
   383→      if (idx === -1) {
   384→        nodes.push(<span key={k}>{p}</span>)
   385→        return
   386→      }
   387→      while (idx >= 0) {
   388→        if (idx > cursor) nodes.push(<span key={`${k}-${cursor}`}>{p.slice(cursor, idx)}</span>)
   389→        nodes.push(
   390→          <mark key={`${k}-${idx}`} className="bg-yellow-300/70 dark:bg-yellow-500/40 rounded px-0.5">
   391→            {p.slice(idx, idx + q.length)}
   392→          </mark>,
   393→        )
   394→        cursor = idx + q.length
   395→        idx = lower.indexOf(q, cursor)
   396→      }
   397→      if (cursor < p.length) nodes.push(<span key={`${k}-end`}>{p.slice(cursor)}</span>)
   398→    } else {
   399→      nodes.push(<span key={`t${i}`}>{p}</span>)
   400→    }
   401→  })
   402→  return <>{nodes}</>
   403→}

   459→  const [seenAt, setSeenAt] = useState<string | null>(() => getSeenAt(activeRoomId))
   460→  const [scrolledRoomId, setScrolledRoomId] = useState<string | null>(null)
   461→
   462→  // Room list — poll every 15s
   463→  const roomsQuery = useQuery<RoomsResponse>({
   464→    queryKey: ['townhall-rooms'],
   465→    queryFn: async () => {

   473→  // Active room + messages — poll every 5s
   474→  const messagesQuery = useQuery<RoomDetailResponse>({
   475→    queryKey: ['townhall-messages', activeRoomId, searchQuery],
   476→    queryFn: async () => {
   477→      if (!activeRoomId) {
   478→        return { room: null as unknown, messages: [], pinnedMessages: [] } as RoomDetailResponse
   479→      }
   480→      const qs = searchQuery ? `?q=${encodeURIComponent(searchQuery)}` : ''
   481→      const res = await fetch(`/api/townhall/${activeRoomId}${qs}`)
   482→      if (!res.ok) throw new Error('Failed to load messages')
   483→      return res.json() as Promise<RoomDetailResponse>
   484→    },
   485→    enabled: !!activeRoomId,
   486→    refetchInterval: 5_000,
   487→  })
   488→
   489→  // Render-time "adjust state when prop changes" pattern: when activeRoomId changes,
   490→  // reset compose state + read seenAt from localStorage. This avoids setState-in-effect.
   491→  // (See: https://react.dev/learn/you-might-not-need-an-effect#adjusting-some-state-when-a-prop-changes)
   492→  const [prevRoomId, setPrevRoomId] = useState<string | null>(activeRoomId)
   493→  if (activeRoomId !== prevRoomId) {
   494→    setPrevRoomId(activeRoomId)
   495→    setSeenAt(getSeenAt(activeRoomId))
   496→    setScrolledRoomId(null)
   497→    setReplyTo(null)
   498→    setEditingMsg(null)
   499→    setDraft('')
   500→    setAttachment(null)
   501→    setMentionPickerOpen(false)
   502→    setSearchOpen(false)
   503→    setSearchQuery('')
   504→    setPinnedBannerOpen(true)
   505→  }
   506→
   507→  // After ~5s of viewing, persist seenAt to latest message timestamp so divider clears.
   508→  useEffect(() => {
   509→    if (!activeRoomId) return
   510→    const msgs = messagesQuery.data?.messages
   511→    if (!msgs || msgs.length === 0) return
   512→    const latest = msgs[msgs.length - 1]
   513→    const timer = setTimeout(() => {
   514→      setSeenAt(activeRoomId, latest.createdAt)
   515→      setSeenAt(latest.createdAt)
   516→    }, 5_000)
   517→    return () => clearTimeout(timer)
   518→  }, [activeRoomId, messagesQuery.data?.messages])
   519→
   520→  // Auto-scroll: initial scroll to first unread OR bottom; subsequent = stick-to-bottom
   521→  useEffect(() => {
   522→    const el = scrollRef.current
   523→    if (!el) return
   524→    const msgs = messagesQuery.data?.messages
   525→    if (!msgs || msgs.length === 0) return
   526→
   527→    const isInitial = scrolledRoomId !== activeRoomId
   528→    if (isInitial) {
   529→      let scrolled = false
   530→      if (seenAt) {
   531→        const seenTime = new Date(seenAt).getTime()
   532→        const idx = msgs.findIndex((m) => new Date(m.createdAt).getTime() > seenTime)
   533→        if (idx >= 0) {
   534→          const target = el.querySelector(`[data-msg-id="${msgs[idx].id}"]`) as HTMLElement | null
   535→          if (target) {
   536→            target.scrollIntoView({ behavior: 'auto', block: 'center' })

  1422→// Message stream — renders day separators, unread divider, bubbles
  1423→// ===========================================================================
  1424→function MessageStream({
  1425→  messages, currentUserId, roomType, themeCfg, searchQuery, unreadDividerIdx,
  1426→  isStaffViewer, onReply, onEdit, onDelete, onPin, onReact,
  1427→}: {
  1428→  messages: ChatMessage[]
  1429→  currentUserId: string
  1430→  roomType: RoomType
  1431→  themeCfg: ThemeConfig
  1432→  searchQuery: string
  1433→  unreadDividerIdx: number
  1434→  isStaffViewer: boolean
  1435→  onReply: (m: ChatMessage) => void
  1436→  onEdit: (m: ChatMessage) => void
  1437→  onDelete: (m: ChatMessage) => void
  1438→  onPin: (m: ChatMessage) => void
  1439→  onReact: (m: ChatMessage, emoji: string) => void
  1440→}) {
  1441→  // Group by day
  1442→  const groups: { day: string; items: ChatMessage[] }[] = []
  1443→  let lastDay = ''
  1444→  for (const m of messages) {
  1445→    const dl = dayLabel(new Date(m.createdAt))
  1446→    if (dl !== lastDay) {
  1447→      groups.push({ day: dl, items: [m] })
  1448→      lastDay = dl
  1449→    } else {
  1450→      groups[groups.length - 1].items.push(m)
  1451→    }
  1452→  }
  1453→
  1454→  let runningIdx = -1
  1455→
  1456→  return (
  1457→    <div className="space-y-1">
  1458→      {groups.map((grp, gi) => (
  1459→        <div key={gi} className="space-y-1">
  1460→          <div className="flex items-center justify-center my-3">
  1461→            <span className="text-[10px] uppercase tracking-wide font-medium text-muted-foreground bg-muted/60 rounded-full px-2.5 py-1">
  1462→              {grp.day}
  1463→            </span>
  1464→          </div>
  1465→          {grp.items.map((m, idx) => {
  1466→            runningIdx += 1
  1467→            const mine = m.senderId === currentUserId
  1468→            const prev = idx > 0 ? grp.items[idx - 1] : null
  1469→            const showSenderInfo = !prev
  1470→              || prev.senderId !== m.senderId
  1471→              || (new Date(m.createdAt).getTime() - new Date(prev.createdAt).getTime()) > 5 * 60_000
  1472→            const showUnreadDivider = runningIdx === unreadDividerIdx
  1473→            return (
  1474→              <div key={m.id} data-msg-id={m.id}>
  1475→                {showUnreadDivider && (
  1476→                  <div className="flex items-center gap-2 my-3 px-1">
  1477→                    <div className="h-px bg-primary/40 flex-1" />
  1478→                    <span className="text-[10px] font-semibold uppercase tracking-wide text-primary bg-primary/10 rounded-full px-2 py-0.5">
  1479→                      New messages
  1480→                    </span>
  1481→                    <div className="h-px bg-primary/40 flex-1" />
  1482→                  </div>
  1483→                )}
  1484→                <MessageBubble
  1485→                  message={m}
  1486→                  mine={mine}
  1487→                  showSenderInfo={showSenderInfo}
  1488→                  showAvatar={roomType !== 'direct'}
  1489→                  themeCfg={themeCfg}
  1490→                  searchQuery={searchQuery}
  1491→                  currentUserId={currentUserId}
  1492→                  isStaffViewer={isStaffViewer}
  1493→                  onReply={onReply}
  1494→                  onEdit={onEdit}
  1495→                  onDelete={onDelete}
  1496→                  onPin={onPin}
  1497→                  onReact={onReact}
  1498→                />
  1499→              </div>
  1500→            )
  1501→          })}
  1502→        </div>
  1503→      ))}
  1504→    </div>
  1505→  )
  1506→}

  1511→function MessageBubble({
  1512→  message, mine, showSenderInfo, showAvatar, themeCfg, searchQuery, currentUserId,

  1531→  const visibleReplies = showAllReplies ? replies : replies.slice(0, 2)
  1532→  const hiddenCount = replies.length - visibleReplies.length
  1533→
  1534→  const deleted = isDeletedBody(message.body)
  1535→  const edited = !!message.editedAt && !deleted
  1536→  const canEdit = mine && within5Min(message.createdAt) && !deleted
  1537→  const canDelete = mine || isStaffViewer
  1538→  const canPin = mine || isStaffViewer
  1539→
  1540→  // Build reaction chips
  1541→  const reactionEntries = Object.entries(message.reactions ?? {})
  1542→
  1543→  return (
  1544→    <div className={cn('flex gap-2 group', mine ? 'flex-row-reverse' : 'flex-row')}>
  1545→      {showAvatar && !mine && (
  1546→        <div className="w-8 shrink-0">
  1547→          {showSenderInfo && (
  1548→            <Avatar className="size-8 mt-0.5">
