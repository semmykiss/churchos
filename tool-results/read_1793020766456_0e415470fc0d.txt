   107→// ---------- debounced search hook ----------
   108→function useDebouncedValue<T>(value: T, delay = 350): T {
   109→  const [debounced, setDebounced] = useState(value)
   110→  useEffect(() => {
   111→    const t = setTimeout(() => setDebounced(value), delay)
   112→    return () => clearTimeout(t)
   113→  }, [value, delay])
   114→  return debounced
   115→}

   726→// ---------- helpers ----------
   727→function renderPageButtons(current: number, total: number): (number | '...')[] {
   728→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1)
   729→  const out: (number | '...')[] = [1]
   730→  if (current > 3) out.push('...')
   731→  for (let i = Math.max(2, current - 1); i <= Math.min(total - 1, current + 1); i++) out.push(i)
   732→  if (current < total - 2) out.push('...')
   733→  out.push(total)
   734→  return out
   735→}
