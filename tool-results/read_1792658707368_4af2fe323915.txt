    99→function useChatTheme(): [ChatTheme, (t: ChatTheme) => void] {
   100→  const [theme, setTheme] = useState<ChatTheme>(() => {
   101→    if (typeof localStorage === 'undefined') return 'default'
   102→    const saved = (localStorage.getItem('churchos:chat-theme')) as ChatTheme | null
   103→    return saved && THEMES.some((t) => t.value === saved) ? saved : 'default'
   104→  })
   105→  const setter = useCallback((t: ChatTheme) => {
   106→    setTheme(t)
   107→    try { localStorage.setItem('churchos:chat-theme', t) } catch {}
   108→  }, [])
   109→  return [theme, setter]
   110→}

   139→function formatTime(date: string | Date) {
   140→  const d = typeof date === 'string' ? new Date(date) : date
   141→  return new Intl.DateTimeFormat('en-GB', { hour: '2-digit', minute: '2-digit', timeZone: 'Africa/Lagos' }).format(d)
   142→}
   143→
   144→function isSameDay(a: Date, b: Date) {
   145→  return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate()
   146→}
   147→
   148→function dayLabel(date: Date) {
   149→  const today = new Date()
   150→  const yesterday = new Date()
   151→  yesterday.setDate(today.getDate() - 1)
   152→  if (isSameDay(date, today)) return 'Today'
   153→  if (isSameDay(date, yesterday)) return 'Yesterday'
   154→  return formatDate(date, { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })
   155→}

   157→function renderMentions(body: string) {
   158→  // Split on @FirstName LastName patterns and highlight
   159→  const parts = body.split(/(@[A-Za-z][A-Za-z'-]+(?:\s[A-Za-z][A-Za-z'-]+)?)/g)
   160→  return parts.map((p, i) => {
   161→    if (p.startsWith('@') && p.length > 1) {
   162→      return <span key={i} className="font-semibold text-indigo-600 dark:text-indigo-400">{p}</span>
   163→    }
   164→    return <span key={i}>{p}</span>
   165→  })
   166→}

   663→// ---------- message bubble with thread ----------
   664→function MessageBubble({
   665→  message, mine, showSenderInfo, showAvatar, themeMine, themeOthers, onReply,
   666→}: {
   667→  message: Message
   668→  mine: boolean
   669→  showSenderInfo: boolean
   670→  showAvatar: boolean
   671→  themeMine: string
   672→  themeOthers: string
   673→  onReply: () => void
   674→}) {
   675→  const [showAllReplies, setShowAllReplies] = useState(false)
   676→  const replies = message.replies ?? []
   677→  const visibleReplies = showAllReplies ? replies : replies.slice(0, 2)
   678→  const hiddenCount = replies.length - visibleReplies.length
   679→
   680→  return (
   681→    <div className={cn('flex gap-2 group', mine ? 'flex-row-reverse' : 'flex-row')}>
   682→      {showAvatar && !mine && showSenderInfo && (
   683→        <Avatar className="size-8 mt-0.5 shrink-0">
   684→          <AvatarImage src={message.senderId ? undefined : undefined} alt={message.senderName} />
   685→          <AvatarFallback className="text-[10px] bg-primary/10 text-primary">
   686→            {initials(message.senderName.split(' ')[0], message.senderName.split(' ')[1])}
   687→          </AvatarFallback>
   688→        </Avatar>
   689→      )}
   690→      {showAvatar && !mine && !showSenderInfo && <div className="w-8 shrink-0" />}
   691→      <div className={cn('flex flex-col max-w-[78%] sm:max-w-[68%]', mine ? 'items-end' : 'items-start')}>
   692→        {showSenderInfo && !mine && showAvatar && (
   693→          <span className="text-[11px] font-medium text-muted-foreground mb-1 px-1">{message.senderName}</span>
   694→        )}
   695→        <div className={cn(
   696→          'rounded-2xl border px-3.5 py-2 text-sm relative',
   697→          mine ? themeMine : themeOthers,
   698→        )}>
   699→          <div className="whitespace-pre-wrap break-words leading-relaxed">{renderMentions(message.body)}</div>
   700→          <div className={cn('flex items-center gap-1 mt-1', mine ? 'justify-end' : 'justify-start')}>
   701→            <span className="text-[9px] text-muted-foreground nums">
   702→              {formatTime(message.createdAt)}
   703→            </span>
   704→            {mine && <CheckCheck className="size-3 text-muted-foreground" />}
   705→          </div>
   706→          {/* Reply button — appears on hover */}
   707→          <button
   708→            onClick={onReply}
   709→            className="absolute -top-2.5 opacity-0 group-hover:opacity-100 transition-opacity bg-background border border-border/60 rounded-full size-6 grid place-items-center shadow-premium hover:bg-accent"
   710→            style={mine ? { left: '-10px' } : { right: '-10px' }}
   711→            aria-label="Reply"
   712→          >
   713→            <Reply className="size-3" />
   714→          </button>
   715→        </div>
   716→
   717→        {/* Threaded replies */}
   718→        {replies.length > 0 && (
   719→          <div className={cn('mt-1.5 w-full space-y-1', mine ? 'items-end' : 'items-start')}>
   720→            {visibleReplies.map((r) => {
   721→              const rMine = r.senderId === message.senderId && mine
   722→              return (
   723→                <div key={r.id} className={cn('flex', rMine ? 'justify-end' : 'justify-start')}>
   724→                  <div className={cn(
   725→                    'rounded-xl border px-3 py-1.5 text-xs max-w-[80%]',
   726→                    rMine ? themeMine : themeOthers,
   727→                  )}>
   728→                    {!rMine && (
   729→                      <p className="text-[10px] font-medium text-muted-foreground mb-0.5">{r.senderName}</p>
   730→                    )}
   731→                    <div className="whitespace-pre-wrap break-words">{renderMentions(r.body)}</div>
   732→                    <span className="text-[9px] text-muted-foreground nums block mt-0.5">{formatTime(r.createdAt)}</span>
   733→                  </div>
   734→                </div>
   735→              )
   736→            })}
   737→            {hiddenCount > 0 && (
   738→              <button
   739→                onClick={() => setShowAllReplies(true)}
   740→                className="text-[11px] text-primary hover:underline font-medium flex items-center gap-1"
   741→              >
   742→                <CircleDot className="size-3" /> View {hiddenCount} more {hiddenCount === 1 ? 'reply' : 'replies'}
   743→              </button>
   744→            )}
   745→          </div>
   746→        )}
   747→      </div>
   748→    </div>
   749→  )
   750→}
