    59→function roleStyle(role: string): string {
    60→  return ROLES.find((r) => r.value === role)?.color ?? 'bg-muted text-muted-foreground border-border'
    61→}
    62→function roleLabel(role: string): string {
    63→  return ROLES.find((r) => r.value === role)?.label ?? titleCase(role)
    64→}

    66→function initials(name: string): string {
    67→  const parts = name.split(' ').filter(Boolean)
    68→  return ((parts[0]?.[0] ?? '') + (parts[1]?.[0] ?? '')).toUpperCase() || '?'
    69→}

   146→// ============= MY ACCOUNT =============
   147→function MyAccountSection() {
   148→  const { user, organization } = useAuth()
   149→  const role = user?.role ?? 'member'
   150→  const queryClient = useQueryClient()
   151→  const [name, setName] = useState(user?.name ?? '')
   152→  const [phone, setPhone] = useState(user?.phone ?? '')
   153→  const [avatarUrl, setAvatarUrl] = useState(user?.avatarUrl ?? '')
   154→  const [oldPassword, setOldPassword] = useState('')
   155→  const [newPassword, setNewPassword] = useState('')
   156→  const [confirmPassword, setConfirmPassword] = useState('')
   157→
   158→  useEffect(() => {
   159→    if (user) {
   160→      // Defer to keep effect body free of synchronous setState.
   161→      void Promise.resolve().then(() => {
   162→        setName(user.name ?? '')
   163→        setPhone(user.phone ?? '')
   164→        setAvatarUrl(user.avatarUrl ?? '')
   165→      })
   166→    }
   167→  }, [user])
   168→
   169→  const profileMutation = useMutation({
   170→    mutationFn: async () => {
   171→      const res = await fetch('/api/profile', {
   172→        method: 'PATCH',
   173→        headers: { 'Content-Type': 'application/json' },
   174→        body: JSON.stringify({
   175→          name,
   176→          phone: phone || null,
   177→          avatarUrl: avatarUrl || null,
   178→        }),
   179→      })
   180→      if (!res.ok) {
   181→        const e = await res.json().catch(() => ({}))
   182→        throw new Error(e.error || 'Update failed')
   183→      }
   184→      return res.json()
   185→    },
   186→    onSuccess: () => {
   187→      toast.success('Profile updated')
   188→      // Reload session to reflect new name everywhere
   189→      fetch('/api/me', { cache: 'no-store' })
   190→        .then((r) => r.json())
   191→        .then((data) => {
   192→          if (data.user) {
   193→            useAuth.setState({ user: data.user, organization: data.organization })
   194→          }
   195→        })
   196→      queryClient.invalidateQueries({ queryKey: ['profile'] })
   197→    },
   198→    onError: (err: Error) => toast.error(err.message),
   199→  })
   200→
   201→  const passwordMutation = useMutation({
   202→    mutationFn: async () => {
   203→      const res = await fetch('/api/profile', {
   204→        method: 'POST',
   205→        headers: { 'Content-Type': 'application/json' },
   206→        body: JSON.stringify({
   207→          action: 'changePassword',
   208→          oldPassword,
   209→          newPassword,
   210→        }),
   211→      })
   212→      if (!res.ok) {
   213→        const e = await res.json().catch(() => ({}))
   214→        throw new Error(e.error || 'Password change failed')
   215→      }
   216→      return res.json()
   217→    },
   218→    onSuccess: () => {
   219→      toast.success('Password changed')
   220→      setOldPassword(''); setNewPassword(''); setConfirmPassword('')
   221→    },
   222→    onError: (err: Error) => toast.error(err.message),
   223→  })
   224→
   225→  return (
   226→    <div className="grid gap-6 lg:grid-cols-2">
   227→      {/* Profile card */}
   228→      <SectionCard
   229→        title="My profile"
   230→        description="Update your name, phone, and avatar URL."
   231→        action={<UserIcon className="size-4 text-muted-foreground" />}
   232→      >
   233→        <div className="space-y-4">
   234→          <div className="flex items-center gap-4">
   235→            <Avatar className="size-16">
   236→              {avatarUrl ? <AvatarImage src={avatarUrl} alt={name || user?.name || '?'} /> : null}
   237→              <AvatarFallback className="bg-primary/10 text-primary">
   238→                {initials(name || user?.name || '?')}
   239→              </AvatarFallback>
   240→            </Avatar>
   241→            <div>
   242→              <p className="font-medium">{user?.name}</p>
   243→              <p className="text-sm text-muted-foreground">{user?.email}</p>
   244→              <Badge variant="outline" className={`mt-1 text-[11px] font-medium ${roleStyle(role)}`}>{roleLabel(role)}</Badge>
   245→            </div>
   246→          </div>
   247→          <div className="grid gap-2">
   248→            <Label htmlFor="name" className="text-sm font-medium">Full name</Label>
   249→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} className="h-9" />
   250→          </div>
   251→          <div className="grid gap-2">
   252→            <Label htmlFor="phone" className="text-sm font-medium">Phone</Label>
   253→            <Input id="phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+234 800 000 0000" className="h-9" />
   254→          </div>
   255→          <div className="grid gap-2">
   256→            <Label htmlFor="avatarUrl" className="text-sm font-medium">Avatar URL</Label>
   257→            <Input id="avatarUrl" value={avatarUrl} onChange={(e) => setAvatarUrl(e.target.value)} placeholder="https://…" className="h-9" />
   258→          </div>
   259→          <div className="rounded-xl bg-muted/40 p-3 text-xs text-muted-foreground">
   260→            <p className="flex items-center gap-1.5"><Mail className="size-3" /> Email and role are managed by administrators and cannot be edited here.</p>
   261→          </div>
   262→          <Button onClick={() => profileMutation.mutate()} disabled={profileMutation.isPending || !name.trim()} className="gap-2">
   263→            <Save className="size-4" /> {profileMutation.isPending ? 'Saving…' : 'Save profile'}
   264→          </Button>
   265→        </div>
   266→      </SectionCard>
   267→
   268→      {/* Password card */}
   269→      <SectionCard
   270→        title="Change password"
   271→        description="For your security, verify your current password before setting a new one."
   272→        action={<KeyRound className="size-4 text-muted-foreground" />}
   273→      >
   274→        <div className="space-y-4">
   275→          <div className="grid gap-2">
   276→            <Label htmlFor="oldPassword" className="text-sm font-medium">Current password</Label>
   277→            <Input id="oldPassword" type="password" value={oldPassword} onChange={(e) => setOldPassword(e.target.value)} className="h-9" />
   278→          </div>
   279→          <div className="grid gap-2">
   280→            <Label htmlFor="newPassword" className="text-sm font-medium">New password</Label>
   281→            <Input id="newPassword" type="password" value={newPassword} onChange={(e) => setNewPassword(e.target.value)} placeholder="At least 6 characters" className="h-9" />
   282→          </div>
   283→          <div className="grid gap-2">
   284→            <Label htmlFor="confirmPassword" className="text-sm font-medium">Confirm new password</Label>
   285→            <Input id="confirmPassword" type="password" value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} className="h-9" />
   286→            {confirmPassword && newPassword && confirmPassword !== newPassword && (
   287→              <p className="text-xs text-rose-600">Passwords do not match</p>
   288→            )}
   289→          </div>
   290→          <Button
   291→            onClick={() => passwordMutation.mutate()}
   292→            disabled={
   293→              passwordMutation.isPending ||
   294→              !oldPassword ||
   295→              !newPassword ||
   296→              newPassword.length < 6 ||
   297→              newPassword !== confirmPassword
   298→            }
   299→            className="gap-2"
   300→          >
   301→            {passwordMutation.isPending ? 'Updating…' : 'Update password'}
   302→          </Button>
   303→          <p className="rounded-xl bg-amber-500/10 border border-amber-500/20 p-2.5 text-xs text-amber-700 dark:text-amber-300">
   304→            <Lock className="mr-1 inline size-3" />
   305→            Password changes are audited and visible to administrators in the Audit Log.
   306→          </p>
   307→        </div>
   308→      </SectionCard>
   309→    </div>
   310→  )
   311→}
