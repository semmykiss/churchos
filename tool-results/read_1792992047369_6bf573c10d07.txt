    89→// ---------- Local thread store (serverless-friendly) ----------
    90→// On Vercel each lambda has an ephemeral DB copy, so the server-side message
    91→// history can be incomplete between requests. We keep the full thread in
    92→// localStorage per conversation and merge it with whatever the server returns,
    93→// so the UI thread never loses messages. Also sent as `history` to the chat
    94→// API for multi-turn context when server history is empty.
    95→const threadKey = (convId: string) => `churchos:ai:thread:${convId}`
    96→function loadLocalThread(convId: string | null): ChatMessage[] {
    97→  if (!convId) return []
    98→  try {
    99→    const raw = localStorage.getItem(threadKey(convId))
   100→    if (!raw) return []
   101→    const arr = JSON.parse(raw)
   102→    return Array.isArray(arr)
   103→      ? arr.filter((m: ChatMessage) => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string')
   104→      : []
   105→  } catch {
   106→    return []
   107→  }
   108→}
   109→function saveLocalThread(convId: string | null, msgs: ChatMessage[]) {
   110→  if (!convId) return
   111→  try {
   112→    localStorage.setItem(threadKey(convId), JSON.stringify(msgs.slice(-60)))
   113→  } catch { /* quota — ignore */ }
   114→}
   115→function mergeThreads(server: ChatMessage[], local: ChatMessage[]): ChatMessage[] {
   116→  // Server messages win for the same id; local fills gaps (dedupe by id + content+role).
   117→  const seen = new Set<string>()
   118→  const out: ChatMessage[] = []
   119→  for (const m of server) {
   120→    const k = m.id
   121→    if (k && seen.has(k)) continue
   122→    if (k) seen.add(k)
   123→    out.push(m)
   124→  }
   125→  for (const m of local) {
   126→    const kContent = `${m.role}:${m.content.slice(0, 120)}`
   127→    if (m.id && seen.has(m.id)) continue
   128→    if (seen.has(kContent)) continue
   129→    seen.add(m.id || kContent)
   130→    out.push(m)
   131→  }
   132→  return out.sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''))
   133→}

   230→export function AiAssistant() {
   231→  const user = useAuth((s) => s.user)
   232→  const canView = useAuth((s) => s.canClient('ai', 'view'))
   233→  const [open, setOpen] = useState(false)

   238→  const [activeId, setActiveId] = useState<string | null>(() => loadStoredConversationId())
   239→
   240→  // Persist activeId to localStorage whenever it changes.
   241→  useEffect(() => {
   242→    saveStoredConversationId(activeId)
   243→  }, [activeId])
   244→
   245→  // Graceful degradation — render nothing when user can't access the ai module.
   246→  if (!user || !canView) return null
   247→
   248→  return (
   249→    <>
   250→      <FloatingButton open={open} onOpen={() => setOpen(true)} />
   251→      <ChatPanel
   252→        open={open}
   253→        onClose={() => setOpen(false)}
   254→        activeId={activeId}
   255→        setActiveId={setActiveId}
   256→      />
   257→    </>

   593→      setLiveAssistantContent('')

   595→      const controller = new AbortController()

   597→      let streamedText = '' // accumulated delta text (closures see fresh value)
   598→
   599→      try {
   600→        // Build client-side history (last 20 turns of the merged thread, before
   601→        // this message) — used by the server when its DB history is empty
   602→        // (per-lambda ephemeral DB on serverless).
   603→        const threadBefore = mergeThreads(convQ.data?.messages ?? [], loadLocalThread(convId))

   625→        while (true) {
   626→          const { done, value } = await reader.read()

   633→              .split('\n')
   634→              .filter((l) => l.startsWith('data:'))

   640→              data = JSON.parse(payload)
   641→            } catch {
   642→              continue
   643→            }

   645→              // server confirmed the conversationId
   646→            } else if (data.type === 'delta') {
   647→              streamedText += data.delta
   648→              setLiveAssistantContent((prev) => prev + data.delta)

   650→              setStreaming(false)
   651→              setLiveAssistantContent('')

   661→                    content: streamedText,
   662→                    createdAt: new Date().toISOString(),

   668→              // then clear the optimistic user message (avoids duplicate display).
   669→              try {
   670→                await queryClient.invalidateQueries({
   671→                  queryKey: ['ai', 'conversation', convId],
   672→                })
   673→              } catch {
   674→                /* ignore */
   675→              }
   676→              queryClient.invalidateQueries({ queryKey: ['ai', 'conversations'] })
   677→              setOptimisticUserMsg(null)
   678→            } else if (data.type === 'error') {
   679→              setStreaming(false)
   680→              setLiveAssistantContent('')
   681→              setErrorMsg(data.message)
   682→              toast.error(data.message || 'AI completion failed')
   683→              // Server persists an assistant error message — refetch to show it.
   684→              try {
   685→                await queryClient.invalidateQueries({
   686→                  queryKey: ['ai', 'conversation', convId],
   687→                })
   688→              } catch {
   689→                /* ignore */
   690→              }
   691→              setOptimisticUserMsg(null)
   692→            }
   693→          }
   694→        }
   695→      } catch (err) {

   699→          // server-saved user message remains visible.

   703→            await queryClient.invalidateQueries({

   708→          }

   712→          setStreaming(false)
   713→          setLiveAssistantContent('')
   714→          const msg = e?.message || 'Streaming failed'
   715→          setErrorMsg(msg)
   716→          toast.error(msg)
   717→        }
   718→      } finally {
   719→        abortRef.current = null
   720→      }
