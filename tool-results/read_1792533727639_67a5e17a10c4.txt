    61→function roleStyle(role: string): string {
    62→  return ROLES.find((r) => r.value === role)?.color ?? 'bg-muted text-muted-foreground border-border'
    63→}
    64→function roleLabel(role: string): string {
    65→  return ROLES.find((r) => r.value === role)?.label ?? titleCase(role)

    68→function initials(name: string): string {
    69→  const parts = name.split(' ').filter(Boolean)
    70→  return ((parts[0]?.[0] ?? '') + (parts[1]?.[0] ?? '')).toUpperCase() || '?'
    71→}

   149→function MyAccountSection() {
   150→  const { user, organization } = useAuth()
   151→  const queryClient = useQueryClient()
   152→  const [name, setName] = useState(user?.name ?? '')
   153→  const [phone, setPhone] = useState(user?.phone ?? '')
   154→  const [avatarUrl, setAvatarUrl] = useState(user?.avatarUrl ?? '')
   155→  const [oldPassword, setOldPassword] = useState('')
   156→  const [newPassword, setNewPassword] = useState('')
   157→  const [confirmPassword, setConfirmPassword] = useState('')
   158→
   159→  useEffect(() => {
   160→    if (user) {
   161→      // Defer to keep effect body free of synchronous setState.
   162→      void Promise.resolve().then(() => {
   163→        setName(user.name ?? '')
   164→        setPhone(user.phone ?? '')
   165→        setAvatarUrl(user.avatarUrl ?? '')
   166→      })
   167→    }
   168→  }, [user])
   169→
   170→  const profileMutation = useMutation({
   171→    mutationFn: async () => {
   172→      const res = await fetch('/api/profile', {
   173→        method: 'PATCH',
   174→        headers: { 'Content-Type': 'application/json' },
   175→        body: JSON.stringify({
   176→          name,
   177→          phone: phone || null,
   178→          avatarUrl: avatarUrl || null,
   179→        }),
   180→      })
   181→      if (!res.ok) {
   182→        const e = await res.json().catch(() => ({}))
   183→        throw new Error(e.error || 'Update failed')
   184→      }
   185→      return res.json()
   186→    },
   187→    onSuccess: () => {
   188→      toast.success('Profile updated')
   189→      // Reload session to reflect new name everywhere
   190→      fetch('/api/me', { cache: 'no-store' })
   191→        .then((r) => r.json())
   192→        .then((data) => {
   193→          if (data.user) {
   194→            useAuth.setState({ user: data.user, organization: data.organization })
   195→          }
   196→        })
   197→      queryClient.invalidateQueries({ queryKey: ['profile'] })
   198→    },
   199→    onError: (err: Error) => toast.error(err.message),
   200→  })
   201→
   202→  const passwordMutation = useMutation({
   203→    mutationFn: async () => {
   204→      const res = await fetch('/api/profile', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify({
   208→          action: 'changePassword',
   209→          oldPassword,
   210→          newPassword,
   211→        }),
   212→      })
   213→      if (!res.ok) {
   214→        const e = await res.json().catch(() => ({}))
   215→        throw new Error(e.error || 'Password change failed')
   216→      }
   217→      return res.json()
   218→    },
   219→    onSuccess: () => {
   220→      toast.success('Password changed')
   221→      setOldPassword(''); setNewPassword(''); setConfirmPassword('')
   222→    },
   223→    onError: (err: Error) => toast.error(err.message),
   224→  })
   225→
   226→  return (
   227→    <div className="grid gap-6 lg:grid-cols-2">
   228→      {/* Profile card */}
   229→      <Card>
   230→        <CardHeader>
   231→          <CardTitle className="flex items-center gap-2"><UserIcon className="size-5" /> My profile</CardTitle>
   232→          <CardDescription>Update your name, phone, and avatar URL.</CardDescription>
   233→        </CardHeader>
   234→        <CardContent className="space-y-4">
   235→          <div className="flex items-center gap-4">
   236→            <Avatar className="size-16">
   237→              {avatarUrl ? <AvatarImage src={avatarUrl} /> : null}
   238→              <AvatarFallback className="bg-primary/10 text-primary">
   239→                {initials(name || user?.name || '?')}
   240→              </AvatarFallback>
   241→            </Avatar>
   242→            <div>
   243→              <p className="font-medium">{user?.name}</p>
   244→              <p className="text-sm text-muted-foreground">{user?.email}</p>
   245→              <Badge variant="outline" className={`mt-1 ${roleStyle(role)}`}>{roleLabel(role)}</Badge>
   246→            </div>
   247→          </div>
   248→          <div className="grid gap-2">
   249→            <Label htmlFor="name">Full name</Label>
   250→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} />
   251→          </div>
   252→          <div className="grid gap-2">
   253→            <Label htmlFor="phone">Phone</Label>
   254→            <Input id="phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+234 800 000 0000" />
   255→          </div>
   256→          <div className="grid gap-2">
   257→            <Label htmlFor="avatarUrl">Avatar URL</Label>
   258→            <Input id="avatarUrl" value={avatarUrl} onChange={(e) => setAvatarUrl(e.target.value)} placeholder="https://…" />
   259→          </div>
   260→          <div className="rounded-md bg-muted/50 p-3 text-xs text-muted-foreground">
   261→            <p className="flex items-center gap-1"><Mail className="size-3" /> Email and role are managed by administrators and cannot be edited here.</p>
   262→          </div>
   263→          <Button onClick={() => profileMutation.mutate()} disabled={profileMutation.isPending || !name.trim()} className="gap-2">
   264→            <Save className="size-4" /> {profileMutation.isPending ? 'Saving…' : 'Save profile'}
   265→          </Button>
   266→        </CardContent>
   267→      </Card>
   268→
   269→      {/* Password card */}
   270→      <Card>
   271→        <CardHeader>
   272→          <CardTitle className="flex items-center gap-2"><KeyRound className="size-5" /> Change password</CardTitle>
   273→          <CardDescription>For your security, verify your current password before setting a new one.</CardDescription>
   274→        </CardHeader>
   275→        <CardContent className="space-y-4">
   276→          <div className="grid gap-2">
   277→            <Label htmlFor="oldPassword">Current password</Label>
   278→            <Input id="oldPassword" type="password" value={oldPassword} onChange={(e) => setOldPassword(e.target.value)} />
   279→          </div>
   280→          <div className="grid gap-2">
   281→            <Label htmlFor="newPassword">New password</Label>
   282→            <Input id="newPassword" type="password" value={newPassword} onChange={(e) => setNewPassword(e.target.value)} placeholder="At least 6 characters" />
   283→          </div>
   284→          <div className="grid gap-2">
   285→            <Label htmlFor="confirmPassword">Confirm new password</Label>
   286→            <Input id="confirmPassword" type="password" value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} />
   287→            {confirmPassword && newPassword && confirmPassword !== newPassword && (
   288→              <p className="text-xs text-rose-600">Passwords do not match</p>
   289→            )}
   290→          </div>
   291→          <Button
   292→            onClick={() => passwordMutation.mutate()}
   293→            disabled={
   294→              passwordMutation.isPending ||
   295→              !oldPassword ||
   296→              !newPassword ||
   297→              newPassword.length < 6 ||
   298→              newPassword !== confirmPassword
   299→            }
   300→            className="gap-2"
   301→          >
   302→            {passwordMutation.isPending ? 'Updating…' : 'Update password'}
   303→          </Button>
   304→          <p className="rounded-md bg-amber-500/10 p-2 text-xs text-amber-700 dark:text-amber-300">
   305→            <Lock className="mr-1 inline size-3" />
   306→            Password changes are audited and visible to administrators in the Audit Log.
   307→          </p>
   308→        </CardContent>
   309→      </Card>
   310→    </div>
   311→  )
   312→}
