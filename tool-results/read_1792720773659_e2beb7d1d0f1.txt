   252→function useChatTheme(): [ChatTheme, (t: ChatTheme) => void] {
   253→  const [theme, setThemeState] = useState<ChatTheme>(() => {
   254→    if (typeof localStorage === 'undefined') return 'default'
   255→    try {
   256→      const saved = localStorage.getItem('churchos:chat-theme-v2') as ChatTheme | null
   257→      return saved && THEMES.some((t) => t.value === saved) ? saved : 'default'
   258→    } catch {
   259→      return 'default'
   260→    }
   261→  })
   262→  const setter = useCallback((t: ChatTheme) => {
   263→    setThemeState(t)
   264→    try { localStorage.setItem('churchos:chat-theme-v2', t) } catch { /* ignore */ }
   265→  }, [])
   266→  return [theme, setter]
   267→}

   324→function isSameDay(a: Date, b: Date): boolean {
   325→  return a.getFullYear() === b.getFullYear()
   326→    && a.getMonth() === b.getMonth()
   327→    && a.getDate() === b.getDate()
   328→}

   330→function dayLabel(date: Date): string {
   331→  const today = new Date()
   332→  const yesterday = new Date()
   333→  yesterday.setDate(today.getDate() - 1)
   334→  if (isSameDay(date, today)) return 'Today'
   335→  if (isSameDay(date, yesterday)) return 'Yesterday'
   336→  return formatDate(date, { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })
   337→}

   352→function nameInitials(name: string): string {
   353→  const parts = name.trim().split(/\s+/)
   354→  if (parts.length === 0) return '?'
   355→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase()
   356→  return (parts[0][0] + parts[1][0]).toUpperCase()
   357→}

   359→function renderBody(body: string, search?: string): ReactNode {
   360→  const mentionParts = body.split(/(@[A-Za-z][A-Za-z'-]+(?:\s[A-Za-z][A-Za-z'-]+)?)/g)
   361→  const q = search?.trim().toLowerCase()
   362→  const nodes: ReactNode[] = []
   363→  mentionParts.forEach((p, i) => {
   364→    if (!p) return
   365→    if (p.startsWith('@') && p.length > 1) {
   366→      if (q && p.toLowerCase().includes(q)) {
   367→        nodes.push(
   368→          <mark key={`m${i}`} className="bg-yellow-300/70 dark:bg-yellow-500/40 rounded px-0.5">
   369→            <span className="font-semibold text-primary">{p}</span>
   370→          </mark>,
   371→        )
   372→      } else {
   373→        nodes.push(<span key={`m${i}`} className="font-semibold text-primary">{p}</span>)
   374→      }
   375→      return
   376→    }
   377→    if (q) {
   378→      const lower = p.toLowerCase()
   379→      let idx = lower.indexOf(q)
   380→      let cursor = 0
   381→      const k = `t${i}`
   382→      if (idx === -1) {
   383→        nodes.push(<span key={k}>{p}</span>)
   384→        return
   385→      }
   386→      while (idx >= 0) {
   387→        if (idx > cursor) nodes.push(<span key={`${k}-${cursor}`}>{p.slice(cursor, idx)}</span>)
   388→        nodes.push(
   389→          <mark key={`${k}-${idx}`} className="bg-yellow-300/70 dark:bg-yellow-500/40 rounded px-0.5">
   390→            {p.slice(idx, idx + q.length)}
   391→          </mark>,
   392→        )
   393→        cursor = idx + q.length
   394→        idx = lower.indexOf(q, cursor)
   395→      }
   396→      if (cursor < p.length) nodes.push(<span key={`${k}-end`}>{p.slice(cursor)}</span>)
   397→    } else {
   398→      nodes.push(<span key={`t${i}`}>{p}</span>)
   399→    }
   400→  })
   401→  return <>{nodes}</>
   402→}

   459→  const [scrolledRoomId, setScrolledRoomId] = useState<string | null>(null)
   460→

   462→  const roomsQuery = useQuery<RoomsResponse>({
   463→    queryKey: ['townhall-rooms'],
   464→    queryFn: async () => {

   474→    queryKey: ['townhall-messages', activeRoomId, searchQuery],
   475→    queryFn: async () => {
   476→      if (!activeRoomId) {
   477→        return { room: null as unknown, messages: [], pinnedMessages: [] } as RoomDetailResponse
   478→      }
   479→      const qs = searchQuery ? `?q=${encodeURIComponent(searchQuery)}` : ''
   480→      const res = await fetch(`/api/townhall/${activeRoomId}${qs}`)
   481→      if (!res.ok) throw new Error('Failed to load messages')
   482→      return res.json() as Promise<RoomDetailResponse>
   483→    },
   484→    enabled: !!activeRoomId,
   485→    refetchInterval: 5_000,
   486→  })
   487→
   488→  // Render-time "adjust state when prop changes" pattern: when activeRoomId changes,
   489→  // reset compose state + read seenAt from localStorage. This avoids setState-in-effect.
   490→  const [prevRoomId, setPrevRoomId] = useState<string | null>(activeRoomId)
   491→  if (activeRoomId !== prevRoomId) {
   492→    setPrevRoomId(activeRoomId)
   493→    setSeenAt(getSeenAt(activeRoomId))
   494→    setScrolledRoomId(null)
   495→    setReplyTo(null)

   497→    setDraft('')
   498→    setAttachment(null)

   500→    setSearchOpen(false)
   501→    setSearchQuery('')
   502→    setPinnedBannerOpen(true)
   503→  }
   504→
   505→  // After ~5s of viewing, persist seenAt to latest message timestamp so divider clears.
   506→  useEffect(() => {
   507→    if (!activeRoomId) return
   508→    const msgs = messagesQuery.data?.messages
   509→    if (!msgs || msgs.length === 0) return
   510→    const latest = msgs[msgs.length - 1]
   511→    const timer = setTimeout(() => {
   512→      setSeenAt(activeRoomId, latest.createdAt)
   513→      setSeenAt(latest.createdAt)
   514→    }, 5_000)
   515→    return () => clearTimeout(timer)
   516→  }, [activeRoomId, messagesQuery.data?.messages])
   517→
   518→  // Auto-scroll: initial scroll to first unread OR bottom; subsequent = stick-to-bottom
   519→  useEffect(() => {
   520→    const el = scrollRef.current
   521→    if (!el) return
   522→    const msgs = messagesQuery.data?.messages
   523→    if (!msgs || msgs.length === 0) return
   524→
   525→    const isInitial = scrolledRoomId !== activeRoomId
   526→    if (isInitial) {
   527→      let scrolled = false
   528→      if (seenAt) {
   529→        const seenTime = new Date(seenAt).getTime()
   530→        const idx = msgs.findIndex((m) => new Date(m.createdAt).getTime() > seenTime)
   531→        if (idx >= 0) {
   532→          const target = el.querySelector(`[data-msg-id="${msgs[idx].id}"]`) as HTMLElement | null
   533→          if (target) {
   534→            target.scrollIntoView({ behavior: 'auto', block: 'center' })
   535→            scrolled = true
   536→          }
   537→        }
   538→      }
