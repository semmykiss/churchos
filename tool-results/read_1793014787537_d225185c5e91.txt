   509→// Posted journal entry (header) — every posted transaction has balanced debits = credits
   510→model JournalEntry {
   511→  id             String   @id @default(cuid())
   512→  organizationId String
   513→  reference      String   @unique // unique traceable reference (e.g. "JE-2026-00001")
   514→  transactionDate DateTime
   515→  description    String
   516→  source         String   // manual, collection, expense, fund_transfer, bank_recon, adjustment, reversal
   517→  sourceRefId    String?  // ID of the source record (collectionSession, expenseRequest, etc.)
   518→  status         String   @default("posted") // posted, reversed
   519→  reversedByRef  String?  // reference of the reversing entry (if reversed)
   520→  reversalReason String?
   521→  periodLocked   Boolean  @default(false) // true if the accounting period was closed before this (defensive)
   522→  postedById     String?
   523→  postedByName   String?
   524→  createdAt      DateTime @default(now())
   525→  organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
   526→  ledgerEntries  LedgerEntry[]
   527→  @@index([organizationId])
   528→  @@index([transactionDate])
   529→  @@index([reference])
   530→  @@index([source])
   531→}
