   124→function useDebouncedValue<T>(value: T, delay = 350): T {
   125→  const [debounced, setDebounced] = useState(value)
   126→  useEffect(() => {
   127→    const t = setTimeout(() => setDebounced(value), delay)
   128→    return () => clearTimeout(t)
   129→  }, [value, delay])
   130→  return debounced
   131→}

   537→              You are about to permanently delete &quot;{deleteName}&quot; from the asset register. This cannot be undone.
   538→            </AlertDialogDescription>
   539→          </AlertDialogHeader>
   540→          <AlertDialogFooter>
   541→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   542→            <AlertDialogAction
   543→              className="bg-destructive text-white hover:bg-destructive/90"
   544→              onClick={async () => {
   545→                if (!deleteId) return
   546→                try {
   547→                  const res = await fetch(`/api/assets/${deleteId}`, { method: 'DELETE' })
   548→                  if (!res.ok) {
   549→                    const e = await res.json().catch(() => ({}))
   550→                    throw new Error(e.error || 'Delete failed')
   551→                  }
   552→                  toast.success('Asset deleted')
   553→                  invalidate()
   554→                } catch (e) {
   555→                  toast.error((e as Error).message)
   556→                } finally {
   557→                  setDeleteId(null)
   558→                }
   559→              }}
   560→            >
   561→              Delete
   562→            </AlertDialogAction>
   563→          </AlertDialogFooter>
   564→        </AlertDialogContent>
   565→      </AlertDialog>
   566→    </div>
   567→  )
   568→}

   570→// ---------- Asset Detail Dialog ----------
   571→function AssetDetailDialog({
   572→  id,
   573→  open,
   574→  onOpenChange,
   575→}: {
   576→  id: string | null
   577→  open: boolean
   578→  onOpenChange: (v: boolean) => void
   579→}) {
   580→  const { data, isLoading } = useQuery({
   581→    queryKey: ['asset', 'detail', id],
   582→    queryFn: async () => {
   583→      const res = await fetch(`/api/assets/${id}`)
   584→      if (!res.ok) {
   585→        const e = await res.json().catch(() => ({}))
   586→        throw new Error(e.error || 'Failed to load asset')
   587→      }
   588→      return res.json()
   589→    },
   590→    enabled: !!id && open,
   591→  })
   592→  const { organization } = useAuth()
   593→  const currency = organization?.currency ?? 'NGN'
   594→
   595→  return (
   596→    <Dialog open={open} onOpenChange={onOpenChange}>
   597→      <DialogContent className="sm:max-w-lg">
   598→        <DialogHeader>
   599→          <DialogTitle>Asset details</DialogTitle>
   600→          <DialogDescription>
   601→            {data ? data.name : 'Loading…'}
   602→          </DialogDescription>
   603→        </DialogHeader>
   604→        {isLoading ? (
   605→          <div className="space-y-3 py-4">
   606→            <Skeleton className="h-4 w-full" />
   607→            <Skeleton className="h-4 w-3/4" />
   608→            <Skeleton className="h-4 w-1/2" />
   609→          </div>
   610→        ) : data ? (
   611→          <div className="grid grid-cols-2 gap-3 py-2 text-sm">
   612→            <DetailRow label="Tag" value={data.tag} />
   613→            <DetailRow label="Category" value={titleCase(data.category)} />
   614→            <DetailRow label="Condition" value={titleCase(data.condition)} />
   615→            <DetailRow label="Location" value={data.location} />
   616→            <DetailRow label="Custodian" value={data.custodian} />
   617→            <DetailRow label="Acquired" value={data.acquiredAt ? formatDate(data.acquiredAt) : null} />
   618→            <DetailRow label="Value" value={data.value != null ? formatCurrency(data.value, currency) : null} />
   619→            <DetailRow label="Created" value={formatDate(data.createdAt)} />
   620→          </div>
   621→        ) : null}
   622→      </DialogContent>
   623→    </Dialog>
   624→  )
   625→}

   636→// ---------- Asset Form Dialog (create / edit) ----------
   637→function AssetFormDialog({
   638→  open,
   639→  onOpenChange,
   640→  editId,
   641→  onSaved,
   642→  currency,
   643→}: {
   644→  open: boolean
   645→  onOpenChange: (v: boolean) => void
   646→  editId?: string | null
   647→  onSaved: () => void
   648→  currency: string
   649→}) {
   650→  const isEdit = !!editId
   651→  const [name, setName] = useState('')
   652→  const [tag, setTag] = useState('')
   653→  const [category, setCategory] = useState('equipment')
   654→  const [location, setLocation] = useState('')
   655→  const [custodian, setCustodian] = useState('')
   656→  const [condition, setCondition] = useState('good')
   657→  const [acquiredAt, setAcquiredAt] = useState('')
   658→  const [value, setValue] = useState('')
   659→
   660→  useEffect(() => {
   661→    if (open && editId) {
   662→      fetch(`/api/assets/${editId}`)
   663→        .then((r) => r.json())
   664→        .then((d) => {
   665→          if (d && !d.error) {
   666→            setName(d.name || '')
   667→            setTag(d.tag || '')
   668→            setCategory(d.category || 'equipment')
   669→            setLocation(d.location || '')
   670→            setCustodian(d.custodian || '')
   671→            setCondition(d.condition || 'good')
   672→            setAcquiredAt(d.acquiredAt ? d.acquiredAt.slice(0, 10) : '')
   673→            setValue(d.value != null ? String(d.value) : '')
   674→          }
   675→        })
   676→        .catch(() => {})
   677→    } else if (open && !editId) {
   678→      void Promise.resolve().then(() => {
   679→        setName(''); setTag(''); setCategory('equipment'); setLocation('')
   680→        setCustodian(''); setCondition('good'); setAcquiredAt(''); setValue('')
   681→      })
   682→    }
   683→  }, [open, editId])
   684→
   685→  const saveMutation = useMutation({
   686→    mutationFn: async () => {
   687→      const payload: Record<string, unknown> = {
   688→        name,
   689→        tag: tag.trim() || null,
   690→        category,
   691→        location: location.trim() || null,
   692→        custodian: custodian.trim() || null,
   693→        condition,
   694→        acquiredAt: acquiredAt ? new Date(acquiredAt).toISOString() : null,
   695→        value: value !== '' ? Number(value) : null,
   696→      }
   697→      const url = isEdit ? `/api/assets/${editId}` : '/api/assets'
   698→      const method = isEdit ? 'PATCH' : 'POST'
   699→      const res = await fetch(url, {
   700→        method,
   701→        headers: { 'Content-Type': 'application/json' },
   702→        body: JSON.stringify(payload),
   703→      })
   704→      if (!res.ok) {
   705→        const e = await res.json().catch(() => ({}))
   706→        throw new Error(e.error || 'Save failed')
   707→      }
   708→      return res.json()
   709→    },
   710→    onSuccess: () => {
   711→      toast.success(isEdit ? 'Asset updated' : 'Asset added')
   712→      onOpenChange(false)
   713→      onSaved()
   714→    },
   715→    onError: (err: Error) => toast.error(err.message),
   716→  })
   717→
   718→  return (
   719→    <Dialog open={open} onOpenChange={onOpenChange}>
   720→      <DialogContent className="sm:max-w-2xl">
   721→        <DialogHeader>
   722→          <DialogTitle>{isEdit ? 'Edit Asset' : 'Add Asset'}</DialogTitle>
   723→          <DialogDescription>
   724→            {isEdit ? 'Update asset details.' : 'Register a new church asset.'} ({currency})
   725→          </DialogDescription>
   726→        </DialogHeader>
   727→        <div className="grid grid-cols-1 gap-4 py-2 sm:grid-cols-2">
   728→          <div className="grid gap-2 sm:col-span-2">
   729→            <Label htmlFor="name">Name</Label>
   730→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Yamaha PSR-S975 Keyboard" />
   731→          </div>
   732→          <div className="grid gap-2">
   733→            <Label htmlFor="tag">Tag / Asset ID</Label>
   734→            <Input id="tag" value={tag} onChange={(e) => setTag(e.target.value)} placeholder="e.g. GCI-AST-1234" />
   735→          </div>
   736→          <div className="grid gap-2">
   737→            <Label>Category</Label>
   738→            <Select value={category} onValueChange={setCategory}>
   739→              <SelectTrigger><SelectValue /></SelectTrigger>
   740→              <SelectContent>
   741→                {CATEGORY_OPTIONS.filter((c) => c.value).map((c) => (
   742→                  <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   743→                ))}
   744→              </SelectContent>
   745→            </Select>
   746→          </div>
   747→          <div className="grid gap-2">
   748→            <Label htmlFor="location">Location</Label>
   749→            <Input id="location" value={location} onChange={(e) => setLocation(e.target.value)} placeholder="e.g. Main Hall" />
   750→          </div>
   751→          <div className="grid gap-2">
   752→            <Label htmlFor="custodian">Custodian</Label>
   753→            <Input id="custodian" value={custodian} onChange={(e) => setCustodian(e.target.value)} placeholder="e.g. Media Dept" />
   754→          </div>
   755→          <div className="grid gap-2">
   756→            <Label>Condition</Label>
   757→            <Select value={condition} onValueChange={setCondition}>
   758→              <SelectTrigger><SelectValue /></SelectTrigger>
   759→              <SelectContent>
   760→                {CONDITION_OPTIONS.filter((c) => c.value).map((c) => (
   761→                  <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   762→                ))}
   763→              </SelectContent>
   764→            </Select>
   765→          </div>
   766→          <div className="grid gap-2">
   767→            <Label htmlFor="acquiredAt">Acquired date</Label>
   768→            <Input id="acquiredAt" type="date" value={acquiredAt} onChange={(e) => setAcquiredAt(e.target.value)} />
   769→          </div>
   770→          <div className="grid gap-2 sm:col-span-2">
   771→            <Label htmlFor="value">Value ({currency})</Label>
   772→            <Input
   773→              id="value"
   774→              type="number"
   775→              step="0.01"
   776→              value={value}
   777→              onChange={(e) => setValue(e.target.value)}
   778→              placeholder="Optional — current value in organization currency"
   779→            />
   780→          </div>
   781→        </div>
   782→        <DialogFooter>
   783→          <DialogClose asChild>
   784→            <Button variant="outline">Cancel</Button>
   785→          </DialogClose>
   786→          <Button onClick={() => saveMutation.mutate()} disabled={saveMutation.isPending || !name.trim()} className="gap-2">
   787→            {saveMutation.isPending ? 'Saving…' : isEdit ? 'Save changes' : 'Add asset'}
   788→          </Button>
   789→        </DialogFooter>
   790→      </DialogContent>
   791→    </Dialog>
   792→  )
   793→}
