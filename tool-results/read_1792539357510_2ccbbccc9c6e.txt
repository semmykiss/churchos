    61→function roleStyle(role: string): string {
    62→  return ROLES.find((r) => r.value === role)?.color ?? 'bg-muted text-muted-foreground border-border'
    63→}
    64→function roleLabel(role: string): string {
    65→  return ROLES.find((r) => r.value === role)?.label ?? titleCase(role)

    68→function initials(name: string): string {
    69→  const parts = name.split(' ').filter(Boolean)
    70→  return ((parts[0]?.[0] ?? '') + (parts[1]?.[0] ?? '')).toUpperCase() || '?'
    71→}

   149→function MyAccountSection() {
   150→  const { user, organization } = useAuth()
   151→  const role = user?.role ?? 'member'
   152→  const queryClient = useQueryClient()
   153→  const [name, setName] = useState(user?.name ?? '')
   154→  const [phone, setPhone] = useState(user?.phone ?? '')
   155→  const [avatarUrl, setAvatarUrl] = useState(user?.avatarUrl ?? '')
   156→  const [oldPassword, setOldPassword] = useState('')
   157→  const [newPassword, setNewPassword] = useState('')
   158→  const [confirmPassword, setConfirmPassword] = useState('')
   159→
   160→  useEffect(() => {
   161→    if (user) {
   162→      // Defer to keep effect body free of synchronous setState.
   163→      void Promise.resolve().then(() => {
   164→        setName(user.name ?? '')
   165→        setPhone(user.phone ?? '')
   166→        setAvatarUrl(user.avatarUrl ?? '')
   167→      })
   168→    }
   169→  }, [user])
   170→
   171→  const profileMutation = useMutation({
   172→    mutationFn: async () => {
   173→      const res = await fetch('/api/profile', {
   174→        method: 'PATCH',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({
   177→          name,
   178→          phone: phone || null,
   179→          avatarUrl: avatarUrl || null,
   180→        }),
   181→      })
   182→      if (!res.ok) {
   183→        const e = await res.json().catch(() => ({}))
   184→        throw new Error(e.error || 'Update failed')
   185→      }
   186→      return res.json()
   187→    },
   188→    onSuccess: () => {
   189→      toast.success('Profile updated')
   190→      // Reload session to reflect new name everywhere
   191→      fetch('/api/me', { cache: 'no-store' })
   192→        .then((r) => r.json())
   193→        .then((data) => {
   194→          if (data.user) {
   195→            useAuth.setState({ user: data.user, organization: data.organization })
   196→          }
   197→        })
   198→      queryClient.invalidateQueries({ queryKey: ['profile'] })
   199→    },
   200→    onError: (err: Error) => toast.error(err.message),
   201→  })
   202→
   203→  const passwordMutation = useMutation({
   204→    mutationFn: async () => {
   205→      const res = await fetch('/api/profile', {
   206→        method: 'POST',
   207→        headers: { 'Content-Type': 'application/json' },
   208→        body: JSON.stringify({
   209→          action: 'changePassword',
   210→          oldPassword,
   211→          newPassword,
   212→        }),
   213→      })
   214→      if (!res.ok) {
   215→        const e = await res.json().catch(() => ({}))
   216→        throw new Error(e.error || 'Password change failed')
   217→      }
   218→      return res.json()
   219→    },
   220→    onSuccess: () => {
   221→      toast.success('Password changed')
   222→      setOldPassword(''); setNewPassword(''); setConfirmPassword('')
   223→    },
   224→    onError: (err: Error) => toast.error(err.message),
   225→  })
   226→
   227→  return (
   228→    <div className="grid gap-6 lg:grid-cols-2">
   229→      {/* Profile card */}
   230→      <Card>
   231→        <CardHeader>
   232→          <CardTitle className="flex items-center gap-2"><UserIcon className="size-5" /> My profile</CardTitle>
   233→          <CardDescription>Update your name, phone, and avatar URL.</CardDescription>
   234→        </CardHeader>
   235→        <CardContent className="space-y-4">
   236→          <div className="flex items-center gap-4">
   237→            <Avatar className="size-16">
   238→              {avatarUrl ? <AvatarImage src={avatarUrl} /> : null}
   239→              <AvatarFallback className="bg-primary/10 text-primary">
   240→                {initials(name || user?.name || '?')}
   241→              </AvatarFallback>
   242→            </Avatar>
   243→            <div>
   244→              <p className="font-medium">{user?.name}</p>
   245→              <p className="text-sm text-muted-foreground">{user?.email}</p>
   246→              <Badge variant="outline" className={`mt-1 ${roleStyle(role)}`}>{roleLabel(role)}</Badge>
   247→            </div>
   248→          </div>
   249→          <div className="grid gap-2">
   250→            <Label htmlFor="name">Full name</Label>
   251→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} />
   252→          </div>
   253→          <div className="grid gap-2">
   254→            <Label htmlFor="phone">Phone</Label>
   255→            <Input id="phone" value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="+234 800 000 0000" />
   256→          </div>
   257→          <div className="grid gap-2">
   258→            <Label htmlFor="avatarUrl">Avatar URL</Label>
   259→            <Input id="avatarUrl" value={avatarUrl} onChange={(e) => setAvatarUrl(e.target.value)} placeholder="https://…" />
   260→          </div>
   261→          <div className="rounded-md bg-muted/50 p-3 text-xs text-muted-foreground">
   262→            <p className="flex items-center gap-1"><Mail className="size-3" /> Email and role are managed by administrators and cannot be edited here.</p>
   263→          </div>
   264→          <Button onClick={() => profileMutation.mutate()} disabled={profileMutation.isPending || !name.trim()} className="gap-2">
   265→            <Save className="size-4" /> {profileMutation.isPending ? 'Saving…' : 'Save profile'}
   266→          </Button>
   267→        </CardContent>
   268→      </Card>
   269→
   270→      {/* Password card */}
   271→      <Card>
   272→        <CardHeader>
   273→          <CardTitle className="flex items-center gap-2"><KeyRound className="size-5" /> Change password</CardTitle>
   274→          <CardDescription>For your security, verify your current password before setting a new one.</CardDescription>
   275→        </CardHeader>
   276→        <CardContent className="space-y-4">
   277→          <div className="grid gap-2">
   278→            <Label htmlFor="oldPassword">Current password</Label>
   279→            <Input id="oldPassword" type="password" value={oldPassword} onChange={(e) => setOldPassword(e.target.value)} />
   280→          </div>
   281→          <div className="grid gap-2">
   282→            <Label htmlFor="newPassword">New password</Label>
   283→            <Input id="newPassword" type="password" value={newPassword} onChange={(e) => setNewPassword(e.target.value)} placeholder="At least 6 characters" />
   284→          </div>
   285→          <div className="grid gap-2">
   286→            <Label htmlFor="confirmPassword">Confirm new password</Label>
   287→            <Input id="confirmPassword" type="password" value={confirmPassword} onChange={(e) => setConfirmPassword(e.target.value)} />
   288→            {confirmPassword && newPassword && confirmPassword !== newPassword && (
   289→              <p className="text-xs text-rose-600">Passwords do not match</p>
   290→            )}
   291→          </div>
   292→          <Button
   293→            onClick={() => passwordMutation.mutate()}
   294→            disabled={
   295→              passwordMutation.isPending ||
   296→              !oldPassword ||
   297→              !newPassword ||
   298→              newPassword.length < 6 ||
   299→              newPassword !== confirmPassword
   300→            }
   301→            className="gap-2"
   302→          >
   303→            {passwordMutation.isPending ? 'Updating…' : 'Update password'}
   304→          </Button>
   305→          <p className="rounded-md bg-amber-500/10 p-2 text-xs text-amber-700 dark:text-amber-300">
   306→            <Lock className="mr-1 inline size-3" />
   307→            Password changes are audited and visible to administrators in the Audit Log.
   308→          </p>
   309→        </CardContent>
   310→      </Card>
   311→    </div>
   312→  )
   313→}
