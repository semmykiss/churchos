    97→function FileIcon({ filename, className }: { filename?: string | null; className?: string }) {
    98→  const ext = (filename ?? '').split('.').pop()?.toLowerCase() ?? ''
    99→  if (['jpg', 'jpeg', 'png', 'gif'].includes(ext)) return <FileImage className={className} />
   100→  if (['xlsx', 'xls', 'csv'].includes(ext)) return <FileSpreadsheet className={className} />
   101→  if (['pptx', 'ppt'].includes(ext)) return <Presentation className={className} />
   102→  if (ext === 'txt') return <FileType className={className} />
   103→  return <File className={className} />
   104→}

   460→const ALLOWED_EXTS = ['pdf', 'doc', 'docx', 'xlsx', 'pptx', 'txt', 'jpg', 'jpeg', 'png']

   462→
   463→function DocumentFormDialog({
   464→  open, onOpenChange, editId, onSaved,
   465→}: {
   466→  open: boolean
   467→  onOpenChange: (v: boolean) => void
   468→  editId?: string
   469→  onSaved: () => void
   470→}) {
   471→  const [title, setTitle] = useState('')
   472→  const [description, setDescription] = useState('')
   473→  const [category, setCategory] = useState('policy')
   474→  const [accessLevel, setAccessLevel] = useState('staff')
   475→  const [file, setFile] = useState<File | null>(null)
   476→  const [loading, setLoading] = useState(false)
   477→
   478→  useQuery({
   479→    queryKey: ['document', editId] as const,
   480→    queryFn: async () => {
   481→      if (!editId) return null
   482→      const res = await fetch(`/api/documents/${editId}`)
   483→      if (!res.ok) {
   484→        const e = await res.json().catch(() => ({}))
   485→        throw new Error(e.error || 'Failed to load')
   486→      }
   487→      const d: DocumentItem = await res.json()
   488→      setTitle(d.title)
   489→      setDescription(d.description ?? '')
   490→      setCategory(d.category)
   491→      setAccessLevel(d.accessLevel)
   492→      return d
   493→    },
   494→    enabled: !!editId && open,
   495→  })
   496→
   497→  useEffect(() => {
   498→    if (!open && !editId) {
   499→      setTitle('')
   500→      setDescription('')
   501→      setCategory('policy')
   502→      setAccessLevel('staff')
   503→      setFile(null)
   504→    }
   505→  }, [open, editId])
   506→
   507→  const isEdit = !!editId
   508→
   509→  const validateFile = (f: File): string | null => {
   510→    if (f.size > MAX_FILE_MB * 1024 * 1024) {
   511→      return `File too large. Max ${MAX_FILE_MB}MB.`
   512→    }
   513→    const ext = (f.name.split('.').pop() ?? '').toLowerCase()
   514→    if (!ALLOWED_EXTS.includes(ext)) {
   515→      return `File type .${ext || '???'} not allowed. Allowed: ${ALLOWED_EXTS.join(', ')}`
   516→    }
   517→    return null
   518→  }
   519→
   520→  const onFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   521→    const f = e.target.files?.[0]
   522→    if (!f) {
   523→      setFile(null)
   524→      return
   525→    }
   526→    const err = validateFile(f)
   527→    if (err) {
   528→      toast.error(err)
   529→      e.target.value = ''
   530→      setFile(null)
   531→      return
   532→    }
   533→    setFile(f)
   534→  }
   535→
   536→  const save = async () => {
   537→    if (!title.trim()) {
   538→      toast.error('Title is required')
   539→      return
   540→    }
   541→    setLoading(true)
   542→    try {
   543→      if (isEdit) {
   544→        // Edit: only metadata — file replace is not supported via this dialog (would need a separate replace flow)
   545→        const res = await fetch(`/api/documents/${editId}`, {
   546→          method: 'PATCH',
   547→          headers: { 'Content-Type': 'application/json' },
   548→          body: JSON.stringify({
   549→            title: title.trim(),
   550→            description: description.trim() || null,
   551→            category,
   552→            accessLevel,
   553→          }),
   554→        })
   555→        if (!res.ok) {
   556→          const e = await res.json().catch(() => ({}))
   557→          throw new Error(e.error || 'Save failed')
   558→        }
   559→        toast.success('Document updated')
   560→        onSaved()
   561→        return
   562→      }
   563→      // Create with multipart
   564→      const form = new FormData()
   565→      form.set('title', title.trim())
   566→      form.set('description', description.trim())
   567→      form.set('category', category)
   568→      form.set('accessLevel', accessLevel)
   569→      if (file) form.set('file', file)
   570→      const res = await fetch('/api/documents', { method: 'POST', body: form })
   571→      if (!res.ok) {
   572→        const e = await res.json().catch(() => ({}))
   573→        throw new Error(e.error || 'Upload failed')
   574→      }
   575→      const data = await res.json()
   576→      toast.success(file ? 'Document uploaded' : 'Document created')
   577→      onSaved()
   578→    } catch (e: unknown) {
   579→      toast.error(e instanceof Error ? e.message : 'Save failed')
   580→    } finally {
   581→      setLoading(false)
   582→    }
   583→  }
   584→
   585→  return (
   586→    <Dialog open={open} onOpenChange={onOpenChange}>
   587→      <DialogContent className="sm:max-w-xl max-h-[90vh] overflow-y-auto">
   588→        <DialogHeader>
   589→          <DialogTitle>{isEdit ? 'Edit Document' : 'Upload Document'}</DialogTitle>
   590→          <DialogDescription>
   591→            {isEdit
   592→              ? 'Update document metadata. File replace is not supported here.'
   593→              : 'Upload a file or create metadata-only entry. Access levels enforce server-side visibility.'}
   594→          </DialogDescription>
   595→        </DialogHeader>
   596→        <div className="space-y-4 py-2">
   597→          <div>
   598→            <Label htmlFor="doc-title">Title <span className="text-destructive">*</span></Label>
   599→            <Input
   600→              id="doc-title"
   601→              value={title}
   602→              onChange={(e) => setTitle(e.target.value)}
   603→              placeholder="e.g. Volunteer Code of Conduct"
   604→              className="mt-1.5"
   605→              maxLength={200}
   606→            />
   607→          </div>
   608→          <div>
   609→            <Label htmlFor="doc-desc">Description</Label>
   610→            <Textarea
   611→              id="doc-desc"
   612→              value={description}
   613→              onChange={(e) => setDescription(e.target.value)}
   614→              placeholder="Short summary of the document…"
   615→              rows={3}
   616→              className="mt-1.5"
   617→              maxLength={1000}
   618→            />
   619→          </div>
   620→          <div className="grid grid-cols-2 gap-3">
   621→            <div>
   622→              <Label htmlFor="doc-cat">Category</Label>
   623→              <Select value={category} onValueChange={setCategory}>
   624→                <SelectTrigger id="doc-cat" className="mt-1.5"><SelectValue /></SelectTrigger>
   625→                <SelectContent>
   626→                  {CATEGORIES.filter((c) => c.value).map((c) => (
   627→                    <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   628→                  ))}
   629→                </SelectContent>
   630→              </Select>
   631→            </div>
   632→            <div>
   633→              <Label htmlFor="doc-access">Access level</Label>
   634→              <Select value={accessLevel} onValueChange={setAccessLevel}>
   635→                <SelectTrigger id="doc-access" className="mt-1.5"><SelectValue /></SelectTrigger>
   636→                <SelectContent>
   637→                  {ACCESS_LEVELS.filter((a) => a.value).map((a) => (
   638→                    <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>
   639→                  ))}
   640→                </SelectContent>
   641→              </Select>
   642→            </div>
   643→          </div>
   644→          {!isEdit && (
   645→            <div>
   646→              <Label htmlFor="doc-file">File (optional)</Label>
   647→              <div className="mt-1.5 border-2 border-dashed border-border/60 rounded-lg p-5 text-center hover:border-primary/40 transition-colors">
   648→                <input
   649→                  id="doc-file"
   650→                  type="file"
   651→                  onChange={onFileChange}
   652→                  accept=".pdf,.doc,.docx,.xlsx,.pptx,.txt,.jpg,.jpeg,.png"
   653→                  className="hidden"
   654→                />
   655→                <Label htmlFor="doc-file" className="cursor-pointer inline-flex flex-col items-center">
   656→                  <Upload className="size-6 text-muted-foreground mb-2" />
   657→                  <span className="text-sm font-medium">{file ? file.name : 'Click to choose a file'}</span>
   658→                  <span className="text-xs text-muted-foreground mt-1">
   659→                    Max {MAX_FILE_MB}MB · {ALLOWED_EXTS.join(', ')}
   660→                  </span>
   661→                </Label>
   662→              </div>
   663→              {file && (
   664→                <div className="mt-2 flex items-center justify-between rounded-md bg-muted/40 px-3 py-2 text-xs">
   665→                  <span className="truncate max-w-[80%]">{file.name}</span>
   666→                  <span className="text-muted-foreground">{(file.size / 1024 / 1024).toFixed(2)}MB</span>
   667→                  <button
   668→                    type="button"
   669→                    onClick={() => setFile(null)}
   670→                    className="text-muted-foreground hover:text-destructive"
   671→                    aria-label="Remove file"
   672→                  >
   673→                    <X className="size-3.5" />
   674→                  </button>
   675→                </div>
   676→              )}
   677→            </div>
   678→          )}
   679→          <div className="rounded-md bg-muted/40 border border-border/60 p-3 text-xs text-muted-foreground flex items-start gap-2">
   680→            <Info className="size-3.5 mt-0.5 shrink-0" />
   681→            <div>
   682→              <span className="font-medium text-foreground">Access levels:</span>{' '}
   683→              public (everyone), members (member+), staff (church_admin/pastor/dept_head+), leaders (pastor/church_admin/dept_head), finance (finance_officer+), pastoral (pastor/super_admin).
   684→            </div>
   685→          </div>
   686→        </div>
   687→        <DialogFooter>
   688→          <DialogClose asChild><Button variant="outline">Cancel</Button></DialogClose>
   689→          <Button onClick={save} disabled={loading || !title.trim()}>
   690→            {loading ? 'Saving…' : isEdit ? 'Save changes' : 'Upload'}
   691→          </Button>
   692→        </DialogFooter>
   693→      </DialogContent>
   694→    </Dialog>
   695→  )
   696→}
