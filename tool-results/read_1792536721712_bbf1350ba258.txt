   102→// ---------- debounced search hook ----------
   103→function useDebouncedValue<T>(value: T, delay = 350): T {
   104→  const [debounced, setDebounced] = useState(value)
   105→  useEffect(() => {
   106→    const t = setTimeout(() => setDebounced(value), delay)
   107→    return () => clearTimeout(t)
   108→  }, [value, delay])
   109→  return debounced
   110→}

   634→function renderPageButtons(current: number, total: number): (number | '...')[] {
   635→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1)
   636→  const out: (number | '...')[] = [1]
   637→  if (current > 3) out.push('...')
   638→  for (let i = Math.max(2, current - 1); i <= Math.min(total - 1, current + 1); i++) out.push(i)
   639→  if (current < total - 2) out.push('...')
   640→  out.push(total)
   641→  return out
   642→}

   908→  const [saving, setSaving] = useState(false)
   909→  const [branches, setBranches] = useState<{ id: string; name: string }[]>([])

   968→    setForm((f) => ({ ...f, [k]: v }))

   972→      toast.error('First and last name are required')

   974→    }

   977→      const payload: Record<string, unknown> = {
   978→        firstName: form.firstName.trim(),
   979→        lastName: form.lastName.trim(),
   980→        email: form.email.trim() || null,
   981→        phone: form.phone.trim() || null,
   982→        gender: form.gender || null,
   983→        dateOfBirth: form.dateOfBirth || null,
   984→        branchId: form.branchId || null,
   985→        baptized: form.baptized,
   986→        baptizedDate: form.baptizedDate || null,
   987→        address: form.address.trim() || null,

   989→        emergencyPhone: form.emergencyPhone.trim() || null,
   990→        notes: form.notes.trim() || null,
   991→        memberId: form.memberId.trim() || null,
   992→      }

   994→      if (canEditStatus) payload.privateNotes = form.privateNotes.trim() || null
   995→
   996→      const url = mode === 'create' ? '/api/members' : `/api/members/${memberId}`
   997→      const method = mode === 'create' ? 'POST' : 'PATCH'

   999→        method,
  1000→        headers: { 'Content-Type': 'application/json' },

  1015→  }

  1018→    <Dialog open={open} onOpenChange={onOpenChange}>
