    83→// ---------- Local thread store (serverless-friendly) ----------
    84→// On Vercel each lambda has an ephemeral DB copy, so the server-side message
    85→// history can be incomplete between requests. We keep the full thread in
    86→// localStorage per conversation and merge it with whatever the server returns,
    87→// so the UI thread never loses messages. Also sent as `history` to the chat
    88→// API for multi-turn context when server history is empty.
    89→const threadKey = (convId: string) => `churchos:ai:thread:${convId}`
    90→function loadLocalThread(convId: string | null): ChatMessage[] {
    91→  if (!convId) return []
    92→  try {
    93→    const raw = localStorage.getItem(threadKey(convId))
    94→    if (!raw) return []
    95→    const arr = JSON.parse(raw)
    96→    return Array.isArray(arr)
    97→      ? arr.filter((m: ChatMessage) => m && (m.role === 'user' || m.role === 'assistant') && typeof m.content === 'string')
    98→      : []
    99→  } catch {
   100→    return []
   101→  }
   102→}
   103→function saveLocalThread(convId: string | null, msgs: ChatMessage[]) {
   104→  if (!convId) return
   105→  try {
   106→    localStorage.setItem(threadKey(convId), JSON.stringify(msgs.slice(-60)))
   107→  } catch { /* quota — ignore */ }
   108→}
   109→function mergeThreads(server: ChatMessage[], local: ChatMessage[]): ChatMessage[] {
   110→  // Server messages win for the same id; local fills gaps (dedupe by id + content+role).
   111→  const seen = new Set<string>()
   112→  const out: ChatMessage[] = []
   113→  for (const m of server) {
   114→    const k = m.id
   115→    if (k && seen.has(k)) continue
   116→    if (k) seen.add(k)
   117→    out.push(m)
   118→  }
   119→  for (const m of local) {
   120→    const kContent = `${m.role}:${m.content.slice(0, 120)}`
   121→    if (m.id && seen.has(m.id)) continue
   122→    if (seen.has(kContent)) continue
   123→    seen.add(m.id || kContent)
   124→    out.push(m)
   125→  }
   126→  return out.sort((a, b) => (a.createdAt || '').localeCompare(b.createdAt || ''))
   127→}

   244→      <FloatingButton open={open} onOpen={() => setOpen(true)} />
   245→      <ChatPanel
   246→        open={open}
   247→        onClose={() => setOpen(false)}
   248→        activeId={activeId}
   249→        setActiveId={setActiveId}

   551→  const sendMessage = useCallback(

   554→      if (!trimmed || streaming) return

   696→          try {
   697→            await queryClient.invalidateQueries({

   702→          }

   709→          setErrorMsg(msg)
   710→          toast.error(msg)
   711→        }
   712→      } finally {
   713→        abortRef.current = null
   714→      }
