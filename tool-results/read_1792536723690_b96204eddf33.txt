   102→   102→// ---------- debounced search hook ----------
   103→   103→function useDebouncedValue<T>(value: T, delay = 350): T {
   104→   104→  const [debounced, setDebounced] = useState(value)
   105→   105→  useEffect(() => {
   106→   106→    const t = setTimeout(() => setDebounced(value), delay)
   107→   107→    return () => clearTimeout(t)
   108→   108→  }, [value, delay])
   109→   109→  return debounced
   110→   110→}

   634→   634→function renderPageButtons(current: number, total: number): (number | '...')[] {
   635→   635→  if (total <= 7) return Array.from({ length: total }, (_, i) => i + 1)
   636→   636→  const out: (number | '...')[] = [1]
   637→   637→  if (current > 3) out.push('...')
   638→   638→  for (let i = Math.max(2, current - 1); i <= Math.min(total - 1, current + 1); i++) out.push(i)
   639→   639→  if (current < total - 2) out.push('...')
   640→   640→  out.push(total)
   641→   641→  return out
   642→   642→}

   894→   894→function MemberFormDialog({
   895→   895→  mode, memberId, open, onOpenChange, onSaved, currency: _currency,

   898→   898→  memberId?: string
   899→   899→  open: boolean
   900→   900→  onOpenChange: (o: boolean) => void
   901→   901→  onSaved: () => void
   902→   902→  currency?: string
   903→   903→}) {
   904→   904→  const { user } = useAuth()
   905→   905→  const role = user?.role ?? 'member'
   906→   906→  const canEditStatus = can(role, 'members', 'update')
   907→   907→  const [form, setForm] = useState<FormState>(EMPTY_FORM)
   908→   908→  const [saving, setSaving] = useState(false)
   909→   909→  const [branches, setBranches] = useState<{ id: string; name: string }[]>([])
   910→   910→
   911→   911→  // Load branches (small list, no permissions endpoint exists — use dashboard-friendly fetch via /api/me? No, fetch directly)
   912→   912→  // Branches are referenced in member PATCH, so we provide a static fetch.
   913→   913→  useEffect(() => {
   914→   914→    if (!open) return
   915→   915→    ;(async () => {
   916→   916→      try {
   917→   917→        const res = await fetch('/api/members?pageSize=1')
   918→   918→        // We don't have a /api/branches endpoint — branch is optional. Provide a graceful fallback list using known seed branch IDs.
   919→   919→        setBranches([
   920→   920→          { id: 'branch-main', name: 'Main Campus — Lagos' },
   921→   921→          { id: 'branch-east', name: 'East Campus — Lekki' },
   922→   922→        ])
   923→   923→        void res
   924→   924→      } catch {
   925→   925→        setBranches([])
   926→   926→      }
   927→   927→    })()
   928→   928→  }, [open])
   929→   929→
   930→   930→  // For edit mode, load the existing member
   931→   931→  useEffect(() => {
   932→   932→    if (!open) return
   933→   933→    if (mode === 'edit' && memberId) {
   934→   934→      ;(async () => {
   935→   935→        try {
   936→   936→          const res = await fetch(`/api/members/${memberId}`)
   937→   937→          if (!res.ok) throw new Error('Failed')
   938→   938→          const m: MemberDetail = await res.json()
   939→   939→          setForm({
   940→   940→            firstName: m.firstName,
   941→   941→            lastName: m.lastName,
   942→   942→            email: m.email ?? '',
   943→   943→            phone: m.phone ?? '',
   944→   944→            gender: m.gender ?? '',
   945→   945→            dateOfBirth: m.dateOfBirth ? new Date(m.dateOfBirth).toISOString().slice(0, 10) : '',
   946→   946→            status: m.status,
   947→   947→            branchId: m.branch?.id ?? '',
   948→   948→            baptized: m.baptized,
   949→   949→            baptizedDate: m.baptizedDate ? new Date(m.baptizedDate).toISOString().slice(0, 10) : '',
   950→   950→            address: m.address ?? '',
   951→   951→            emergencyName: m.emergencyName ?? '',
   952→   952→            emergencyPhone: m.emergencyPhone ?? '',
   953→   953→            notes: m.notes ?? '',
   954→   954→            privateNotes: m.privateNotes ?? '',
   955→   955→            memberId: m.memberId ?? '',
   956→   956→          })
   957→   957→        } catch {
   958→   958→          toast.error('Failed to load member for editing')
   959→   959→          onOpenChange(false)
   960→   960→        }
   961→   961→      })()
   962→   962→    } else {
   963→   963→      setForm(EMPTY_FORM)
   964→   964→    }
   965→   965→  }, [mode, memberId, open, onOpenChange])
   966→   966→
   967→   967→  const set = <K extends keyof FormState>(k: K, v: FormState[K]) =>
   968→   968→    setForm((f) => ({ ...f, [k]: v }))
   969→   969→
   970→   970→  const submit = async () => {
   971→   971→    if (!form.firstName.trim() || !form.lastName.trim()) {
   972→   972→      toast.error('First and last name are required')
   973→   973→      return
   974→   974→    }
   975→   975→    setSaving(true)
   976→   976→    try {
   977→   977→      const payload: Record<string, unknown> = {
   978→   978→        firstName: form.firstName.trim(),
   979→   979→        lastName: form.lastName.trim(),
   980→   980→        email: form.email.trim() || null,
   981→   981→        phone: form.phone.trim() || null,
   982→   982→        gender: form.gender || null,
   983→   983→        dateOfBirth: form.dateOfBirth || null,
   984→   984→        branchId: form.branchId || null,
   985→   985→        baptized: form.baptized,
   986→   986→        baptizedDate: form.baptizedDate || null,
   987→   987→        address: form.address.trim() || null,
   988→   988→        emergencyName: form.emergencyName.trim() || null,
   989→   989→        emergencyPhone: form.emergencyPhone.trim() || null,
   990→   990→        notes: form.notes.trim() || null,
   991→   991→        memberId: form.memberId.trim() || null,
   992→   992→      }
   993→   993→      if (canEditStatus) payload.status = form.status
   994→   994→      if (canEditStatus) payload.privateNotes = form.privateNotes.trim() || null
   995→   995→
   996→   996→      const url = mode === 'create' ? '/api/members' : `/api/members/${memberId}`
   997→   997→      const method = mode === 'create' ? 'POST' : 'PATCH'
   998→   998→      const res = await fetch(url, {
   999→   999→        method,
  1000→  1000→        headers: { 'Content-Type': 'application/json' },
  1001→  1001→        body: JSON.stringify(payload),
  1002→  1002→      })
  1003→  1003→      if (!res.ok) {
  1004→  1004→        const e = await res.json().catch(() => ({}))
  1005→  1005→        throw new Error(e.error || `Failed to ${mode} member`)
  1006→  1006→      }
  1007→  1007→      toast.success(mode === 'create' ? 'Member added' : 'Member updated')
  1008→  1008→      onSaved()
  1009→  1009→    } catch (e: unknown) {

  1011→  1011→      toast.error(msg)
  1012→  1012→    } finally {
  1013→  1013→      setSaving(false)
  1014→  1014→    }
  1015→  1015→  }
  1016→  1016→
  1017→  1017→  return (

  1020→  1020→        <DialogHeader>

  1026→  1026→          </DialogDescription>
  1027→  1027→        </DialogHeader>
  1028→  1028→
  1029→  1029→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1030→  1030→          <Field label="First name *">
  1031→  1031→            <Input value={form.firstName} onChange={(e) => set('firstName', e.target.value)} />
