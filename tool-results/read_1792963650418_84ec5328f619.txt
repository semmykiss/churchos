   506→// Posted journal entry (header) — every posted transaction has balanced debits = credits
   507→model JournalEntry {
   508→  id             String   @id @default(cuid())
   509→  organizationId String
   510→  reference      String   @unique // unique traceable reference (e.g. "JE-2026-00001")
   511→  transactionDate DateTime
   512→  description    String
   513→  source         String   // manual, collection, expense, fund_transfer, bank_recon, adjustment, reversal
   514→  sourceRefId    String?  // ID of the source record (collectionSession, expenseRequest, etc.)
   515→  status         String   @default("posted") // posted, reversed
   516→  reversedByRef  String?  // reference of the reversing entry (if reversed)
   517→  reversalReason String?
   518→  periodLocked   Boolean  @default(false) // true if the accounting period was closed before this (defensive)
   519→  postedById     String?
   520→  postedByName   String?
   521→  createdAt      DateTime @default(now())
   522→  organization   Organization @relation(fields: [organizationId], references: [id], onDelete: Cascade)
   523→  ledgerEntries  LedgerEntry[]
   524→  @@index([organizationId])
   525→  @@index([transactionDate])
   526→  @@index([reference])
   527→  @@index([source])
   528→}
