   388→      {/* Edit dialog */}

   398→            queryClient.invalidateQueries({ queryKey: ['dashboard'] })

   400→        />
   401→      )}
   402→
   403→      {/* Delete confirmation */}
   404→      <AlertDialog open={!!deleteId} onOpenChange={(o) => { if (!o) setDeleteId(null) }}>
   405→        <AlertDialogContent>
   406→          <AlertDialogHeader>
   407→            <AlertDialogTitle>Delete announcement?</AlertDialogTitle>
   408→            <AlertDialogDescription>
   409→              This will permanently remove &quot;{deleteTitle}&quot; and its in-app notifications.

   418→              Delete
   419→            </AlertDialogAction>
   420→          </AlertDialogFooter>
   421→        </AlertDialogContent>

   423→    </div>

   425→}

   427→// ---------- announcement card ----------
   428→function AnnouncementCard({
   429→  a, canUpdate, canDelete, onEdit, onDelete,
   430→}: {
   431→  a: Announcement
   432→  canUpdate: boolean
   433→  canDelete: boolean
   434→  onEdit: () => void
   435→  onDelete: () => void
   436→}) {
   437→  return (
   438→    <Card className={`border-border/60 rounded-xl hover:shadow-premium-md hover:-translate-y-0.5 transition-all ${a.pinned ? 'ring-1 ring-inset ring-amber-500/20' : ''}`}>
   439→      <CardContent className="p-5">
   440→        <div className="flex items-start justify-between gap-3 mb-2">
   441→          <div className="flex items-center gap-2 flex-wrap min-w-0">
   442→            {a.pinned && (
   443→              <Badge variant="outline" className="gap-1 bg-amber-500/10 text-amber-700 border-amber-500/20 dark:text-amber-300 text-[10px] font-medium">
   444→                <Pin className="size-3" /> Pinned
   445→              </Badge>
   446→            )}
   447→            <Badge variant="outline" className={`text-[10px] font-medium ${AUDIENCE_BADGE_STYLES[a.audience] ?? ''}`}>
   448→              {a.audience === 'all' && <Globe className="size-3 mr-1" />}
   449→              {titleCase(a.audience)}
   450→            </Badge>
   451→            {!a.published && (
   452→              <Badge variant="outline" className="text-[10px] font-medium bg-muted text-muted-foreground">Draft</Badge>
   453→            )}
   454→          </div>
   455→          {(canUpdate || canDelete) && (
   456→            <DropdownMenu>
   457→              <DropdownMenuTrigger asChild>
   458→                <Button variant="ghost" size="icon" className="size-8 shrink-0" aria-label="Announcement actions">
   459→                  <MoreHorizontal className="size-4" />
   460→                </Button>
   461→              </DropdownMenuTrigger>
   462→              <DropdownMenuContent align="end">
   463→                {canUpdate && (
   464→                  <DropdownMenuItem onClick={onEdit}>
   465→                    <Pencil className="size-3.5 mr-2" /> Edit
   466→                  </DropdownMenuItem>
   467→                )}
   468→                {canDelete && (
   469→                  <>
   470→                    {canUpdate && <DropdownMenuSeparator />}
   471→                    <DropdownMenuItem onClick={onDelete} className="text-destructive">
   472→                      <Trash2 className="size-3.5 mr-2" /> Delete
   473→                    </DropdownMenuItem>
   474→                  </>
   475→                )}
   476→              </DropdownMenuContent>
   477→            </DropdownMenu>
   478→          )}
   479→        </div>
   480→        <h3 className="font-semibold text-base leading-snug mb-1.5">{a.title}</h3>
   481→        <p className="text-sm text-muted-foreground line-clamp-3 whitespace-pre-wrap">{a.body}</p>
   482→        <div className="flex items-center gap-3 mt-4 pt-3 border-t border-border/60 text-xs text-muted-foreground">
   483→          <span className="inline-flex items-center gap-1 nums">
   484→            <Send className="size-3" /> {a.published ? formatDate(a.publishedAt) : 'Draft'}
   485→          </span>
   486→          {a.notificationsQueued !== undefined && a.notificationsQueued > 0 && (
   487→            <span className="inline-flex items-center gap-1 text-emerald-700 dark:text-emerald-400 nums">
   488→              <Bell className="size-3" /> {a.notificationsQueued} notified
   489→            </span>
   490→          )}
   491→          <span className="ml-auto nums">{relativeTime(a.publishedAt || a.createdAt)}</span>
   492→        </div>
   493→      </CardContent>
   494→    </Card>
   495→  )
   496→}

   498→// ---------- form dialog (create/edit) ----------
   499→function AnnouncementFormDialog({
   500→  open, onOpenChange, editId, onSaved,
   501→}: {
   502→  open: boolean
   503→  onOpenChange: (v: boolean) => void
   504→  editId?: string
   505→  onSaved: () => void
   506→}) {
   507→  const [title, setTitle] = useState('')
   508→  const [body, setBody] = useState('')
   509→  const [audience, setAudience] = useState('all')
   510→  const [departmentId, setDepartmentId] = useState('')
   511→  const [pinned, setPinned] = useState(false)
   512→  const [published, setPublished] = useState(true)
   513→  const [departments, setDepartments] = useState<DepartmentRef[]>([])
   514→  const [loading, setLoading] = useState(false)
   515→
   516→  // Load existing announcement when editing
   517→  useQuery({
   518→    queryKey: ['announcement', editId] as const,
   519→    queryFn: async () => {
   520→      if (!editId) return null
   521→      const res = await fetch(`/api/announcements/${editId}`)
   522→      if (!res.ok) {
   523→        const e = await res.json().catch(() => ({}))
   524→        throw new Error(e.error || 'Failed to load')
   525→      }
   526→      const data: Announcement = await res.json()
   527→      setTitle(data.title)
   528→      setBody(data.body)
   529→      setAudience(data.audience)
   530→      setDepartmentId(data.departmentId ?? '')
   531→      setPinned(data.pinned)
   532→      setPublished(data.published)
   533→      return data
   534→    },
   535→    enabled: !!editId && open,
   536→  })
   537→
   538→  // Load departments list (for audience=department)
   539→  useQuery({
   540→    queryKey: ['departments', 'list', 'short'] as const,
   541→    queryFn: async () => {
   542→      const res = await fetch('/api/departments?pageSize=100')
   543→      if (!res.ok) return []
   544→      const data = await res.json()
   545→      const list = (data.items ?? []).map((d: any) => ({ id: d.id, name: d.name }))
   546→      setDepartments(list)
   547→      return list
   548→    },
   549→    enabled: open,
   550→  })
   551→
   552→  // Reset when closed
   553→  useEffect(() => {
   554→    if (!open && !editId) {
   555→      setTitle('')
   556→      setBody('')
   557→      setAudience('all')
   558→      setDepartmentId('')
   559→      setPinned(false)
   560→      setPublished(true)
   561→    }
   562→  }, [open, editId])
   563→
   564→  const isEdit = !!editId
   565→
   566→  const save = async () => {
   567→    if (!title.trim()) {
   568→      toast.error('Title is required')
   569→      return
   570→    }
   571→    if (!body.trim()) {
   572→      toast.error('Message body is required')
   573→      return
   574→    }
   575→    if (audience === 'department' && !departmentId) {
   576→      toast.error('Please select a department')
   577→      return
   578→    }
   579→    setLoading(true)
   580→    try {
   581→      const payload: any = {
   582→        title: title.trim(),
   583→        body: body.trim(),
   584→        audience,
   585→        pinned,
   586→        published,
   587→      }
   588→      if (audience === 'department') payload.departmentId = departmentId
   589→      const url = isEdit ? `/api/announcements/${editId}` : '/api/announcements'
   590→      const method = isEdit ? 'PATCH' : 'POST'
   591→      const res = await fetch(url, {
   592→        method,
   593→        headers: { 'Content-Type': 'application/json' },
   594→        body: JSON.stringify(payload),
   595→      })
   596→      if (!res.ok) {
   597→        const e = await res.json().catch(() => ({}))
   598→        throw new Error(e.error || 'Save failed')
   599→      }
   600→      const data = await res.json()
   601→      const queued = data.notificationsQueued ?? 0
   602→      toast.success(
   603→        isEdit
   604→          ? 'Announcement updated'
   605→          : `Announcement created${queued > 0 ? ` — ${queued} notification${queued === 1 ? '' : 's'} queued` : ''}`,
   606→      )
   607→      onSaved()
   608→    } catch (e: unknown) {
   609→      const msg = e instanceof Error ? e.message : 'Save failed'
   610→      toast.error(msg)
   611→    } finally {
   612→      setLoading(false)
   613→    }
   614→  }
   615→
   616→  return (
   617→    <Dialog open={open} onOpenChange={onOpenChange}>
   618→      <DialogContent className="sm:max-w-[600px] max-h-[90vh] overflow-y-auto scroll-thin">
   619→        <DialogHeader>
   620→          <DialogTitle>{isEdit ? 'Edit Announcement' : 'New Announcement'}</DialogTitle>
   621→          <DialogDescription>
   622→            Compose a church announcement. In-app notifications will be queued for the selected audience.
   623→          </DialogDescription>
   624→        </DialogHeader>
   625→        <div className="space-y-4 py-2">
   626→          <div>
   627→            <Label htmlFor="ann-title" className="text-sm font-medium">Title <span className="text-destructive">*</span></Label>
   628→            <Input
   629→              id="ann-title"
   630→              value={title}
   631→              onChange={(e) => setTitle(e.target.value)}
   632→              placeholder="e.g. Sunday Service Time Change"
   633→              className="mt-1.5 h-9"
   634→              maxLength={140}
   635→            />
   636→          </div>
   637→          <div>
   638→            <Label htmlFor="ann-body" className="text-sm font-medium">Message <span className="text-destructive">*</span></Label>
   639→            <Textarea
   640→              id="ann-body"
   641→              value={body}
   642→              onChange={(e) => setBody(e.target.value)}
   643→              placeholder="Write the announcement message…"
   644→              rows={5}
   645→              className="mt-1.5"
   646→              maxLength={4000}
   647→            />
   648→            <p className="text-xs text-muted-foreground mt-1 nums">{body.length}/4000 characters</p>
   649→          </div>
   650→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   651→            <div>
   652→              <Label htmlFor="ann-audience" className="text-sm font-medium">Audience</Label>
   653→              <Select value={audience} onValueChange={setAudience}>
   654→                <SelectTrigger id="ann-audience" className="mt-1.5 h-9">
   655→                  <SelectValue />
   656→                </SelectTrigger>
   657→                <SelectContent>
   658→                  {AUDIENCE_OPTIONS.map((o) => (
   659→                    <SelectItem key={o.value} value={o.value}>
   660→                      {o.label}
   661→                    </SelectItem>
   662→                  ))}
   663→                </SelectContent>
   664→              </Select>
   665→            </div>
   666→            {audience === 'department' && (
   667→              <div>
   668→                <Label htmlFor="ann-dept" className="text-sm font-medium">Department</Label>
   669→                <Select value={departmentId} onValueChange={setDepartmentId}>
   670→                  <SelectTrigger id="ann-dept" className="mt-1.5 h-9">
   671→                    <SelectValue placeholder="Select department" />
   672→                  </SelectTrigger>
   673→                  <SelectContent>
   674→                    {departments.map((d) => (
   675→                      <SelectItem key={d.id} value={d.id}>{d.name}</SelectItem>
   676→                    ))}
   677→                  </SelectContent>
   678→                </Select>
   679→              </div>
   680→            )}
   681→          </div>
   682→          <div className="grid grid-cols-2 gap-4">
   683→            <div className="flex items-center justify-between rounded-xl border border-border/60 p-3">
   684→              <div>
   685→                <Label htmlFor="ann-pinned" className="font-medium text-sm">Pin to top</Label>
   686→                <p className="text-xs text-muted-foreground">Highlight on the announcements list</p>
   687→              </div>
   688→              <Switch id="ann-pinned" checked={pinned} onCheckedChange={setPinned} />
   689→            </div>
   690→            <div className="flex items-center justify-between rounded-xl border border-border/60 p-3">
   691→              <div>
   692→                <Label htmlFor="ann-published" className="font-medium text-sm">Published</Label>
   693→                <p className="text-xs text-muted-foreground">Publish immediately to audience</p>
   694→              </div>
   695→              <Switch id="ann-published" checked={published} onCheckedChange={setPublished} />
   696→            </div>
   697→          </div>
   698→          <div className="rounded-lg bg-muted/40 border border-border/60 p-3 text-xs text-muted-foreground flex items-start gap-2">
   699→            <Mail className="size-3.5 mt-0.5 shrink-0" />
   700→            <div>
   701→              <span className="font-medium text-foreground">Provider status:</span>{' '}
   702→              Email, SMS and WhatsApp adapters are not configured — only in-app notifications will be queued.
   703→            </div>
   704→          </div>
   705→        </div>
   706→        <DialogFooter>
   707→          <DialogClose asChild>
   708→            <Button variant="outline">Cancel</Button>
   709→          </DialogClose>
   710→          <Button onClick={save} disabled={loading} className="gap-2">
   711→            {loading ? 'Saving…' : isEdit ? 'Save changes' : 'Create & queue'}
   712→          </Button>
   713→        </DialogFooter>
   714→      </DialogContent>
   715→    </Dialog>
   716→  )

   720→function NotificationsPanel() {
   721→  const queryClient = useQueryClient()
   722→  const [onlyUnread, setOnlyUnread] = useState(false)
   723→
   724→  const { data, isLoading, isError, refetch } = useQuery<NotificationListResponse>({
   725→    queryKey: ['notifications', 'list', onlyUnread] as const,
   726→    queryFn: async () => {
   727→      const params = new URLSearchParams({ pageSize: '50' })
   728→      if (onlyUnread) params.set('unread', '1')
   729→      const res = await fetch(`/api/notifications?${params}`)
   730→      if (!res.ok) {
   731→        const e = await res.json().catch(() => ({}))
   732→        throw new Error(e.error || 'Failed to load notifications')
   733→      }
   734→      return res.json()
   735→    },
   736→  })
   737→
   738→  const markAllRead = useMutation({
   739→    mutationFn: async () => {
   740→      const res = await fetch('/api/notifications', {
   741→        method: 'PATCH',
   742→        headers: { 'Content-Type': 'application/json' },
   743→        body: JSON.stringify({ all: true }),
   744→      })
   745→      if (!res.ok) {
   746→        const e = await res.json().catch(() => ({}))
   747→        throw new Error(e.error || 'Failed')
   748→      }
   749→      return res.json()
   750→    },
   751→    onSuccess: (data) => {
   752→      toast.success(`Marked ${data.updated ?? 0} notifications as read`)
   753→      queryClient.invalidateQueries({ queryKey: ['notifications'] })
   754→    },
   755→    onError: (e: Error) => toast.error(e.message),
   756→  })
   757→
   758→  const markOneRead = useMutation({
   759→    mutationFn: async (id: string) => {
   760→      const res = await fetch('/api/notifications', {
   761→        method: 'PATCH',
   762→        headers: { 'Content-Type': 'application/json' },
   763→        body: JSON.stringify({ id }),
   764→      })
   765→      if (!res.ok) {
   766→        const e = await res.json().catch(() => ({}))
   767→        throw new Error(e.error || 'Failed')
   768→      }
   769→      return res.json()
   770→    },
   771→    onSuccess: () => {
   772→      queryClient.invalidateQueries({ queryKey: ['notifications'] })
   773→    },
   774→    onError: (e: Error) => toast.error(e.message),
   775→  })
   776→
   777→  const unread = data?.unread ?? 0
   778→
   779→  return (
   780→    <Card className="border-border/60">
   781→      <CardHeader className="pb-3">
   782→        <div className="flex items-center justify-between gap-3">
   783→          <div>
   784→            <CardTitle className="flex items-center gap-2 text-base">
   785→              <Bell className="size-4" /> Your Notifications
   786→              {unread > 0 && (
   787→                <Badge className="bg-primary text-primary-foreground nums">{unread} unread</Badge>
   788→              )}
   789→            </CardTitle>
   790→            <CardDescription className="text-xs">
   791→              In-app notifications sent to you from announcements and other modules.
   792→            </CardDescription>
   793→          </div>
   794→          <div className="flex items-center gap-2">
   795→            <Button
   796→              size="sm"
   797→              variant={onlyUnread ? 'default' : 'outline'}
   798→              onClick={() => setOnlyUnread((v) => !v)}
   799→            >
   800→              {onlyUnread ? 'Showing unread' : 'Unread only'}
   801→            </Button>
   802→            {unread > 0 && (
   803→              <Button
   804→                size="sm"
   805→                variant="outline"
   806→                onClick={() => markAllRead.mutate()}
   807→                disabled={markAllRead.isPending}
   808→                className="gap-2"
   809→              >
   810→                <CheckCheck className="size-3.5" /> Mark all read
   811→              </Button>
   812→            )}
   813→          </div>
   814→        </div>
   815→      </CardHeader>
   816→      <CardContent>
   817→        {isLoading ? (
   818→          <div className="space-y-3">
   819→            {Array.from({ length: 4 }).map((_, i) => (
   820→              <Skeleton key={i} className="h-16 w-full rounded-lg" />
   821→            ))}
   822→          </div>
   823→        ) : isError ? (
   824→          <EmptyState
   825→            icon={AlertTriangle}
   826→            title="Failed to load notifications"
   827→            description="Please try again."
   828→            action={
   829→              <Button variant="outline" size="sm" onClick={() => refetch()} className="gap-2">
   830→                <RefreshCw className="size-3.5" /> Retry
   831→              </Button>
   832→            }
   833→          />
   834→        ) : !data || data.items.length === 0 ? (
   835→          <EmptyState
   836→            icon={Bell}
   837→            title={onlyUnread ? 'No unread notifications' : 'No notifications yet'}
   838→            description={onlyUnread ? 'You are all caught up.' : 'New announcements and activity will appear here.'}
   839→          />
   840→        ) : (
   841→          <ul className="divide-y divide-border/60 max-h-[600px] overflow-y-auto scroll-thin -mx-2">
   842→            {data.items.map((n) => (
   843→              <li
   844→                key={n.id}
   845→                className={`flex items-start gap-3 px-2 py-3 transition-colors relative ${n.read ? 'opacity-70' : 'bg-primary/5'}`}
   846→              >
   847→                {!n.read && <span className="absolute left-0 top-0 bottom-0 w-0.5 bg-primary" aria-hidden />}
   848→                <div className={`size-2 rounded-full mt-2 shrink-0 ${n.read ? 'bg-muted-foreground/40' : 'bg-primary'}`} />
   849→                <div className="flex-1 min-w-0">
   850→                  <p className="text-sm font-medium leading-snug">{n.title}</p>
   851→                  <p className="text-sm text-muted-foreground mt-0.5">{n.message}</p>

   853→                </div>
   854→                <div className="flex items-center gap-1 shrink-0">
   855→                  {!n.read && (
   856→                    <Button
   857→                      size="sm"
   858→                      variant="ghost"
   859→                      className="size-8 p-0"
   860→                      onClick={() => markOneRead.mutate(n.id)}
   861→                      disabled={markOneRead.isPending}
   862→                      aria-label="Mark as read"
   863→                    >
   864→                      <CheckCheck className="size-3.5" />
   865→                    </Button>
   866→                  )}
   867→                </div>
   868→              </li>
   869→            ))}
   870→          </ul>
   871→        )}
   872→      </CardContent>
   873→    </Card>
   874→  )
