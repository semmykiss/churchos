   107→// ---------- debounced search hook ----------
   108→function useDebouncedValue<T>(value: T, delay = 350): T {
   109→  const [debounced, setDebounced] = useState(value)
   110→  useEffect(() => {
   111→    const t = setTimeout(() => setDebounced(value), delay)
   112→    return () => clearTimeout(t)
   113→  }, [value, delay])
   114→  return debounced
   115→}
