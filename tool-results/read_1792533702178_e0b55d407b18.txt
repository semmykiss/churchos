   255→function PrayerRequestFormCard({ enabled }: { enabled: boolean }) {
   256→  const queryClient = useQueryClient()
   257→  const [request, setRequest] = useState('')
   258→  const [isPrivate, setIsPrivate] = useState(true)
   259→  const [loading, setLoading] = useState(false)
   260→
   261→  const submit = async () => {
   262→    if (!request.trim()) {
   263→      toast.error('Please write your prayer request first')
   264→      return
   265→    }
   266→    setLoading(true)
   267→    try {
   268→      const res = await fetch('/api/pastoral/prayer', {
   269→        method: 'POST',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({ request: request.trim(), private: isPrivate }),
   272→      })
   273→      if (!res.ok) {
   274→        const e = await res.json().catch(() => ({}))
   275→        throw new Error(e.error || 'Submit failed')
   276→      }
   277→      toast.success('Prayer request submitted — our team is praying with you.')
   278→      setRequest('')
   279→      queryClient.invalidateQueries({ queryKey: ['pastoral', 'prayer'] })
   280→    } catch (e: unknown) {
   281→      toast.error(e instanceof Error ? e.message : 'Submit failed')
   282→    } finally {
   283→      setLoading(false)
   284→    }
   285→  }
   286→
   287→  return (
   288→    <Card className={enabled ? 'border-border/60' : 'opacity-60'}>
   289→      <CardHeader>
   290→        <CardTitle className="flex items-center gap-2 text-base">
   291→          <MessageCircle className="size-4 text-primary" /> Submit a Prayer Request
   292→        </CardTitle>
   293→        <CardDescription className="text-xs">
   294→          Share what you would like the pastoral team to pray about.
   295→        </CardDescription>
   296→      </CardHeader>
   297→      <CardContent className="space-y-3">
   298→        <Textarea
   299→          value={request}
   300→          onChange={(e) => setRequest(e.target.value)}
   301→          placeholder="e.g. Please pray for healing for my mother…"
   302→          rows={5}
   303→          maxLength={2000}
   304→        />
   305→        <div className="flex items-center justify-between rounded-lg border border-border/60 p-3">
   306→          <div>
   307→            <Label htmlFor="pr-private" className="font-medium text-sm">Keep private</Label>
   308→            <p className="text-xs text-muted-foreground">Only pastors will see your name</p>
   309→          </div>
   310→          <Switch id="pr-private" checked={isPrivate} onCheckedChange={setIsPrivate} />
   311→        </div>
   312→        <Button onClick={submit} disabled={!enabled || loading || !request.trim()} className="w-full">
   313→          {loading ? 'Submitting…' : 'Submit Prayer Request'}
   314→        </Button>
   315→      </CardContent>
   316→    </Card>
   317→  )
   318→}

   408→// ---------- cases tab ----------
   409→function CasesTab({
   410→  canCreate, canUpdate, canDelete, canSeePrivate,
   411→}: {
   412→  canCreate: boolean
   413→  canUpdate: boolean
   414→  canDelete: boolean
   415→  canSeePrivate: boolean
   416→}) {
   417→  const queryClient = useQueryClient()
   418→  const [typeFilter, setTypeFilter] = useState('')
   419→  const [statusFilter, setStatusFilter] = useState('')
   420→  const [priorityFilter, setPriorityFilter] = useState('')
   421→  const [search, setSearch] = useState('')
   422→
   423→  // Detail dialog
   424→  const [detailId, setDetailId] = useState<string | null>(null)
   425→  const [detailOpen, setDetailOpen] = useState(false)
   426→  // New case dialog
   427→  const [addOpen, setAddOpen] = useState(false)
   428→  // Delete
   429→  const [deleteId, setDeleteId] = useState<string | null>(null)
   430→
   431→  const queryKey = useMemo(
   432→    () => ['pastoral', 'cases', typeFilter, statusFilter, priorityFilter, search] as const,
   433→    [typeFilter, statusFilter, priorityFilter, search],
   434→  )
   435→
   436→  const { data, isLoading, isError, refetch } = useQuery<PastoralListResponse>({
   437→    queryKey,
   438→    queryFn: async () => {
   439→      const params = new URLSearchParams({ pageSize: '100' })
   440→      if (typeFilter) params.set('type', typeFilter)
   441→      if (statusFilter) params.set('status', statusFilter)
   442→      if (priorityFilter) params.set('priority', priorityFilter)
   443→      if (search) params.set('q', search)
   444→      const res = await fetch(`/api/pastoral?${params}`)
   445→      if (!res.ok) {
   446→        const e = await res.json().catch(() => ({}))
   447→        throw new Error(e.error || 'Failed to load cases')
   448→      }
   449→      return res.json()
   450→    },
   451→  })
   452→
   453→  const deleteMutation = useMutation({
   454→    mutationFn: async (id: string) => {
   455→      const res = await fetch(`/api/pastoral/${id}`, { method: 'DELETE' })
   456→      if (!res.ok) {
   457→        const e = await res.json().catch(() => ({}))
   458→        throw new Error(e.error || 'Delete failed')
   459→      }
   460→      return res.json()
   461→    },
   462→    onSuccess: () => {
   463→      toast.success('Case deleted')
   464→      queryClient.invalidateQueries({ queryKey: ['pastoral'] })
   465→      setDeleteId(null)
   466→    },
   467→    onError: (e: Error) => toast.error(e.message),
   468→  })
   469→
   470→  const openDetail = useCallback((id: string) => {
   471→    setDetailId(id)
   472→    setDetailOpen(true)
   473→  }, [])
   474→
   475→  const stats = data?.stats ?? { total: 0, open: 0, inProgress: 0, resolved: 0, urgent: 0 }
   476→
   477→  return (
   478→    <div className="space-y-4">
   479→      {/* Stats */}
   480→      <div className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   481→        <StatMini icon={HeartHandshake} label="Total" value={stats.total} accent="bg-primary/10 text-primary" />
   482→        <StatMini icon={Clock} label="Open" value={stats.open} accent="bg-rose-500/10 text-rose-600" />
   483→        <StatMini icon={RefreshCw} label="In Progress" value={stats.inProgress} accent="bg-amber-500/10 text-amber-600" />
   484→        <StatMini icon={UserCheck} label="Resolved" value={stats.resolved} accent="bg-emerald-500/10 text-emerald-600" />
   485→        <StatMini icon={Flame} label="Urgent" value={stats.urgent} accent="bg-orange-500/10 text-orange-600" />
   486→      </div>
   487→
   488→      {/* Filters */}
   489→      <Card className="border-border/60">
   490→        <CardContent className="p-4">
   491→          <div className="flex flex-col md:flex-row md:items-center gap-3">
   492→            <div className="relative flex-1">
   493→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground pointer-events-none" />
   494→              <Input
   495→                value={search}
   496→                onChange={(e) => setSearch(e.target.value)}
   497→                placeholder="Search by member name or description…"
   498→                className="pl-9"
   499→                aria-label="Search cases"
   500→              />
   501→              {search && (
   502→                <button
   503→                  type="button"
   504→                  onClick={() => setSearch('')}
   505→                  className="absolute right-2 top-1/2 -translate-y-1/2 size-6 grid place-items-center rounded-md hover:bg-accent text-muted-foreground"
   506→                  aria-label="Clear search"
   507→                >
   508→                  <X className="size-3.5" />
   509→                </button>
   510→              )}
   511→            </div>
   512→            <Select value={typeFilter} onValueChange={(v) => setTypeFilter(v === 'all' ? '' : v)}>
   513→              <SelectTrigger className="w-full md:w-40" aria-label="Filter by type">
   514→                <SelectValue placeholder="All types" />
   515→              </SelectTrigger>
   516→              <SelectContent>
   517→                <SelectItem value="all">All types</SelectItem>
   518→                {CASE_TYPES.map((t) => (
   519→                  <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>
   520→                ))}
   521→              </SelectContent>
   522→            </Select>
   523→            <Select value={statusFilter} onValueChange={(v) => setStatusFilter(v === 'all' ? '' : v)}>
   524→              <SelectTrigger className="w-full md:w-40" aria-label="Filter by status">
   525→                <SelectValue placeholder="All statuses" />
   526→              </SelectTrigger>
