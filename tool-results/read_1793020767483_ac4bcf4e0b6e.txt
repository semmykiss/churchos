   102→   102→function FileIcon({ filename, className }: { filename?: string | null; className?: string }) {
   103→   103→  const ext = (filename ?? '').split('.').pop()?.toLowerCase() ?? ''
   104→   104→  if (['jpg', 'jpeg', 'png', 'gif'].includes(ext)) return <FileImage className={className} />
   105→   105→  if (['xlsx', 'xls', 'csv'].includes(ext)) return <FileSpreadsheet className={className} />
   106→   106→  if (['pptx', 'ppt'].includes(ext)) return <Presentation className={className} />
   107→   107→  if (ext === 'txt') return <FileType className={className} />
   108→   108→  return <File className={className} />
   109→   109→}

   422→   422→            queryClient.invalidateQueries({ queryKey: ['documents'] })
   423→   423→          }}
   424→   424→        />
   425→   425→      )}
   426→   426→
   427→   427→      {/* Edit dialog */}
   428→   428→      {editOpen && editId && (
   429→   429→        <DocumentFormDialog
   430→   430→          open={editOpen}
   431→   431→          onOpenChange={setEditOpen}
   432→   432→          editId={editId}
   433→   433→          onSaved={() => {
   434→   434→            setEditOpen(false)
   435→   435→            setEditId(null)
   436→   436→            queryClient.invalidateQueries({ queryKey: ['documents'] })
   437→   437→          }}
   438→   438→        />
   439→   439→      )}
   440→   440→
   441→   441→      {/* Delete confirm */}
   442→   442→      <AlertDialog open={!!deleteId} onOpenChange={(o) => { if (!o) setDeleteId(null) }}>
   443→   443→        <AlertDialogContent>
   444→   444→          <AlertDialogHeader>
   445→   445→            <AlertDialogTitle>Delete document?</AlertDialogTitle>
   446→   446→            <AlertDialogDescription>
   447→   447→              This will permanently remove &quot;{deleteTitle}&quot; and its attached file. This cannot be undone.
   448→   448→            </AlertDialogDescription>
   449→   449→          </AlertDialogHeader>
   450→   450→          <AlertDialogFooter>
   451→   451→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   452→   452→            <AlertDialogAction
   453→   453→              onClick={() => deleteId && deleteMutation.mutate(deleteId)}
   454→   454→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   455→   455→            >
   456→   456→              Delete
   457→   457→            </AlertDialogAction>
   458→   458→          </AlertDialogFooter>
   459→   459→        </AlertDialogContent>
   460→   460→      </AlertDialog>
   461→   461→    </div>
   462→   462→  )
   463→   463→}

   465→   465→// ---------- form dialog ----------
   466→   466→const ALLOWED_EXTS = ['pdf', 'doc', 'docx', 'xlsx', 'pptx', 'txt', 'jpg', 'jpeg', 'png']
   467→   467→const MAX_FILE_MB = 10
   468→   468→
   469→   469→function DocumentFormDialog({
   470→   470→  open, onOpenChange, editId, onSaved,
   471→   471→}: {
   472→   472→  open: boolean
   473→   473→  onOpenChange: (v: boolean) => void
   474→   474→  editId?: string
   475→   475→  onSaved: () => void
   476→   476→}) {
   477→   477→  const [title, setTitle] = useState('')
   478→   478→  const [description, setDescription] = useState('')
   479→   479→  const [category, setCategory] = useState('policy')
   480→   480→  const [accessLevel, setAccessLevel] = useState('staff')
   481→   481→  const [file, setFile] = useState<File | null>(null)
   482→   482→  const [loading, setLoading] = useState(false)
   483→   483→
   484→   484→  useQuery({
   485→   485→    queryKey: ['document', editId] as const,
   486→   486→    queryFn: async () => {
   487→   487→      if (!editId) return null
   488→   488→      const res = await fetch(`/api/documents/${editId}`)
   489→   489→      if (!res.ok) {
   490→   490→        const e = await res.json().catch(() => ({}))
   491→   491→        throw new Error(e.error || 'Failed to load')
   492→   492→      }
   493→   493→      const d: DocumentItem = await res.json()
   494→   494→      setTitle(d.title)
   495→   495→      setDescription(d.description ?? '')
   496→   496→      setCategory(d.category)
   497→   497→      setAccessLevel(d.accessLevel)
   498→   498→      return d
   499→   499→    },
   500→   500→    enabled: !!editId && open,
   501→   501→  })
   502→   502→
   503→   503→  useEffect(() => {
   504→   504→    if (!open && !editId) {
   505→   505→      setTitle('')
   506→   506→      setDescription('')
   507→   507→      setCategory('policy')
   508→   508→      setAccessLevel('staff')
   509→   509→      setFile(null)
   510→   510→    }
   511→   511→  }, [open, editId])
   512→   512→
   513→   513→  const isEdit = !!editId
   514→   514→
   515→   515→  const validateFile = (f: File): string | null => {
   516→   516→    if (f.size > MAX_FILE_MB * 1024 * 1024) {
   517→   517→      return `File too large. Max ${MAX_FILE_MB}MB.`
   518→   518→    }
   519→   519→    const ext = (f.name.split('.').pop() ?? '').toLowerCase()
   520→   520→    if (!ALLOWED_EXTS.includes(ext)) {
   521→   521→      return `File type .${ext || '???'} not allowed. Allowed: ${ALLOWED_EXTS.join(', ')}`
   522→   522→    }
   523→   523→    return null
   524→   524→  }
   525→   525→
   526→   526→  const onFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   527→   527→    const f = e.target.files?.[0]
   528→   528→    if (!f) {
   529→   529→      setFile(null)
   530→   530→      return
   531→   531→    }
   532→   532→    const err = validateFile(f)
   533→   533→    if (err) {
   534→   534→      toast.error(err)
   535→   535→      e.target.value = ''
   536→   536→      setFile(null)
   537→   537→      return
   538→   538→    }
   539→   539→    setFile(f)
   540→   540→  }
   541→   541→
   542→   542→  const save = async () => {
   543→   543→    if (!title.trim()) {
   544→   544→      toast.error('Title is required')
   545→   545→      return
   546→   546→    }
   547→   547→    setLoading(true)
   548→   548→    try {
   549→   549→      if (isEdit) {
   550→   550→        // Edit: only metadata — file replace is not supported via this dialog (would need a separate replace flow)
   551→   551→        const res = await fetch(`/api/documents/${editId}`, {
   552→   552→          method: 'PATCH',
   553→   553→          headers: { 'Content-Type': 'application/json' },
   554→   554→          body: JSON.stringify({
   555→   555→            title: title.trim(),
   556→   556→            description: description.trim() || null,
   557→   557→            category,
   558→   558→            accessLevel,
   559→   559→          }),
   560→   560→        })
   561→   561→        if (!res.ok) {
   562→   562→          const e = await res.json().catch(() => ({}))
   563→   563→          throw new Error(e.error || 'Save failed')
   564→   564→        }
   565→   565→        toast.success('Document updated')
   566→   566→        onSaved()
   567→   567→        return
   568→   568→      }
   569→   569→      // Create with multipart
   570→   570→      const form = new FormData()
   571→   571→      form.set('title', title.trim())
   572→   572→      form.set('description', description.trim())
   573→   573→      form.set('category', category)
   574→   574→      form.set('accessLevel', accessLevel)
   575→   575→      if (file) form.set('file', file)
   576→   576→      const res = await fetch('/api/documents', { method: 'POST', body: form })
   577→   577→      if (!res.ok) {
   578→   578→        const e = await res.json().catch(() => ({}))
   579→   579→        throw new Error(e.error || 'Upload failed')
   580→   580→      }
   581→   581→      const data = await res.json()
   582→   582→      toast.success(file ? 'Document uploaded' : 'Document created')
   583→   583→      onSaved()
   584→   584→    } catch (e: unknown) {
   585→   585→      toast.error(e instanceof Error ? e.message : 'Save failed')
   586→   586→    } finally {
   587→   587→      setLoading(false)
   588→   588→    }
   589→   589→  }
   590→   590→
   591→   591→  return (
   592→   592→    <Dialog open={open} onOpenChange={onOpenChange}>
   593→   593→      <DialogContent className="sm:max-w-[600px] max-h-[90vh] overflow-y-auto scroll-thin">
   594→   594→        <DialogHeader>
   595→   595→          <DialogTitle>{isEdit ? 'Edit Document' : 'Upload Document'}</DialogTitle>
   596→   596→          <DialogDescription>
   597→   597→            {isEdit
   598→   598→              ? 'Update document metadata. File replace is not supported here.'
   599→   599→              : 'Upload a file or create metadata-only entry. Access levels enforce server-side visibility.'}
   600→   600→          </DialogDescription>
   601→   601→        </DialogHeader>
   602→   602→        <div className="space-y-4 py-2">
   603→   603→          <div>
   604→   604→            <Label htmlFor="doc-title" className="text-sm font-medium">Title <span className="text-destructive">*</span></Label>
   605→   605→            <Input
   606→   606→              id="doc-title"
   607→   607→              value={title}
   608→   608→              onChange={(e) => setTitle(e.target.value)}
   609→   609→              placeholder="e.g. Volunteer Code of Conduct"
   610→   610→              className="mt-1.5 h-9"
   611→   611→              maxLength={200}
   612→   612→            />
   613→   613→          </div>
   614→   614→          <div>
   615→   615→            <Label htmlFor="doc-desc" className="text-sm font-medium">Description</Label>
   616→   616→            <Textarea
   617→   617→              id="doc-desc"
   618→   618→              value={description}
   619→   619→              onChange={(e) => setDescription(e.target.value)}
   620→   620→              placeholder="Short summary of the document…"
   621→   621→              rows={3}
   622→   622→              className="mt-1.5"
   623→   623→              maxLength={1000}
   624→   624→            />
   625→   625→          </div>
   626→   626→          <div className="grid grid-cols-2 gap-3">
   627→   627→            <div>
   628→   628→              <Label htmlFor="doc-cat" className="text-sm font-medium">Category</Label>
   629→   629→              <Select value={category} onValueChange={setCategory}>
   630→   630→                <SelectTrigger id="doc-cat" className="mt-1.5 h-9"><SelectValue /></SelectTrigger>
   631→   631→                <SelectContent>
   632→   632→                  {CATEGORIES.filter((c) => c.value).map((c) => (
   633→   633→                    <SelectItem key={c.value} value={c.value}>{c.label}</SelectItem>
   634→   634→                  ))}
   635→   635→                </SelectContent>
   636→   636→              </Select>
   637→   637→            </div>
   638→   638→            <div>
   639→   639→              <Label htmlFor="doc-access" className="text-sm font-medium">Access level</Label>
   640→   640→              <Select value={accessLevel} onValueChange={setAccessLevel}>
   641→   641→                <SelectTrigger id="doc-access" className="mt-1.5 h-9"><SelectValue /></SelectTrigger>
   642→   642→                <SelectContent>
   643→   643→                  {ACCESS_LEVELS.filter((a) => a.value).map((a) => (
   644→   644→                    <SelectItem key={a.value} value={a.value}>{a.label}</SelectItem>
   645→   645→                  ))}
   646→   646→                </SelectContent>
   647→   647→              </Select>
   648→   648→            </div>
   649→   649→          </div>
   650→   650→          {!isEdit && (
   651→   651→            <div>
   652→   652→              <Label htmlFor="doc-file" className="text-sm font-medium">File (optional)</Label>
   653→   653→              <div className="mt-1.5 border-2 border-dashed border-border/60 rounded-xl p-5 text-center hover:border-primary/40 transition-colors">
   654→   654→                <input
   655→   655→                  id="doc-file"
   656→   656→                  type="file"
   657→   657→                  onChange={onFileChange}
   658→   658→                  accept=".pdf,.doc,.docx,.xlsx,.pptx,.txt,.jpg,.jpeg,.png"
   659→   659→                  className="hidden"
   660→   660→                />
   661→   661→                <Label htmlFor="doc-file" className="cursor-pointer inline-flex flex-col items-center">
   662→   662→                  <Upload className="size-6 text-muted-foreground mb-2" />
   663→   663→                  <span className="text-sm font-medium">{file ? file.name : 'Click to choose a file'}</span>
   664→   664→                  <span className="text-xs text-muted-foreground mt-1">
   665→   665→                    Max {MAX_FILE_MB}MB · {ALLOWED_EXTS.join(', ')}
   666→   666→                  </span>
   667→   667→                </Label>
   668→   668→              </div>
   669→   669→              {file && (
   670→   670→                <div className="mt-2 flex items-center justify-between rounded-lg bg-muted/40 px-3 py-2 text-xs">

   676→   676→                    className="text-muted-foreground hover:text-destructive"

   685→   685→          <div className="rounded-lg bg-muted/40 border border-border/60 p-3 text-xs text-muted-foreground flex items-start gap-2">

   689→   689→              public (everyone), members (member+), staff (church_admin/pastor/dept_head+), leaders (pastor/church_admin/dept_head), finance (finance_officer+), pastoral (pastor/super_admin).
   690→   690→            </div>
